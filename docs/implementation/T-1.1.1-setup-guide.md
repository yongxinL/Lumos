## Step 10: Create Source Files

### Create `src/main/index.ts` (Electron Main Process)

```typescript
import { app, BrowserWindow } from 'electron';
import path from 'path';

let mainWindow: BrowserWindow | null = null;

function createWindow() {
  mainWindow = new BrowserWindow({
    width: 1200,
    height: 800,
    webPreferences: {
      preload: path.join(__dirname, '../preload/index.js'),
      nodeIntegration: false,
      contextIsolation: true,
    },
  });

  // Load the renderer
  if (process.env.NODE_ENV === 'development') {
    mainWindow.loadURL('http://localhost:5173');
    mainWindow.webContents.openDevTools();
  } else {
    mainWindow.loadFile(path.join(__dirname, '../renderer/index.html'));
  }

  mainWindow.on('closed', () => {
    mainWindow = null;
  });
}

app.whenReady().then(() => {
  createWindow();

  app.on('activate', () => {
    if (BrowserWindow.getAllWindows().length === 0) {
      createWindow();
    }
  });
});

app.on('window-all-closed', () => {
  if (process.platform !== 'darwin') {
    app.quit();
  }
});
```

### Create `src/preload/index.ts` (Preload Script)

```typescript
import { contextBridge, ipcRenderer } from 'electron';

// Expose protected methods that allow the renderer process to use
// the ipcRenderer without exposing the entire object
contextBridge.exposeInMainWorld('electron', {
  // Add IPC methods here as needed
  send: (channel: string, data: unknown) => {
    ipcRenderer.send(channel, data);
  },
  receive: (channel: string, func: (...args: unknown[]) => void) => {
    ipcRenderer.on(channel, (_event, ...args) => func(...args));
  },
});
```

## Troubleshooting

### Issue: Electron window doesn't open

**Solution:**

```bash
# Check if Vite server is running
curl http://localhost:5173

# Check Electron logs
NODE_ENV=development electron . --enable-logging
```

### Issue: Type errors in preload script

**Solution:**
Ensure `@types/node` is installed:

```bash
pnpm add -D @types/node
```

### Issue: Vite HMR not working

**Solution:**
Check that `vite.config.ts` has correct server configuration and port is not in use:

```bash
lsof -i :5173
```

### Issue: Pre-commit hooks not running

**Solution:**
Reinstall hooks:

```bash
pnpm exec husky install
chmod +x .husky/pre-commit
```
