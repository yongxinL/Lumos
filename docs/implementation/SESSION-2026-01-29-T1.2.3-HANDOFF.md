### 2. Preload Script Enhancement

**File:** [src/preload/index.ts](../../src/preload/index.ts)

**Before:** Basic send/receive pattern with string channels
**After:** Fully type-safe API with TypeScript inference

```typescript
// Type-safe invoke with automatic type inference
async function invoke<C extends IPCChannel>(
  channel: C,
  payload: IPCChannelPayloads[C]
): Promise<IPCChannelResponses[C]>;

// Event subscription with cleanup
function on<E extends IPCEventType>(
  event: E,
  callback: (payload: IPCEventPayloads[E]) => void
): () => void; // Returns unsubscribe function
```

**Key Features:**

- Automatic error unwrapping and re-throwing
- Type inference from channel/event names
- Memory leak prevention via unsubscribe functions
- Backward compatibility with legacy API

## Patterns & Best Practices

### 1. Handler Implementation Pattern

```typescript
// src/main/ipc/domain.handlers.ts
import { registerHandler } from './handlers';
import { validatePayloadOrThrow } from './validation';

export function registerDomainHandlers(): void {
  registerHandler('domain:operation', async (payload) => {
    // Optional: validate payload
    validatePayloadOrThrow('domain:operation', payload);

    // Business logic
    const result = await service.doOperation(payload);

    // Return result (automatically wrapped in IPCResponse)
    return result;
  });
}
```

### 2. Event Emission Pattern

```typescript
// In main process service
import { emitEvent } from '../ipc/events';

// Emit event to renderer
emitEvent('domain:something-happened', {
  data: 'event payload',
});

// For high-frequency events, use throttling
const emitUpdate = createThrottledEmitter('transcription:update', 100);
emitUpdate({ partial: 'Hello...', confidence: 0.8 });
```

### 3. React Hook Usage Pattern

```typescript
// In renderer component
function MyComponent() {
  const { invoke } = useIPC();
  const [data, setData] = useState(null);

  // Subscribe to events
  useIPCEvent('domain:update', (payload) => {
    setData(payload);
  });

  // Invoke handlers
  const handleAction = async () => {
    const result = await invoke('domain:action', { param: 'value' });
    console.log(result);
  };

  return <div>...</div>;
}
```

### 4. Mutation Hook Pattern

```typescript
// For commands that change state
function MyComponent() {
  const { mutate, isLoading, error } = useIPCMutation('domain:create', {
    onSuccess: (data) => {
      console.log('Created:', data);
    },
    onError: (error) => {
      console.error('Failed:', error);
    },
  });

  const handleSubmit = (formData) => {
    mutate(formData);
  };

  return (
    <button onClick={handleSubmit} disabled={isLoading}>
      {isLoading ? 'Creating...' : 'Create'}
    </button>
  );
}
```
