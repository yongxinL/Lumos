### Skill Registry Implementation

```typescript
// src/main/services/governance/skill-registry.ts
import * as fs from 'fs/promises';
import * as path from 'path';
import * as yaml from 'js-yaml';
import * as crypto from 'crypto';
import Ajv from 'ajv';
import { SKILL_SCHEMA } from '../../types/governance/skill';
import { watch } from 'chokidar';

export class SkillRegistry {
  private skills: Map<string, Skill> = new Map();
  private operationIndex: Map<OperationType, Set<string>> = new Map();
  private validator: Ajv;
  private validateSkill: any;
  private skillsDirectory: string;
  private watcher?: any;

  constructor(skillsDirectory: string) {
    this.skillsDirectory = skillsDirectory;
    this.validator = new Ajv({ allErrors: true });
    this.validateSkill = this.validator.compile(SKILL_SCHEMA);
  }

  async initialize(): Promise<void> {
    // Ensure skills directory exists
    await fs.mkdir(this.skillsDirectory, { recursive: true });

    // Load all skill files
    await this.loadAllSkills();

    // Setup hot reload in development
    if (process.env.NODE_ENV === 'development') {
      this.setupHotReload();
    }

    console.log(`Loaded ${this.skills.size} skills`);
  }

  async loadSkill(filePath: string): Promise<Skill> {
    // Read YAML file
    const content = await fs.readFile(filePath, 'utf-8');
    const parsed = yaml.load(content);

    // Validate schema
    const valid = this.validateSkill(parsed);
    if (!valid) {
      const errors = this.validator.errorsText(this.validateSkill.errors);
      throw new SkillValidationError(`Invalid skill schema: ${errors}`);
    }

    // Compute version hash (immutable identifier)
    const versionHash = this.computeVersionHash(parsed);

    // Create skill object
    const skill: Skill = {
      ...parsed,
      version_hash: versionHash,
      created_at: parsed.created_at || new Date().toISOString(),
    };

    // Validate business rules
    this.validateBusinessRules(skill);

    return skill;
  }

  async registerSkill(skill: Skill): Promise<void> {
    // Check for conflicts
    const conflicts = this.detectConflicts(skill);
    if (conflicts.length > 0) {
      console.warn(`Skill ${skill.id} has conflicts with:`, conflicts);
    }

    // Register skill
    this.skills.set(skill.id, skill);

    // Update operation index
    for (const operation of skill.enabled_operations) {
      if (!this.operationIndex.has(operation)) {
        this.operationIndex.set(operation, new Set());
      }
      this.operationIndex.get(operation)!.add(skill.id);
    }

    console.log(`Registered skill: ${skill.name} (${skill.id})`);
  }

  async activateSkill(skillId: string, activatedBy: string): Promise<void> {
    const skill = this.skills.get(skillId);
    if (!skill) {
      throw new Error(`Skill ${skillId} not found`);
    }

    if (skill.status === 'active') {
      throw new Error(`Skill ${skillId} is already active`);
    }

    skill.status = 'active';
    skill.activated_at = new Date().toISOString();
    skill.activated_by = activatedBy;

    // Persist to database
    await this.persistSkill(skill);

    console.log(`Activated skill: ${skill.name}`);
  }

  async deactivateSkill(skillId: string): Promise<void> {
    const skill = this.skills.get(skillId);
    if (!skill) {
      throw new Error(`Skill ${skillId} not found`);
    }

    skill.status = 'deprecated';

    // Persist to database
    await this.persistSkill(skill);

    console.log(`Deactivated skill: ${skill.name}`);
  }

  getSkill(skillId: string): Skill | undefined {
    return this.skills.get(skillId);
  }

  getSkillByHash(versionHash: VersionHash): Skill | undefined {
    return Array.from(this.skills.values()).find((s) => s.version_hash === versionHash);
  }

  findSkillsByOperation(operation: OperationType): Skill[] {
    const skillIds = this.operationIndex.get(operation);
    if (!skillIds) return [];

    const skills = Array.from(skillIds)
      .map((id) => this.skills.get(id))
      .filter((s): s is Skill => s !== undefined && s.status === 'active');

    // Sort by priority (higher first)
    return skills.sort((a, b) => b.priority - a.priority);
  }

  getAllSkills(): Skill[] {
    return Array.from(this.skills.values());
  }

  getActiveSkills(): Skill[] {
    return Array.from(this.skills.values()).filter((s) => s.status === 'active');
  }

  private async loadAllSkills(): Promise<void> {
    try {
      const files = await fs.readdir(this.skillsDirectory);
      const yamlFiles = files.filter((f) => f.endsWith('.yaml') || f.endsWith('.yml'));

      for (const file of yamlFiles) {
        const filePath = path.join(this.skillsDirectory, file);
        try {
          const skill = await this.loadSkill(filePath);
          await this.registerSkill(skill);
        } catch (error) {
          console.error(`Failed to load skill ${file}:`, error);
        }
      }
    } catch (error) {
      console.error('Failed to load skills:', error);
    }
  }

  private computeVersionHash(skillData: any): VersionHash {
    // Create canonical JSON (sorted keys)
    const canonical = JSON.stringify(skillData, Object.keys(skillData).sort());

    // Compute SHA-256 hash
    const hash = crypto.createHash('sha256').update(canonical).digest('hex');

    return hash as VersionHash;
  }

  private validateBusinessRules(skill: Skill): void {
    // Priority must be reasonable
    if (skill.priority < 0 || skill.priority > 100) {
      throw new SkillValidationError('Priority must be between 0 and 100');
    }

    // Must have at least one enabled operation
    if (skill.enabled_operations.length === 0) {
      throw new SkillValidationError('Must have at least one enabled operation');
    }

    // Constraints must be valid TypeScript expressions
    for (const constraint of skill.policy_constraints) {
      if (!constraint.rule || constraint.rule.trim().length === 0) {
        throw new SkillValidationError('Constraint rule cannot be empty');
      }
    }
  }

  private detectConflicts(skill: Skill): string[] {
    const conflicts: string[] = [];

    for (const operation of skill.enabled_operations) {
      const existingSkills = this.findSkillsByOperation(operation);

      for (const existing of existingSkills) {
        // Check if same priority (potential conflict)
        if (existing.priority === skill.priority && existing.id !== skill.id) {
          conflicts.push(`${existing.id} (same priority for ${operation})`);
        }
      }
    }

    return conflicts;
  }

  private async persistSkill(skill: Skill): Promise<void> {
    // Persist to database (implementation depends on storage layer)
    // This is a placeholder - actual implementation would use storage service
  }

  private setupHotReload(): void {
    this.watcher = watch(this.skillsDirectory, {
      ignored: /(^|[\/\\])\../, // Ignore dotfiles
      persistent: true,
    });

    this.watcher.on('change', async (filePath: string) => {
      console.log(`Skill file changed: ${filePath}`);
      try {
        const skill = await this.loadSkill(filePath);
        await this.registerSkill(skill);
        console.log(`Reloaded skill: ${skill.name}`);
      } catch (error) {
        console.error(`Failed to reload skill:`, error);
      }
    });

    this.watcher.on('add', async (filePath: string) => {
      console.log(`New skill file added: ${filePath}`);
      try {
        const skill = await this.loadSkill(filePath);
        await this.registerSkill(skill);
      } catch (error) {
        console.error(`Failed to load new skill:`, error);
      }
    });
  }

  async shutdown(): Promise<void> {
    if (this.watcher) {
      await this.watcher.close();
    }
  }
}

// Errors
export class SkillValidationError extends Error {
  constructor(message: string) {
    super(message);
    this.name = 'SkillValidationError';
  }
}
```

### Example Skill File

```yaml
# skills/create-ticket.yaml
id: 'skill-create-ticket-001'
name: 'Create ServiceNow Ticket'
description: 'Create a new incident or task ticket in ServiceNow'
status: active
origin: human
enabled_operations:
  - create_ticket
policy_constraints:
  - type: precondition
    rule: "proposal.data_classification !== 'restricted'"
    error_message: 'Cannot create tickets with restricted data'
  - type: precondition
    rule: 'proposal.confidence >= 0.7'
    error_message: 'Confidence too low for automatic ticket creation'
  - type: invariant
    rule: "proposal.target_entity === null || proposal.target_entity.source === 'servicenow'"
    error_message: 'Target must be ServiceNow or null'
required_trust_level: SUPERVISED
priority: 80
data_domain: enterprise
conflict_resolution: policy_escalation
proposal_schema:
  type: object
  required:
    - intent
    - operation
  properties:
    intent:
      type: string
    operation:
      enum:
        - create_ticket
rollback_specification:
  type: FULL
  method: 'Delete created ticket via ServiceNow API'
created_at: '2026-01-28T00:00:00Z'
activated_at: '2026-01-28T10:00:00Z'
```
