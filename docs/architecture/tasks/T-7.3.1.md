## Technical Notes

```typescript
// tests/security/constraint-sandbox.test.ts
describe('Constraint Sandbox Security', () => {
  it('should block require() attempts', () => {
    const constraint: PolicyConstraint = {
      type: 'precondition',
      rule: 'require("fs").readFileSync("/etc/passwd")',
      error_message: 'Test',
    };

    const result = evaluator.evaluateConstraint(constraint, context);
    expect(result.passed).toBe(false);
    expect(result.error).toBeDefined();
  });

  it('should block process access', () => {
    const constraint: PolicyConstraint = {
      type: 'precondition',
      rule: 'process.exit(1)',
      error_message: 'Test',
    };

    const result = evaluator.evaluateConstraint(constraint, context);
    expect(result.passed).toBe(false);
  });
});

// tests/security/sql-injection.test.ts
describe('SQL Injection Prevention', () => {
  it('should prevent SQL injection in audit query', async () => {
    const malicious = "' OR '1'='1";
    const records = await auditService.query({
      operation: malicious as OperationType,
    });
    // Verify no injection occurred
    expect(records).toEqual([]);
  });
});
```
