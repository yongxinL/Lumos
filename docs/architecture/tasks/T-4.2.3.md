## Objective

## Technical Notes

### Attestation System

```typescript
// src/main/services/governance/attestation.system.ts
import { TrustManagementService } from './trust-management.service';
import { EventEmitter } from 'events';

export class AttestationSystem extends EventEmitter {
  private trustService: TrustManagementService;
  private notificationTimer?: NodeJS.Timeout;
  private config: AttestationConfig;

  constructor(trustService: TrustManagementService, config?: Partial<AttestationConfig>) {
    super();
    this.trustService = trustService;
    this.config = {
      checkInterval: 24 * 60 * 60 * 1000, // 24 hours
      overdueGracePeriod: 7, // days
      notificationsEnabled: true,
      ...config,
    };
  }

  async initialize(): Promise<void> {
    // Start periodic checks
    this.startMonitoring();

    // Listen to trust service events
    this.trustService.on('trust:promoted', (event) => {
      if (event.newLevel === 'DELEGATED') {
        console.log(`New DELEGATED operation: ${event.operation}`);
      }
    });

    console.log('Attestation system initialized');
  }

  /**
   * Get operations requiring attestation
   */
  getAttestationsDue(): AttestationItem[] {
    const checks = this.trustService.checkAttestationDue();

    return checks.map((check) => ({
      operation: check.operation,
      dueDate: check.dueDate,
      daysPastDue: check.daysPastDue,
      status: this.getAttestationStatus(check.daysPastDue),
      action: this.getRequiredAction(check.daysPastDue),
    }));
  }

  /**
   * Confirm attestations
   */
  async confirmAttestations(operations: OperationType[]): Promise<ConfirmationResult> {
    try {
      await this.trustService.confirmAttestation(operations);

      // Record attestation
      await this.recordAttestations(operations);

      return {
        success: true,
        confirmedCount: operations.length,
      };
    } catch (error) {
      return {
        success: false,
        error: error instanceof Error ? error.message : 'Unknown error',
      };
    }
  }

  /**
   * Process overdue attestations
   */
  async processOverdue(): Promise<OverdueResult> {
    const due = this.getAttestationsDue();
    const overdue = due.filter((item) => item.daysPastDue >= this.config.overdueGracePeriod);

    const demoted: OperationType[] = [];

    for (const item of overdue) {
      // Demote to SUPERVISED
      await this.trustService.demoteTrust(item.operation, 'SUPERVISED');
      demoted.push(item.operation);

      // Emit event
      this.emit('attestation:auto-demoted', {
        operation: item.operation,
        reason: `Attestation overdue by ${item.daysPastDue} days`,
      });
    }

    return {
      overdueCount: overdue.length,
      demotedOperations: demoted,
    };
  }

  /**
   * Get attestation history
   */
  async getHistory(operation?: OperationType, limit = 50): Promise<AttestationRecord[]> {
    let query = `
      SELECT * FROM attestation_history
      ${operation ? 'WHERE operation = ?' : ''}
      ORDER BY confirmed_at DESC
      LIMIT ?
    `;

    const params = operation ? [operation, limit] : [limit];
    const rows = await this.db.all(query, params);

    return rows.map((row) => ({
      id: row.id,
      operation: row.operation,
      confirmedAt: row.confirmed_at,
      confirmedBy: row.confirmed_by,
      nextDue: row.next_due,
    }));
  }

  /**
   * Get attestation statistics
   */
  getStatistics(): AttestationStatistics {
    const due = this.getAttestationsDue();

    const stats: AttestationStatistics = {
      total: due.length,
      dueNow: due.filter((item) => item.daysPastDue >= 0).length,
      dueSoon: due.filter((item) => item.daysPastDue < 0 && item.daysPastDue >= -7).length,
      overdue: due.filter((item) => item.daysPastDue >= this.config.overdueGracePeriod).length,
      byOperation: {},
    };

    for (const item of due) {
      if (!stats.byOperation[item.operation]) {
        stats.byOperation[item.operation] = {
          dueDate: item.dueDate,
          daysPastDue: item.daysPastDue,
          status: item.status,
        };
      }
    }

    return stats;
  }

  /**
   * Send notification to user
   */
  private sendNotification(items: AttestationItem[]): void {
    if (!this.config.notificationsEnabled) {
      return;
    }

    const message = this.buildNotificationMessage(items);

    this.emit('attestation:notification', {
      title: 'Attestation Required',
      message,
      items,
    });
  }

  private startMonitoring(): void {
    // Immediate check
    this.performCheck();

    // Periodic checks
    this.notificationTimer = setInterval(() => {
      this.performCheck();
    }, this.config.checkInterval);
  }

  private async performCheck(): Promise<void> {
    const due = this.getAttestationsDue();

    if (due.length > 0) {
      console.log(`${due.length} operations require attestation`);
      this.sendNotification(due);
    }

    // Process overdue (auto-demote)
    const result = await this.processOverdue();
    if (result.demotedOperations.length > 0) {
      console.log(`Auto-demoted ${result.demotedOperations.length} operations`);
    }
  }

  private getAttestationStatus(daysPastDue: number): AttestationStatus {
    if (daysPastDue >= this.config.overdueGracePeriod) {
      return 'overdue';
    } else if (daysPastDue >= 0) {
      return 'due';
    } else if (daysPastDue >= -7) {
      return 'due_soon';
    } else {
      return 'ok';
    }
  }

  private getRequiredAction(daysPastDue: number): string {
    if (daysPastDue >= this.config.overdueGracePeriod) {
      return 'Confirm immediately to prevent automatic demotion';
    } else if (daysPastDue >= 0) {
      return 'Confirm to maintain delegated trust';
    } else if (daysPastDue >= -7) {
      return 'Review upcoming attestation';
    } else {
      return 'No action required';
    }
  }

  private buildNotificationMessage(items: AttestationItem[]): string {
    if (items.length === 1) {
      return `The operation "${items[0].operation}" requires attestation review.`;
    } else {
      return `${items.length} operations require attestation review.`;
    }
  }

  private async recordAttestations(operations: OperationType[]): Promise<void> {
    const now = new Date().toISOString();
    const nextDue = new Date();
    nextDue.setDate(nextDue.getDate() + 30);

    for (const operation of operations) {
      await this.db.run(
        `
        INSERT INTO attestation_history (
          id,
          operation,
          confirmed_at,
          confirmed_by,
          next_due
        ) VALUES (?, ?, ?, ?, ?)
      `,
        [
          crypto.randomUUID(),
          operation,
          now,
          'user', // TODO: Get actual user ID
          nextDue.toISOString(),
        ]
      );
    }
  }

  shutdown(): void {
    if (this.notificationTimer) {
      clearInterval(this.notificationTimer);
    }
  }
}

// Types
interface AttestationConfig {
  checkInterval: number;
  overdueGracePeriod: number;
  notificationsEnabled: boolean;
}

interface AttestationItem {
  operation: OperationType;
  dueDate: ISO8601String;
  daysPastDue: number;
  status: AttestationStatus;
  action: string;
}

type AttestationStatus = 'ok' | 'due_soon' | 'due' | 'overdue';

interface ConfirmationResult {
  success: boolean;
  confirmedCount?: number;
  error?: string;
}

interface OverdueResult {
  overdueCount: number;
  demotedOperations: OperationType[];
}

interface AttestationRecord {
  id: string;
  operation: OperationType;
  confirmedAt: ISO8601String;
  confirmedBy: string;
  nextDue: ISO8601String;
}

interface AttestationStatistics {
  total: number;
  dueNow: number;
  dueSoon: number;
  overdue: number;
  byOperation: Record<
    OperationType,
    {
      dueDate: ISO8601String;
      daysPastDue: number;
      status: AttestationStatus;
    }
  >;
}
```

### Database Schema

```sql
-- Attestation history
CREATE TABLE IF NOT EXISTS attestation_history (
  id TEXT PRIMARY KEY,
  operation TEXT NOT NULL,
  confirmed_at TEXT NOT NULL,
  confirmed_by TEXT NOT NULL,
  next_due TEXT NOT NULL
);

CREATE INDEX idx_attestation_history_operation ON attestation_history(operation);
CREATE INDEX idx_attestation_history_confirmed_at ON attestation_history(confirmed_at);
```

### IPC Integration

```typescript
// Register attestation handlers
registerHandler('attestation:get-due', async () => {
  return attestationSystem.getAttestationsDue();
});

registerHandler('attestation:confirm', async (operations: OperationType[]) => {
  return attestationSystem.confirmAttestations(operations);
});

registerHandler('attestation:get-history', async ({ operation, limit }) => {
  return attestationSystem.getHistory(operation, limit);
});

registerHandler('attestation:get-statistics', async () => {
  return attestationSystem.getStatistics();
});

// Forward events to renderer
attestationSystem.on('attestation:notification', (data) => {
  mainWindow.webContents.send('attestation:notification', data);
});
```
