# T-2.1.1: Fast Path Service

**Milestone:** M2 - Input & Classification
**Module:** 2.1 - Classification Layer
**Status:** Planned
**Priority:** Medium

---

## Objective

Implement the Fast Path Service that uses regex-based pattern matching to quickly classify user input as either action-oriented or conversational, routing requests to the appropriate processor without LLM overhead.

---

## Acceptance Criteria

- [ ] **AC-2.1.1.1**: Pattern matching engine with configurable regex patterns
- [ ] **AC-2.1.1.2**: Action detection patterns (create, update, delete, schedule, etc.)
- [ ] **AC-2.1.1.3**: Non-action patterns (questions, explanations, searches)
- [ ] **AC-2.1.1.4**: Classification confidence scoring
- [ ] **AC-2.1.1.5**: Performance target <10ms per classification
- [ ] **AC-2.1.1.6**: Pattern priority ordering (action patterns first)
- [ ] **AC-2.1.1.7**: Ambiguous input defaults to Evaluation LLM (fail-safe)

### Core Service

```typescript
// src/main/services/fast-path.service.ts
interface PatternMatcher {
  pattern: RegExp;
  classification: 'action' | 'non_action';
  confidence: number;
  route: 'evaluation_llm' | 'expert_ai';
  description: string;
}

export class FastPathService {
  private actionPatterns: PatternMatcher[] = [];
  private nonActionPatterns: PatternMatcher[] = [];

  constructor() {
    this.initializePatterns();
  }

  classify(input: ProcessedInput): FastPathResult {
    const text = input.text.toLowerCase().trim();
    const startTime = performance.now();

    // Try action patterns first (higher priority)
    for (const matcher of this.actionPatterns) {
      if (matcher.pattern.test(text)) {
        const duration = performance.now() - startTime;
        return {
          classification: 'action',
          confidence: matcher.confidence,
          matched_pattern: matcher.description,
          route_to: 'evaluation_llm',
          processing_time_ms: duration
        };
      }
    }

    // Try non-action patterns
    for (const matcher of this.nonActionPatterns) {
      if (matcher.pattern.test(text)) {
        const duration = performance.now() - startTime;
        return {
          classification: 'non_action',
          confidence: matcher.confidence,
          matched_pattern: matcher.description,
          route_to: 'expert_ai',
          processing_time_ms: duration
        };
      }
    }

    // Ambiguous: default to evaluation LLM (fail-safe)
    const duration = performance.now() - startTime;
    return {
      classification: 'action',
      confidence: 0.5,
      matched_pattern: null,
      route_to: 'evaluation_llm',
      processing_time_ms: duration
    };
  }

  private initializePatterns(): void {
    this.actionPatterns = getActionPatterns();
    this.nonActionPatterns = getNonActionPatterns();
  }

  // For testing and monitoring
  getStats(): FastPathStats {
    return {
      action_patterns: this.actionPatterns.length,
      nonaction_patterns: this.nonActionPatterns.length,
      total_patterns: this.actionPatterns.length + this.nonActionPatterns.length
    };
  }
}
```

### Confidence Scoring

```typescript
// Adjust confidence based on additional factors
function adjustConfidence(
  baseConfidence: number,
  input: ProcessedInput
): number {
  let confidence = baseConfidence;

  // Reduce confidence for very short inputs
  if (input.text.length < 10) {
    confidence *= 0.9;
  }

  // Reduce confidence if multiple sentences
  const sentences = input.text.split(/[.!?]+/).filter(s => s.trim().length > 0);
  if (sentences.length > 2) {
    confidence *= 0.85;
  }

  // Boost confidence for imperative mood
  if (/^(please\s+)?[a-z]+\s+/i.test(input.text)) {
    confidence *= 1.05;
  }

  return Math.min(1.0, Math.max(0.0, confidence));
}
```
