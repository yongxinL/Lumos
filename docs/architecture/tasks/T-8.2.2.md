## Technical Notes

```yaml
# resources/policies/default-security.yaml
id: "policy-security-default-001"
name: "Default Security Policy"
description: "Baseline security requirements for all operations"
priority: 100
enabled: true
rules:
  - condition: "proposal.data_classification === 'restricted'"
    action: deny
    reason: "Restricted data operations not permitted"

  - condition: "proposal.risk_level === 'high'"
    action: require_confirmation
    reason: "High-risk operations require user confirmation"

  - condition: "proposal.confidence < 0.6"
    action: deny
    reason: "Confidence too low for automatic operation"

  - condition: "proposal.reversibility === 'IRREVERSIBLE' && !proposal.requires_confirmation"
    action: require_confirmation
    reason: "Irreversible operations must be confirmed"

created_at: "2026-01-28T00:00:00Z"

# resources/policies/data-classification.yaml
id: "policy-data-classification-001"
name: "Data Classification Policy"
description: "Enforce data handling based on classification"
priority: 90
enabled: true
rules:
  - condition: "proposal.data_classification === 'confidential'"
    action: require_confirmation
    reason: "Confidential data requires explicit approval"

  - condition: "proposal.data_domain === 'enterprise' && trustLevel !== 'DELEGATED'"
    action: require_confirmation
    reason: "Enterprise operations require appropriate trust level"

created_at: "2026-01-28T00:00:00Z"

# resources/policies/trust-enforcement.yaml
id: "policy-trust-enforcement-001"
name: "Trust Level Enforcement"
description: "Enforce trust level requirements"
priority: 95
enabled: true
rules:
  - condition: "trustLevel === 'OBSERVE'"
    action: deny
    reason: "OBSERVE level cannot execute actions"

  - condition: "trustLevel === 'SUPERVISED'"
    action: require_confirmation
    reason: "SUPERVISED level requires confirmation"

created_at: "2026-01-28T00:00:00Z"
```
