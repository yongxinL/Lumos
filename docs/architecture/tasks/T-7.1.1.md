// tests/integration/e2e-pipeline.test.ts
describe('End-to-End Pipeline', () => {
  it('should process text input through complete pipeline', async () => {
    // 1. Submit text input
    const input = 'Create a ticket for server downtime';
    const result = await inputHandler.submitText(input);

    // 2. Verify proposal generated
    expect(result.proposal).toBeDefined();
    expect(result.proposal.operation).toBe('create_ticket');

    // 3. Verify policy evaluation
    const policyResult = await policyEngine.evaluate(result.proposal);
    expect(policyResult.allowed).toBe(true);

    // 4. Execute action
    const execution = await executionLayer.execute({
      proposal: result.proposal,
      plan: result.plan,
      userId: 'test-user',
    });
    expect(execution.success).toBe(true);

    // 5. Verify audit log
    const auditRecords = await auditService.query({
      proposalId: result.proposal.id,
    });
    expect(auditRecords.length).toBeGreaterThan(0);
