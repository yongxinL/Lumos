## Technical Notes

### Database Schema

**Core Tables:**
```sql
-- Audit log (append-only)
CREATE TABLE audit_log (
  id TEXT PRIMARY KEY,
  timestamp TEXT NOT NULL,
  event_type TEXT NOT NULL,
  actor_type TEXT NOT NULL,
  actor_id TEXT,
  action_id TEXT,
  operation TEXT,
  outcome TEXT NOT NULL,
  input_hash TEXT,
  output_hash TEXT,
  duration_ms INTEGER,
  error_message TEXT,
  rollback_of TEXT REFERENCES audit_log(id),
  metadata TEXT  -- JSON
);

-- Skills registry
CREATE TABLE skills (
  id TEXT PRIMARY KEY,
  name TEXT NOT NULL,
  description TEXT NOT NULL,
  version_hash TEXT NOT NULL UNIQUE,
  status TEXT NOT NULL,
  origin TEXT NOT NULL,
  enabled_operations TEXT NOT NULL,  -- JSON array
  policy_constraints TEXT NOT NULL,  -- JSON array
  required_trust_level TEXT NOT NULL,
  priority INTEGER NOT NULL,
  data_domain TEXT NOT NULL,
  conflict_resolution TEXT NOT NULL,
  proposal_schema TEXT NOT NULL,  -- JSON
  rollback_specification TEXT NOT NULL,  -- JSON
  created_at TEXT NOT NULL,
  activated_at TEXT,
  activated_by TEXT
);

-- Policies
CREATE TABLE policies (
  id TEXT PRIMARY KEY,
  name TEXT NOT NULL,
  description TEXT NOT NULL,
  version_hash TEXT NOT NULL UNIQUE,
  rules TEXT NOT NULL,  -- JSON array
  priority INTEGER NOT NULL,
  enabled INTEGER NOT NULL DEFAULT 1,
  created_at TEXT NOT NULL,
  updated_at TEXT NOT NULL
);

-- Trust levels
CREATE TABLE trust_levels (
  operation TEXT PRIMARY KEY,
  level TEXT NOT NULL,
  successes INTEGER NOT NULL DEFAULT 0,
  failures INTEGER NOT NULL DEFAULT 0,
  last_attestation TEXT,
  attestation_due TEXT,
  last_updated TEXT NOT NULL
);

-- Action proposals
CREATE TABLE action_proposals (
  id TEXT PRIMARY KEY,
  timestamp TEXT NOT NULL,
  intent TEXT NOT NULL,
  operation TEXT NOT NULL,
  target_entity TEXT,  -- JSON
  data_domain TEXT NOT NULL,
  data_classification TEXT NOT NULL,
  reversibility TEXT NOT NULL,
  confidence REAL NOT NULL,
  related_skills TEXT,  -- JSON array
  rollback_plan TEXT,  -- JSON
  evaluation_model TEXT NOT NULL,
  raw_user_input TEXT NOT NULL,
  requires_confirmation INTEGER NOT NULL,
  risk_level TEXT NOT NULL,
  policy_result TEXT,  -- JSON
  user_decision TEXT,
  executed_at TEXT,
  execution_result TEXT  -- JSON
);

-- MCP servers
CREATE TABLE mcp_servers (
  id TEXT PRIMARY KEY,
  name TEXT NOT NULL,
  version TEXT NOT NULL,
  status TEXT NOT NULL,
  capabilities TEXT NOT NULL,  -- JSON array
  tools TEXT NOT NULL,  -- JSON array
  registered_at TEXT NOT NULL,
  last_health_check TEXT
);
```

### Indexes

```sql
-- Audit log queries
CREATE INDEX idx_audit_timestamp ON audit_log(timestamp);
CREATE INDEX idx_audit_event_type ON audit_log(event_type);
CREATE INDEX idx_audit_operation ON audit_log(operation);
CREATE INDEX idx_audit_outcome ON audit_log(outcome);

-- Skill lookups
CREATE INDEX idx_skills_status ON skills(status);
CREATE INDEX idx_skills_priority ON skills(priority DESC);

-- Proposal history
CREATE INDEX idx_proposals_timestamp ON action_proposals(timestamp);
CREATE INDEX idx_proposals_operation ON action_proposals(operation);
```

### Append-Only Trigger

```sql
-- Prevent UPDATE/DELETE on audit_log
CREATE TRIGGER prevent_audit_update
BEFORE UPDATE ON audit_log
BEGIN
  SELECT RAISE(ABORT, 'Audit log records are immutable');
END;

CREATE TRIGGER prevent_audit_delete
BEFORE DELETE ON audit_log
BEGIN
  SELECT RAISE(ABORT, 'Audit log records cannot be deleted');
END;
```

### Migration Framework

```typescript
interface Migration {
  version: number;
  name: string;
  up: (db: Database) => void;
  down: (db: Database) => void;
}

// Migration tracking table
CREATE TABLE schema_migrations (
  version INTEGER PRIMARY KEY,
  name TEXT NOT NULL,
  applied_at TEXT NOT NULL
);
```

### WAL Mode Configuration

```typescript
db.pragma('journal_mode = WAL');
db.pragma('synchronous = NORMAL');
db.pragma('foreign_keys = ON');
```
