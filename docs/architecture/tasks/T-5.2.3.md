## Technical Notes

### Audit Export Service

```typescript
// src/main/services/audit/audit-export.service.ts
import { AuditService, AuditFilter } from './audit.service';
import * as fs from 'fs/promises';
import { dialog } from 'electron';

type ExportFormat = 'json' | 'csv';

interface ExportOptions {
  format: ExportFormat;
  filter: AuditFilter;
  outputPath?: string;
}

interface ExportResult {
  success: boolean;
  path?: string;
  recordCount?: number;
  error?: string;
}

export class AuditExportService {
  private auditService: AuditService;

  constructor(auditService: AuditService) {
    this.auditService = auditService;
  }

  async export(options: ExportOptions): Promise<ExportResult> {
    try {
      // Get output path if not provided
      let outputPath = options.outputPath;
      if (!outputPath) {
        outputPath = await this.showSaveDialog(options.format);
        if (!outputPath) {
          return {
            success: false,
            error: 'Export cancelled by user'
          };
        }
      }

      // Query audit records
      const records = await this.auditService.query(options.filter);

      // Export based on format
      switch (options.format) {
        case 'json':
          await this.exportJSON(records, outputPath);
          break;
        case 'csv':
          await this.exportCSV(records, outputPath);
          break;
        default:
          throw new Error(`Unsupported format: ${options.format}`);
      }

      return {
        success: true,
        path: outputPath,
        recordCount: records.length
      };

    } catch (error) {
      return {
        success: false,
        error: error instanceof Error ? error.message : 'Export failed'
      };
    }
  }

  async exportStream(
    options: ExportOptions,
    onProgress: (progress: number) => void
  ): Promise<ExportResult> {
    try {
      let outputPath = options.outputPath;
      if (!outputPath) {
        outputPath = await this.showSaveDialog(options.format);
        if (!outputPath) {
          return { success: false, error: 'Export cancelled' };
        }
      }

      // Query records in batches for large datasets
      const batchSize = 1000;
      let offset = 0;
      let totalRecords = 0;
      let fileHandle: fs.FileHandle | null = null;

      try {
        fileHandle = await fs.open(outputPath, 'w');

        // Write header based on format
        if (options.format === 'json') {
          await fileHandle.write('[\n');
        } else if (options.format === 'csv') {
          await fileHandle.write(this.getCSVHeader() + '\n');
        }

        let isFirstBatch = true;

        while (true) {
          const records = await this.auditService.query({
            ...options.filter,
            limit: batchSize,
            offset
          });

          if (records.length === 0) break;

          // Write batch
          if (options.format === 'json') {
            for (let i = 0; i < records.length; i++) {
              const prefix = !isFirstBatch || i > 0 ? ',\n' : '';
              await fileHandle.write(prefix + JSON.stringify(records[i], null, 2));
            }
          } else if (options.format === 'csv') {
            for (const record of records) {
              await fileHandle.write(this.recordToCSV(record) + '\n');
            }
          }

          totalRecords += records.length;
          offset += batchSize;
          isFirstBatch = false;

          // Report progress
          onProgress(totalRecords);

          // Break if less than batch size (last batch)
          if (records.length < batchSize) break;
        }

        // Write footer
        if (options.format === 'json') {
          await fileHandle.write('\n]');
        }

        return {
          success: true,
          path: outputPath,
          recordCount: totalRecords
        };

      } finally {
        if (fileHandle) {
          await fileHandle.close();
        }
      }

    } catch (error) {
      return {
        success: false,
        error: error instanceof Error ? error.message : 'Export failed'
      };
    }
  }

  // Private methods

  private async showSaveDialog(format: ExportFormat): Promise<string | null> {
    const result = await dialog.showSaveDialog({
      title: 'Export Audit Log',
      defaultPath: `audit-log-${new Date().toISOString().split('T')[0]}.${format}`,
      filters: [
        {
          name: format.toUpperCase(),
          extensions: [format]
        }
      ]
    });

    return result.filePath || null;
  }

  private async exportJSON(records: any[], outputPath: string): Promise<void> {
    const json = JSON.stringify(records, null, 2);
    await fs.writeFile(outputPath, json, 'utf-8');
  }

  private async exportCSV(records: any[], outputPath: string): Promise<void> {
    const lines: string[] = [];

    // Header
    lines.push(this.getCSVHeader());

    // Data rows
    for (const record of records) {
      lines.push(this.recordToCSV(record));
    }

    await fs.writeFile(outputPath, lines.join('\n'), 'utf-8');
  }

  private getCSVHeader(): string {
    return [
      'ID',
      'Timestamp',
      'Action ID',
      'Proposal ID',
      'Operation',
      'Status',
      'Actor',
      'Actor ID',
      'Input Hash',
      'Content Hash',
      'Rollback Of'
    ].join(',');
  }

  private recordToCSV(record: any): string {
    const fields = [
      record.id,
      record.timestamp,
      record.action_id,
      record.proposal_id,
      record.operation,
      record.status,
      record.actor,
      record.actor_id,
      record.input_hash,
      record.content_hash,
      record.rollback_of || ''
    ];

    // Escape fields containing commas or quotes
    return fields
      .map(f => {
        const str = String(f);
        if (str.includes(',') || str.includes('"') || str.includes('\n')) {
          return `"${str.replace(/"/g, '""')}"`;
        }
        return str;
      })
      .join(',');
  }
}
```

### IPC Integration

```typescript
// Register export handlers
registerHandler('audit:export', async (options: ExportOptions) => {
  return auditExportService.export(options);
});

registerHandler('audit:export-stream', async (options: ExportOptions) => {
  return new Promise((resolve) => {
    auditExportService.exportStream(options, (progress) => {
      // Send progress updates to renderer
      mainWindow.webContents.send('audit:export-progress', progress);
    }).then(resolve);
  });
});
```

### React Hook for Export

```typescript
// src/renderer/hooks/useAuditExport.ts
import { useIPC } from './useIPC';
import { useState } from 'react';

export function useAuditExport() {
  const { invoke } = useIPC();
  const [exporting, setExporting] = useState(false);
  const [progress, setProgress] = useState(0);

  useIPCEvent('audit:export-progress', (count: number) => {
    setProgress(count);
  });

  const exportAudit = async (
    format: 'json' | 'csv',
    filter: AuditFilter
  ): Promise<ExportResult> => {
    setExporting(true);
    setProgress(0);

    try {
      const result = await invoke('audit:export', { format, filter });
      return result;
    } finally {
      setExporting(false);
      setProgress(0);
    }
  };

  const exportAuditStream = async (
    format: 'json' | 'csv',
    filter: AuditFilter
  ): Promise<ExportResult> => {
    setExporting(true);
    setProgress(0);

    try {
      const result = await invoke('audit:export-stream', { format, filter });
      return result;
    } finally {
      setExporting(false);
      setProgress(0);
    }
  };

  return {
    exportAudit,
    exportAuditStream,
    exporting,
    progress
  };
}
```

### Usage in Component

```typescript
function AuditExportButton() {
  const { exportAudit, exporting, progress } = useAuditExport();

  const handleExport = async (format: 'json' | 'csv') => {
    const result = await exportAudit(format, {
      startDate: '2026-01-01T00:00:00Z'
    });

    if (result.success) {
      console.log(`Exported ${result.recordCount} records to ${result.path}`);
    } else {
      console.error('Export failed:', result.error);
    }
  };

  return (
    <div>
      <button onClick={() => handleExport('json')} disabled={exporting}>
        Export as JSON
      </button>
      <button onClick={() => handleExport('csv')} disabled={exporting}>
        Export as CSV
      </button>
      {exporting && <div>Exporting... {progress} records</div>}
    </div>
  );
}
```
