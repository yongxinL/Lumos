### Rollback Manager Implementation

```typescript
// src/main/services/execution/rollback-manager.ts
import { MCPServerRegistry } from '../mcp/mcp-server-registry';
import { TrustManagementService } from '../governance/trust-management.service';
import { AuditService } from '../audit/audit.service';

interface RollbackRequest {
  actionId: string;
  executedAction: ExecutedAction;
  reason: string;
  requestedBy: string;
}

interface ExecutedAction {
  actionId: string;
  operation: OperationType;
  proposal: ActionProposal;
  stateSnapshot: StateSnapshot | null;
  outputs: any[];
  executedAt: ISO8601String;
}

interface RollbackResult {
  success: boolean;
  actionId: string;
  rollbackType: RollbackType;
  steps: RollbackStep[];
  duration: number;
  error?: string;
}

interface RollbackStep {
  order: number;
  action: string;
  success: boolean;
  result?: any;
  error?: string;
}

export class RollbackManager {
  private mcpRegistry: MCPServerRegistry;
  private trustService: TrustManagementService;
  private auditService: AuditService;
  private rollbackHistory: Map<string, RollbackResult> = new Map();

  constructor(
    mcpRegistry: MCPServerRegistry,
    trustService: TrustManagementService,
    auditService: AuditService
  ) {
    this.mcpRegistry = mcpRegistry;
    this.trustService = trustService;
    this.auditService = auditService;
  }

  async rollback(request: RollbackRequest): Promise<RollbackResult> {
    const startTime = Date.now();
    const rollbackType = request.executedAction.proposal.reversibility;

    try {
      // Log rollback initiation
      await this.auditService.log({
        action_id: request.actionId,
        proposal_id: request.executedAction.proposal.id,
        operation: request.executedAction.operation,
        status: 'rolling_back',
        actor: request.requestedBy,
        data: { reason: request.reason, rollbackType },
      });

      // Execute appropriate rollback strategy
      let steps: RollbackStep[];

      switch (rollbackType) {
        case 'FULL':
          steps = await this.rollbackFull(request.executedAction);
          break;
        case 'PARTIAL':
          steps = await this.rollbackPartial(request.executedAction);
          break;
        case 'COMPENSATABLE':
          steps = await this.rollbackCompensatable(request.executedAction);
          break;
        case 'IRREVERSIBLE':
          throw new RollbackError('Operation is irreversible', 'IRREVERSIBLE');
        default:
          throw new RollbackError('Unknown rollback type', 'UNKNOWN_TYPE');
      }

      const duration = Date.now() - startTime;
      const allSuccess = steps.every((s) => s.success);

      // Log rollback completion
      await this.auditService.log({
        action_id: request.actionId,
        proposal_id: request.executedAction.proposal.id,
        operation: request.executedAction.operation,
        status: allSuccess ? 'rolled_back' : 'rollback_partial',
        actor: request.requestedBy,
        data: { steps, duration },
      });

      // Demote trust level on rollback
      await this.trustService.recordRollback(request.executedAction.operation);

      const result: RollbackResult = {
        success: allSuccess,
        actionId: request.actionId,
        rollbackType,
        steps,
        duration,
      };

      // Store in history
      this.rollbackHistory.set(request.actionId, result);

      return result;
    } catch (error) {
      const duration = Date.now() - startTime;

      // Log rollback failure
      await this.auditService.log({
        action_id: request.actionId,
        proposal_id: request.executedAction.proposal.id,
        operation: request.executedAction.operation,
        status: 'rollback_failed',
        actor: request.requestedBy,
        data: {
          error: error instanceof Error ? error.message : 'Unknown error',
          duration,
        },
      });

      return {
        success: false,
        actionId: request.actionId,
        rollbackType,
        steps: [],
        duration,
        error: error instanceof Error ? error.message : 'Rollback failed',
      };
    }
  }

  async getRollbackHistory(actionId: string): Promise<RollbackResult | null> {
    return this.rollbackHistory.get(actionId) || null;
  }

  canRollback(action: ExecutedAction): CanRollbackResult {
    const reversibility = action.proposal.reversibility;

    if (reversibility === 'IRREVERSIBLE') {
      return {
        canRollback: false,
        reason: 'Operation is irreversible and cannot be undone',
      };
    }

    if (reversibility === 'FULL' && !action.stateSnapshot) {
      return {
        canRollback: false,
        reason: 'No state snapshot available for full rollback',
      };
    }

    return {
      canRollback: true,
      estimatedSuccessRate: this.estimateSuccessRate(action),
    };
  }

  // Private methods - Rollback strategies

  private async rollbackFull(action: ExecutedAction): Promise<RollbackStep[]> {
    const steps: RollbackStep[] = [];

    if (!action.stateSnapshot) {
      throw new RollbackError('No state snapshot for full rollback', 'NO_SNAPSHOT');
    }

    // Restore pre-execution state
    const step = await this.restoreState(action);
    steps.push(step);

    return steps;
  }

  private async rollbackPartial(action: ExecutedAction): Promise<RollbackStep[]> {
    const steps: RollbackStep[] = [];

    // Reverse individual operations
    for (const output of action.outputs.reverse()) {
      try {
        const step = await this.reverseOperation(action.operation, output);
        steps.push(step);
      } catch (error) {
        steps.push({
          order: steps.length + 1,
          action: `Reverse ${action.operation}`,
          success: false,
          error: error instanceof Error ? error.message : 'Unknown error',
        });
      }
    }

    return steps;
  }

  private async rollbackCompensatable(action: ExecutedAction): Promise<RollbackStep[]> {
    const steps: RollbackStep[] = [];

    // Execute compensation actions
    const plan = action.proposal.rollback_plan;
    if (!plan) {
      throw new RollbackError('No rollback plan available', 'NO_PLAN');
    }

    for (const planStep of plan.steps) {
      try {
        const result = await this.executeCompensation(planStep, action);
        steps.push({
          order: planStep.order,
          action: planStep.action,
          success: true,
          result,
        });
      } catch (error) {
        steps.push({
          order: planStep.order,
          action: planStep.action,
          success: false,
          error: error instanceof Error ? error.message : 'Unknown error',
        });
      }
    }

    return steps;
  }

  private async restoreState(action: ExecutedAction): Promise<RollbackStep> {
    const snapshot = action.stateSnapshot!;

    try {
      // Determine restoration method based on operation
      if (action.operation.includes('calendar')) {
        await this.restoreCalendarState(snapshot);
      } else if (action.operation.includes('file')) {
        await this.restoreFileState(snapshot);
      }

      return {
        order: 1,
        action: 'Restore state',
        success: true,
        result: { restored: true },
      };
    } catch (error) {
      return {
        order: 1,
        action: 'Restore state',
        success: false,
        error: error instanceof Error ? error.message : 'Restoration failed',
      };
    }
  }

  private async reverseOperation(operation: OperationType, output: any): Promise<RollbackStep> {
    // Map operation to reverse operation
    const reverseOp = this.getReverseOperation(operation);

    try {
      const location = await this.mcpRegistry.findTool(reverseOp);
      if (!location) {
        throw new Error(`Reverse tool ${reverseOp} not found`);
      }

      // Execute reverse operation
      const result = await this.mcpRegistry.callTool(
        location.serverId,
        reverseOp,
        this.extractReverseArgs(output)
      );

      return {
        order: 1,
        action: reverseOp,
        success: true,
        result,
      };
    } catch (error) {
      return {
        order: 1,
        action: reverseOp,
        success: false,
        error: error instanceof Error ? error.message : 'Reverse failed',
      };
    }
  }

  private async executeCompensation(step: RollbackPlanStep, action: ExecutedAction): Promise<any> {
    // Parse compensation action and execute
    // Simplified implementation
    return { compensated: true };
  }

  private getReverseOperation(operation: OperationType): string {
    const reverseMap: Record<string, string> = {
      create_calendar_event: 'delete_event',
      update_calendar_event: 'update_event',
      delete_calendar_event: 'create_event',
      write_file: 'delete_file',
      create_note: 'delete_file',
    };

    return reverseMap[operation] || operation;
  }

  private extractReverseArgs(output: any): object {
    // Extract arguments for reverse operation
    if (output.result?.id) {
      return { id: output.result.id };
    }
    return {};
  }

  private async restoreCalendarState(snapshot: StateSnapshot): Promise<void> {
    // Restore calendar event to previous state
    if (snapshot.preState) {
      const location = await this.mcpRegistry.findTool('update_event');
      if (location) {
        await this.mcpRegistry.callTool(location.serverId, 'update_event', snapshot.preState);
      }
    }
  }

  private async restoreFileState(snapshot: StateSnapshot): Promise<void> {
    // Restore file to previous content
    if (snapshot.preState) {
      const location = await this.mcpRegistry.findTool('write_file');
      if (location) {
        await this.mcpRegistry.callTool(location.serverId, 'write_file', snapshot.preState);
      }
    }
  }

  private estimateSuccessRate(action: ExecutedAction): number {
    const reversibility = action.proposal.reversibility;

    switch (reversibility) {
      case 'FULL':
        return action.stateSnapshot ? 0.95 : 0.5;
      case 'PARTIAL':
        return 0.75;
      case 'COMPENSATABLE':
        return action.proposal.rollback_plan?.estimated_success_rate || 0.6;
      case 'IRREVERSIBLE':
        return 0.0;
      default:
        return 0.5;
    }
  }
}

// Errors
export class RollbackError extends Error {
  constructor(
    message: string,
    public code: string
  ) {
    super(message);
    this.name = 'RollbackError';
  }
}

interface CanRollbackResult {
  canRollback: boolean;
  reason?: string;
  estimatedSuccessRate?: number;
}

interface RollbackPlanStep {
  order: number;
  action: string;
  reversible: boolean;
}
```

## Test Cases

```typescript
describe('RollbackManager', () => {
  let rollbackManager: RollbackManager;

  beforeEach(() => {
    rollbackManager = new RollbackManager(mcpRegistry, trustService, auditService);
  });

  it('should rollback FULL reversible action', async () => {
    const action = createExecutedAction('FULL', withSnapshot());
    const request: RollbackRequest = {
      actionId: action.actionId,
      executedAction: action,
      reason: 'User requested',
      requestedBy: 'user-123',
    };

    const result = await rollbackManager.rollback(request);
    expect(result.success).toBe(true);
    expect(result.rollbackType).toBe('FULL');
  });

  it('should reject IRREVERSIBLE rollback', async () => {
    const action = createExecutedAction('IRREVERSIBLE');
    const request: RollbackRequest = {
      actionId: action.actionId,
      executedAction: action,
      reason: 'User requested',
      requestedBy: 'user-123',
    };

    await expect(rollbackManager.rollback(request)).rejects.toThrow('IRREVERSIBLE');
  });

  it('should execute compensation for COMPENSATABLE', async () => {
    const action = createExecutedAction('COMPENSATABLE', null, withRollbackPlan());
    const request: RollbackRequest = {
      actionId: action.actionId,
      executedAction: action,
      reason: 'User requested',
      requestedBy: 'user-123',
    };

    const result = await rollbackManager.rollback(request);
    expect(result.steps.length).toBeGreaterThan(0);
  });

  it('should check rollback capability', () => {
    const action = createExecutedAction('FULL', withSnapshot());
    const canRollback = rollbackManager.canRollback(action);
    expect(canRollback.canRollback).toBe(true);
  });

  it('should record rollback in history', async () => {
    const action = createExecutedAction('FULL', withSnapshot());
    const request: RollbackRequest = {
      actionId: action.actionId,
      executedAction: action,
      reason: 'Test',
      requestedBy: 'user',
    };

    await rollbackManager.rollback(request);
    const history = await rollbackManager.getRollbackHistory(action.actionId);
    expect(history).toBeDefined();
  });
});
```
