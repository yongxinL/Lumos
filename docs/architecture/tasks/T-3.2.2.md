### Expert AI Service Implementation

```typescript
// src/main/services/ai/expert-ai.service.ts
import { AIProviderRegistry } from './providers/provider-registry';

interface ExpertAIConfig {
  defaultProvider: string;
  maxConversationLength: number;
  contextWindowSize: number;
}

interface Conversation {
  id: string;
  messages: Message[];
  createdAt: string;
  updatedAt: string;
}

export class ExpertAIService {
  private registry: AIProviderRegistry;
  private config: ExpertAIConfig;
  private conversations: Map<string, Conversation> = new Map();
  private currentProvider: string;

  constructor(registry: AIProviderRegistry, config?: Partial<ExpertAIConfig>) {
    this.registry = registry;
    this.config = {
      defaultProvider: 'ollama',
      maxConversationLength: 50,
      contextWindowSize: 8000,
      ...config
    };
    this.currentProvider = this.config.defaultProvider;
  }

  async initialize(): Promise<void> {
    // Verify at least one provider is available
    const available = await this.registry.listAvailable();
    if (available.length === 0) {
      throw new Error('No AI providers available');
    }

    // Set default provider if available
    const defaultProvider = this.registry.get(this.config.defaultProvider);
    if (defaultProvider && await defaultProvider.checkHealth()) {
      this.currentProvider = this.config.defaultProvider;
    } else {
      // Fallback to first available
      const healthy = await this.registry.getHealthyProvider();
      if (healthy) {
        this.currentProvider = healthy.getName();
      } else {
        throw new Error('No healthy AI provider available');
      }
    }
  }

  async processQuery(input: ProcessedInput): Promise<ExpertResponse> {
    const provider = this.getProvider();

    // Create or get conversation
    const conversationId = input.metadata.conversationId || crypto.randomUUID();
    const conversation = this.getOrCreateConversation(conversationId);

    // Add user message
    conversation.messages.push({
      role: 'user',
      content: input.text
    });

    // Prune context if needed
    this.pruneContext(conversation);

    // Generate response
    const response = await provider.generate({
      messages: conversation.messages,
      options: {
        temperature: 0.7,
        maxTokens: 1000
      }
    });

    // Add assistant message
    conversation.messages.push({
      role: 'assistant',
      content: response.content
    });
    conversation.updatedAt = new Date().toISOString();

    return {
      conversationId,
      content: response.content,
      model: provider.getModel(),
      usage: response.usage
    };
  }

  async processQueryStream(
    input: ProcessedInput,
    onChunk: (chunk: string) => void
  ): Promise<ExpertResponse> {
    const provider = this.getProvider();

    const conversationId = input.metadata.conversationId || crypto.randomUUID();
    const conversation = this.getOrCreateConversation(conversationId);

    conversation.messages.push({
      role: 'user',
      content: input.text
    });

    this.pruneContext(conversation);

    let fullContent = '';
    const response = await provider.generateStream(
      {
        messages: conversation.messages,
        options: {
          temperature: 0.7,
          maxTokens: 1000
        }
      },
      (chunk) => {
        if (!chunk.isComplete) {
          fullContent += chunk.delta;
          onChunk(chunk.delta);
        }
      }
    );

    conversation.messages.push({
      role: 'assistant',
      content: response.content
    });
    conversation.updatedAt = new Date().toISOString();

    return {
      conversationId,
      content: response.content,
      model: provider.getModel(),
      usage: response.usage
    };
  }

  async planAction(proposal: ActionProposal): Promise<ActionPlan> {
    const provider = this.getProvider();

    const prompt = this.buildActionPlanPrompt(proposal);

    const response = await provider.generate({
      messages: [
        {
          role: 'system',
          content: 'You are an AI assistant that creates detailed action plans.'
        },
        {
          role: 'user',
          content: prompt
        }
      ],
      options: {
        temperature: 0.3,
        maxTokens: 1500
      }
    });

    return this.parseActionPlan(response.content, proposal);
  }

  async generateRollbackPlan(action: ExecutedAction): Promise<RollbackPlan> {
    const provider = this.getProvider();

    const prompt = this.buildRollbackPlanPrompt(action);

    const response = await provider.generate({
      messages: [
        {
          role: 'system',
          content: 'You are an AI assistant that creates rollback plans for actions.'
        },
        {
          role: 'user',
          content: prompt
        }
      ],
      options: {
        temperature: 0.1,
        maxTokens: 1000
      }
    });

    return this.parseRollbackPlan(response.content, action);
  }

  async switchModel(providerName: string): Promise<void> {
    const provider = this.registry.get(providerName);
    if (!provider) {
      throw new Error(`Provider ${providerName} not found`);
    }

    const healthy = await provider.checkHealth();
    if (!healthy) {
      throw new Error(`Provider ${providerName} is not healthy`);
    }

    this.currentProvider = providerName;
  }

  getAvailableModels(): Promise<ModelOption[]> {
    return this.registry.listAvailable().then(statuses =>
      statuses.map(s => ({
        name: s.name,
        model: s.model,
        type: s.type,
        healthy: s.healthy
      }))
    );
  }

  clearConversation(conversationId: string): void {
    this.conversations.delete(conversationId);
  }

  private getProvider(): AIProvider {
    const provider = this.registry.get(this.currentProvider);
    if (!provider) {
      throw new Error(`Provider ${this.currentProvider} not available`);
    }
    return provider;
  }

  private getOrCreateConversation(id: string): Conversation {
    if (!this.conversations.has(id)) {
      this.conversations.set(id, {
        id,
        messages: [],
        createdAt: new Date().toISOString(),
        updatedAt: new Date().toISOString()
      });
    }
    return this.conversations.get(id)!;
  }

  private pruneContext(conversation: Conversation): void {
    // Keep system message + recent messages within context window
    if (conversation.messages.length <= this.config.maxConversationLength) {
      return;
    }

    // Keep first message (system) and last N messages
    const systemMessages = conversation.messages.filter(m => m.role === 'system');
    const recentMessages = conversation.messages
      .filter(m => m.role !== 'system')
      .slice(-this.config.maxConversationLength);

    conversation.messages = [...systemMessages, ...recentMessages];
  }

  private buildActionPlanPrompt(proposal: ActionProposal): string {
    return `Create a detailed action plan for the following proposal:

Operation: ${proposal.operation}
Intent: ${proposal.intent}
Target: ${JSON.stringify(proposal.target_entity)}

The action plan should include:
1. Prerequisites to check
2. Step-by-step execution steps
3. Expected outcomes
4. Validation checks

Format as JSON with fields: prerequisites, steps, expectedOutcomes, validationChecks`;
  }

  private buildRollbackPlanPrompt(action: ExecutedAction): string {
    return `Create a rollback plan for the following executed action:

Operation: ${action.operation}
Result: ${JSON.stringify(action.result)}

The rollback plan should include:
1. Rollback type (FULL, PARTIAL, COMPENSATABLE, or IRREVERSIBLE)
2. Method description
3. Step-by-step rollback steps
4. Success rate estimate

Format as JSON.`;
  }

  private parseActionPlan(content: string, proposal: ActionProposal): ActionPlan {
    try {
      const parsed = JSON.parse(content);
      return {
        id: crypto.randomUUID(),
        proposalId: proposal.id,
        steps: parsed.steps || [],
        prerequisites: parsed.prerequisites || [],
        expectedOutcomes: parsed.expectedOutcomes || [],
        validationChecks: parsed.validationChecks || []
      };
    } catch {
      // Fallback to simple plan
      return {
        id: crypto.randomUUID(),
        proposalId: proposal.id,
        steps: [{ order: 1, action: proposal.intent, description: content }],
        prerequisites: [],
        expectedOutcomes: [],
        validationChecks: []
      };
    }
  }

  private parseRollbackPlan(content: string, action: ExecutedAction): RollbackPlan {
    try {
      const parsed = JSON.parse(content);
      return {
        type: parsed.type || 'PARTIAL',
        method: parsed.method || 'Manual intervention',
        steps: parsed.steps || [],
        estimated_success_rate: parsed.estimated_success_rate || 0.5
      };
    } catch {
      return {
        type: 'PARTIAL',
        method: 'Manual intervention required',
        steps: [],
        estimated_success_rate: 0.5
      };
    }
  }
}

// Types
export interface ExpertResponse {
  conversationId: string;
  content: string;
  model: string;
  usage: TokenUsage;
}

export interface ActionPlan {
  id: string;
  proposalId: string;
  steps: ActionStep[];
  prerequisites: string[];
  expectedOutcomes: string[];
  validationChecks: string[];
}

export interface ActionStep {
  order: number;
  action: string;
  description: string;
}

export interface ModelOption {
  name: string;
  model: string;
  type: 'local' | 'cloud';
  healthy: boolean;
}
```
