## Technical Notes

```typescript
// src/renderer/components/chat/ProposalCard.tsx
interface ProposalCardProps {
  proposal: ActionProposal;
  policyResult: PolicyResult;
  onApprove: () => void;
  onReject: () => void;
  loading?: boolean;
}

export function ProposalCard({
  proposal,
  policyResult,
  onApprove,
  onReject,
  loading
}: ProposalCardProps) {
  const getRiskColor = (risk: string) => {
    switch (risk) {
      case 'low': return 'text-green-500';
      case 'medium': return 'text-yellow-500';
      case 'high': return 'text-red-500';
      default: return 'text-gray-500';
    }
  };

  return (
    <div className="border rounded-lg p-4 space-y-4 bg-card">
      <div className="flex items-start justify-between">
        <div className="flex-1">
          <h3 className="font-semibold">{proposal.intent}</h3>
          <p className="text-sm text-muted-foreground mt-1">
            Operation: {proposal.operation}
          </p>
        </div>
        <div className={cn('px-2 py-1 rounded text-xs font-medium', getRiskColor(proposal.risk_level))}>
          {proposal.risk_level.toUpperCase()} RISK
        </div>
      </div>

      <div className="space-y-2 text-sm">
        <div className="flex justify-between">
          <span className="text-muted-foreground">Data Domain:</span>
          <span>{proposal.data_domain}</span>
        </div>
        <div className="flex justify-between">
          <span className="text-muted-foreground">Reversibility:</span>
          <span>{proposal.reversibility}</span>
        </div>
        <div className="flex justify-between">
          <span className="text-muted-foreground">Confidence:</span>
          <span>{(proposal.confidence * 100).toFixed(0)}%</span>
        </div>
      </div>

      {proposal.rollback_plan && (
        <div className="text-xs p-2 bg-accent rounded">
          <p className="font-medium">Rollback Plan:</p>
          <p className="text-muted-foreground mt-1">
            {proposal.rollback_plan.method}
          </p>
        </div>
      )}

      {!policyResult.allowed && (
        <div className="p-2 bg-destructive/10 rounded text-sm text-destructive">
          {policyResult.denial_reason}
        </div>
      )}

      <div className="flex gap-2">
        <button
          onClick={onReject}
          disabled={loading}
          className="flex-1 px-4 py-2 border rounded-md hover:bg-accent"
        >
          Reject
        </button>
        <button
          onClick={onApprove}
          disabled={loading || !policyResult.allowed}
          className="flex-1 px-4 py-2 bg-primary text-primary-foreground rounded-md hover:bg-primary/90 disabled:opacity-50"
        >
          {policyResult.requires_confirmation ? 'Approve' : 'Auto-Execute'}
        </button>
      </div>
    </div>
  );
}
```
