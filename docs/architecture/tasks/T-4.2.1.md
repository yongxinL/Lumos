## Objective

Implement the Trust Level State Machine that defines state transitions between OBSERVE, SUPERVISED, and DELEGATED trust levels with validation rules for promotions and automatic demotion on failures.

- [ ] **AC-4.2.1.3**: Automatic demotion on rollback (DELEGATED→SUPERVISED)
- [ ] **AC-4.2.1.4**: State transition validation

- [ ] **AC-4.2.1.6**: Attestation requirement for DELEGATED (30-day interval)

## Technical Notes

### Trust Level State Machine

```typescript
// src/main/services/governance/trust-level-state-machine.ts

export type TrustLevel = 'OBSERVE' | 'SUPERVISED' | 'DELEGATED';

interface TrustState {
  operation: OperationType;
  level: TrustLevel;
  successes: number;
  failures: number;
  lastAttestation: ISO8601String | null;
  attestationDue: ISO8601String | null;
  lastUpdated: ISO8601String;
  transitionHistory: StateTransition[];
}

interface StateTransition {
  from: TrustLevel;
  to: TrustLevel;
  reason: string;
  timestamp: ISO8601String;
}

interface PromotionRequirement {
  requiredSuccesses: number;
  maxFailures: number;
  attestationRequired: boolean;
}

export class TrustLevelStateMachine {
  private static readonly PROMOTION_RULES: Record<TrustLevel, PromotionRequirement> = {
    OBSERVE: {
      requiredSuccesses: 0, // Can promote immediately
      maxFailures: Infinity,
      attestationRequired: false,
    },
    SUPERVISED: {
      requiredSuccesses: 10, // Need 10 successful operations
      maxFailures: 0, // No failures allowed
      attestationRequired: false,
    },
    DELEGATED: {
      requiredSuccesses: Infinity, // Cannot promote from DELEGATED
      maxFailures: 0,
      attestationRequired: true, // Requires periodic attestation
    },
  };

  private static readonly ATTESTATION_INTERVAL_DAYS = 30;

  /**
   * Check if promotion is allowed
   */
  canPromote(state: TrustState): boolean {
    const currentLevel = state.level;

    // Cannot promote from DELEGATED (it's the highest)
    if (currentLevel === 'DELEGATED') {
      return false;
    }

    // Get next level
    const nextLevel = this.getNextLevel(currentLevel);
    if (!nextLevel) return false;

    // Get promotion requirements for current level
    const requirements = TrustLevelStateMachine.PROMOTION_RULES[currentLevel];

    // Check success count
    if (state.successes < requirements.requiredSuccesses) {
      return false;
    }

    // Check failure count
    if (state.failures > requirements.maxFailures) {
      return false;
    }

    return true;
  }

  /**
   * Promote to next level
   */
  promote(state: TrustState, reason: string): PromoteResult {
    if (!this.canPromote(state)) {
      return {
        success: false,
        error: this.getPromotionBlockReason(state),
      };
    }

    const oldLevel = state.level;
    const newLevel = this.getNextLevel(oldLevel)!;

    // Update state
    state.level = newLevel;
    state.lastUpdated = new Date().toISOString();

    // Set attestation if promoting to DELEGATED
    if (newLevel === 'DELEGATED') {
      state.lastAttestation = new Date().toISOString();
      state.attestationDue = this.calculateAttestationDue();
    }

    // Record transition
    state.transitionHistory.push({
      from: oldLevel,
      to: newLevel,
      reason,
      timestamp: state.lastUpdated,
    });

    return {
      success: true,
      oldLevel,
      newLevel,
    };
  }

  /**
   * Demote to specified level
   */
  demote(state: TrustState, targetLevel: TrustLevel, reason: string): DemoteResult {
    if (!this.canDemote(state, targetLevel)) {
      return {
        success: false,
        error: `Cannot demote from ${state.level} to ${targetLevel}`,
      };
    }

    const oldLevel = state.level;

    // Update state
    state.level = targetLevel;
    state.lastUpdated = new Date().toISOString();

    // Clear attestation if demoting from DELEGATED
    if (oldLevel === 'DELEGATED' && targetLevel !== 'DELEGATED') {
      state.lastAttestation = null;
      state.attestationDue = null;
    }

    // Record transition
    state.transitionHistory.push({
      from: oldLevel,
      to: targetLevel,
      reason,
      timestamp: state.lastUpdated,
    });

    return {
      success: true,
      oldLevel,
      newLevel: targetLevel,
    };
  }

  /**
   * Automatic demotion on rollback
   */
  demoteOnRollback(state: TrustState): DemoteResult {
    if (state.level === 'OBSERVE') {
      // Already at lowest level
      return {
        success: false,
        error: 'Already at lowest trust level',
      };
    }

    const targetLevel = state.level === 'DELEGATED' ? 'SUPERVISED' : 'OBSERVE';

    return this.demote(state, targetLevel, 'Automatic demotion due to rollback');
  }

  /**
   * Record successful operation
   */
  recordSuccess(state: TrustState): void {
    state.successes++;
    state.lastUpdated = new Date().toISOString();
  }

  /**
   * Record failed operation
   */
  recordFailure(state: TrustState): void {
    state.failures++;
    state.lastUpdated = new Date().toISOString();

    // Reset success counter if in promotion track
    if (state.level === 'SUPERVISED') {
      state.successes = 0;
    }
  }

  /**
   * Check if attestation is due
   */
  isAttestationDue(state: TrustState): boolean {
    if (state.level !== 'DELEGATED') {
      return false;
    }

    if (!state.attestationDue) {
      return true; // Should have attestation due date
    }

    const now = new Date();
    const due = new Date(state.attestationDue);

    return now >= due;
  }

  /**
   * Confirm attestation
   */
  confirmAttestation(state: TrustState): void {
    if (state.level !== 'DELEGATED') {
      throw new Error('Attestation only applies to DELEGATED trust level');
    }

    state.lastAttestation = new Date().toISOString();
    state.attestationDue = this.calculateAttestationDue();
    state.lastUpdated = new Date().toISOString();
  }

  /**
   * Get progress toward next promotion
   */
  getPromotionProgress(state: TrustState): PromotionProgress {
    if (state.level === 'DELEGATED') {
      return {
        canPromote: false,
        progress: 1.0,
        reason: 'Already at highest trust level',
      };
    }

    const requirements = TrustLevelStateMachine.PROMOTION_RULES[state.level];

    if (requirements.requiredSuccesses === 0) {
      return {
        canPromote: true,
        progress: 1.0,
        reason: 'Ready to promote',
      };
    }

    const progress = Math.min(state.successes / requirements.requiredSuccesses, 1.0);

    const canPromote = this.canPromote(state);

    return {
      canPromote,
      progress,
      currentSuccesses: state.successes,
      requiredSuccesses: requirements.requiredSuccesses,
      failures: state.failures,
      reason: canPromote ? 'Ready to promote' : this.getPromotionBlockReason(state),
    };
  }

  // Private methods

  private getNextLevel(current: TrustLevel): TrustLevel | null {
    switch (current) {
      case 'OBSERVE':
        return 'SUPERVISED';
      case 'SUPERVISED':
        return 'DELEGATED';
      case 'DELEGATED':
        return null;
    }
  }

  private canDemote(state: TrustState, targetLevel: TrustLevel): boolean {
    const levels: TrustLevel[] = ['OBSERVE', 'SUPERVISED', 'DELEGATED'];
    const currentIndex = levels.indexOf(state.level);
    const targetIndex = levels.indexOf(targetLevel);

    return targetIndex < currentIndex;
  }

  private getPromotionBlockReason(state: TrustState): string {
    if (state.level === 'DELEGATED') {
      return 'Already at highest trust level';
    }

    const requirements = TrustLevelStateMachine.PROMOTION_RULES[state.level];

    if (state.successes < requirements.requiredSuccesses) {
      return `Need ${requirements.requiredSuccesses - state.successes} more successful operations`;
    }

    if (state.failures > requirements.maxFailures) {
      return 'Has failures, must be reset';
    }

    return 'Unknown';
  }

  private calculateAttestationDue(): ISO8601String {
    const due = new Date();
    due.setDate(due.getDate() + TrustLevelStateMachine.ATTESTATION_INTERVAL_DAYS);
    return due.toISOString() as ISO8601String;
  }
}

// Types
interface PromoteResult {
  success: boolean;
  oldLevel?: TrustLevel;
  newLevel?: TrustLevel;
  error?: string;
}

interface DemoteResult {
  success: boolean;
  oldLevel?: TrustLevel;
  newLevel?: TrustLevel;
  error?: string;
}

interface PromotionProgress {
  canPromote: boolean;
  progress: number;
  currentSuccesses?: number;
  requiredSuccesses?: number;
  failures?: number;
  reason: string;
}

export { TrustState, StateTransition, PromotionProgress };
```

### Example State Transitions

```typescript
// Initial state (OBSERVE)
const state: TrustState = {
  operation: 'create_ticket',
  level: 'OBSERVE',
  successes: 0,
  failures: 0,
  lastAttestation: null,
  attestationDue: null,
  lastUpdated: new Date().toISOString(),
  transitionHistory: [],
};

const machine = new TrustLevelStateMachine();

// User requests promotion to SUPERVISED (immediate)
if (machine.canPromote(state)) {
  const result = machine.promote(state, 'User requested promotion');
  console.log(`Promoted from ${result.oldLevel} to ${result.newLevel}`);
}

// Record successful operations
for (let i = 0; i < 10; i++) {
  machine.recordSuccess(state);
}

// Check promotion progress
const progress = machine.getPromotionProgress(state);
console.log(`Promotion progress: ${progress.progress * 100}%`);

// Promote to DELEGATED
if (machine.canPromote(state)) {
  machine.promote(state, '10 successful operations completed');
}

// Rollback occurs - automatic demotion
machine.demoteOnRollback(state);
console.log(`Demoted to ${state.level}`);

// Check attestation
if (machine.isAttestationDue(state)) {
  console.log('Attestation required');
  machine.confirmAttestation(state);
}
```

## Test Cases

```typescript
describe('TrustLevelStateMachine', () => {
  let machine: TrustLevelStateMachine;
  let state: TrustState;

  beforeEach(() => {
    machine = new TrustLevelStateMachine();
    state = {
      operation: 'create_ticket',
      level: 'OBSERVE',
      successes: 0,
      failures: 0,
      lastAttestation: null,
      attestationDue: null,
      lastUpdated: new Date().toISOString(),
      transitionHistory: [],
    };
  });

  it('should promote from OBSERVE to SUPERVISED immediately', () => {
    expect(machine.canPromote(state)).toBe(true);
    const result = machine.promote(state, 'Test');
    expect(result.success).toBe(true);
    expect(state.level).toBe('SUPERVISED');
  });

  it('should require 10 successes for SUPERVISED to DELEGATED', () => {
    state.level = 'SUPERVISED';
    expect(machine.canPromote(state)).toBe(false);

    for (let i = 0; i < 10; i++) {
      machine.recordSuccess(state);
    }

    expect(machine.canPromote(state)).toBe(true);
  });

  it('should reset successes on failure in SUPERVISED', () => {
    state.level = 'SUPERVISED';
    machine.recordSuccess(state);
    machine.recordSuccess(state);
    expect(state.successes).toBe(2);

    machine.recordFailure(state);
    expect(state.successes).toBe(0);
  });

  it('should demote DELEGATED to SUPERVISED on rollback', () => {
    state.level = 'DELEGATED';
    const result = machine.demoteOnRollback(state);
    expect(result.success).toBe(true);
    expect(state.level).toBe('SUPERVISED');
  });

  it('should set attestation when promoting to DELEGATED', () => {
    state.level = 'SUPERVISED';
    state.successes = 10;

    machine.promote(state, 'Test');
    expect(state.level).toBe('DELEGATED');
    expect(state.lastAttestation).toBeDefined();
    expect(state.attestationDue).toBeDefined();
  });

  it('should track transition history', () => {
    machine.promote(state, 'First promotion');
    expect(state.transitionHistory).toHaveLength(1);
    expect(state.transitionHistory[0].from).toBe('OBSERVE');
    expect(state.transitionHistory[0].to).toBe('SUPERVISED');
  });

  it('should calculate promotion progress', () => {
    state.level = 'SUPERVISED';
    state.successes = 5;

    const progress = machine.getPromotionProgress(state);
    expect(progress.progress).toBe(0.5);
    expect(progress.canPromote).toBe(false);
  });
});
```
