### Constrained Decoding Utilities

```typescript
// src/main/services/ai/constrained-json.ts
import Ajv, { JSONSchemaType, ValidateFunction } from 'ajv';
import addFormats from 'ajv-formats';

export class ConstrainedJSONDecoder {
  private validator: Ajv;
  private schemaCache: Map<string, ValidateFunction> = new Map();

  constructor() {
    this.validator = new Ajv({
      allErrors: true,
      verbose: true,
      strict: false
    });
    addFormats(this.validator);
  }

  /**
   * Validate JSON against schema
   */
  validate<T>(data: unknown, schema: JSONSchemaType<T>): ValidationResult<T> {
    const validate = this.getValidator(schema);
    const valid = validate(data);

    if (valid) {
      return {
        success: true,
        data: data as T
      };
    }

    return {
      success: false,
      errors: this.validator.errorsText(validate.errors, { dataVar: 'data' }),
      validationErrors: validate.errors || []
    };
  }

  /**
   * Attempt to parse and fix common JSON errors
   */
  parseWithFallback(jsonString: string): ParseResult {
    // Try direct parse
    try {
      return {
        success: true,
        data: JSON.parse(jsonString)
      };
    } catch (error) {
      // Attempt fixes
      const fixed = this.attemptFix(jsonString);
      if (fixed) {
        try {
          return {
            success: true,
            data: JSON.parse(fixed),
            wasFixed: true
          };
        } catch {
          // Fall through
        }
      }

      return {
        success: false,
        error: error instanceof Error ? error.message : 'Parse failed'
      };
    }
  }

  /**
   * Complete partial JSON (e.g., truncated responses)
   */
  completePartialJSON(partial: string): string {
    let completed = partial.trim();

    // Count braces and brackets
    let braceCount = 0;
    let bracketCount = 0;
    let inString = false;
    let escapeNext = false;

    for (const char of completed) {
      if (escapeNext) {
        escapeNext = false;
        continue;
      }

      if (char === '\\') {
        escapeNext = true;
        continue;
      }

      if (char === '"') {
        inString = !inString;
        continue;
      }

      if (inString) continue;

      if (char === '{') braceCount++;
      if (char === '}') braceCount--;
      if (char === '[') bracketCount++;
      if (char === ']') bracketCount--;
    }

    // Close unclosed strings
    if (inString) {
      completed += '"';
    }

    // Close unclosed objects
    while (braceCount > 0) {
      completed += '}';
      braceCount--;
    }

    // Close unclosed arrays
    while (bracketCount > 0) {
      completed += ']';
      bracketCount--;
    }

    return completed;
  }

  /**
   * Simplify schema for better LLM understanding
   */
  simplifySchema(schema: object): object {
    const simplified = JSON.parse(JSON.stringify(schema));

    // Remove unnecessary fields
    const fieldsToRemove = [
      'examples',
      'default',
      '$comment',
      'title',
      'description' // Optional: remove if token limit is issue
    ];

    this.removeFields(simplified, fieldsToRemove);

    return simplified;
  }

  /**
   * Get common schema templates
   */
  static getTemplate(type: SchemaTemplate): object {
    return SCHEMA_TEMPLATES[type];
  }

  // Private methods

  private getValidator(schema: object): ValidateFunction {
    const key = JSON.stringify(schema);

    if (!this.schemaCache.has(key)) {
      const validate = this.validator.compile(schema);
      this.schemaCache.set(key, validate);
    }

    return this.schemaCache.get(key)!;
  }

  private attemptFix(jsonString: string): string | null {
    let fixed = jsonString.trim();

    // Fix trailing commas
    fixed = fixed.replace(/,(\s*[}\]])/g, '$1');

    // Fix unquoted keys
    fixed = fixed.replace(/([{,]\s*)([a-zA-Z_][a-zA-Z0-9_]*)\s*:/g, '$1"$2":');

    // Fix single quotes
    fixed = fixed.replace(/'/g, '"');

    // Complete partial JSON
    fixed = this.completePartialJSON(fixed);

    return fixed;
  }

  private removeFields(obj: any, fields: string[]): void {
    if (typeof obj !== 'object' || obj === null) return;

    for (const field of fields) {
      delete obj[field];
    }

    for (const key in obj) {
      if (typeof obj[key] === 'object') {
        this.removeFields(obj[key], fields);
      }
    }
  }
}

  success: boolean;
  data?: T;
  errors?: string;
  validationErrors?: any[];

  data?: any;
```

### Usage Examples

```typescript
// Example 1: Validate proposal
const decoder = new ConstrainedJSONDecoder();
const schema = ConstrainedJSONDecoder.getTemplate('action_proposal');

const result = decoder.validate(proposalData, schema);
if (result.success) {
  console.log('Valid proposal:', result.data);
} else {
  console.error('Validation errors:', result.errors);
}

// Example 2: Parse with fallback
const parseResult = decoder.parseWithFallback(llmResponse);
if (parseResult.success) {
  if (parseResult.wasFixed) {
    console.warn('JSON was auto-fixed');
  }
  console.log('Parsed data:', parseResult.data);
} else {
  console.error('Parse failed:', parseResult.error);
}

// Example 3: Complete partial JSON
const partial = '{"intent":"Create ticket","operation":"create_ticket"';
const completed = decoder.completePartialJSON(partial);
console.log('Completed:', completed);
// Output: {"intent":"Create ticket","operation":"create_ticket"}

// Example 4: Simplify schema for LLM
const fullSchema = ACTION_PROPOSAL_SCHEMA;
const simplified = decoder.simplifySchema(fullSchema);
```

### Validation Error Reporting

```typescript
export class ValidationReporter {
  static formatErrors(errors: any[]): string[] {
    return errors.map(error => {
      const path = error.instancePath || 'root';
      const message = error.message || 'unknown error';
      return `${path}: ${message}`;
    });
  }

  static getActionableMessage(errors: any[]): string {
    const formatted = this.formatErrors(errors);
    return `Validation failed:\n${formatted.join('\n')}\n\nPlease check the generated JSON structure.`;
  }

  static hasRequiredFieldErrors(errors: any[]): boolean {
    return errors.some(e => e.keyword === 'required');
  }

  static hasTypeErrors(errors: any[]): boolean {
    return errors.some(e => e.keyword === 'type');
  }
}
```

## Test Cases

```typescript
describe('ConstrainedJSONDecoder', () => {
  let decoder: ConstrainedJSONDecoder;

  beforeEach(() => {
    decoder = new ConstrainedJSONDecoder();
  });

  describe('validate', () => {
    it('should validate correct proposal', () => {
      const schema = ConstrainedJSONDecoder.getTemplate('action_proposal');
      const data = {
        intent: 'Create ticket',
        operation: 'create_ticket',
        data_domain: 'enterprise',
        data_classification: 'internal',
        reversibility: 'FULL',
        confidence: 0.9,
        requires_confirmation: true,
        risk_level: 'medium'
      };

      const result = decoder.validate(data, schema as any);
      expect(result.success).toBe(true);
    });

    it('should reject invalid proposal', () => {
      const schema = ConstrainedJSONDecoder.getTemplate('action_proposal');
      const data = {
        intent: 'Create ticket',
        // Missing required fields
      };

      const result = decoder.validate(data, schema as any);
      expect(result.success).toBe(false);
      expect(result.errors).toBeDefined();
    });
  });

  describe('parseWithFallback', () => {
    it('should parse valid JSON', () => {
      const json = '{"key":"value"}';
      const result = decoder.parseWithFallback(json);
      expect(result.success).toBe(true);
      expect(result.data).toEqual({ key: 'value' });
    });

    it('should fix trailing commas', () => {
      const json = '{"key":"value",}';
      const result = decoder.parseWithFallback(json);
      expect(result.success).toBe(true);
      expect(result.wasFixed).toBe(true);
    });

    it('should fix single quotes', () => {
      const json = "{'key':'value'}";
      const result = decoder.parseWithFallback(json);
      expect(result.success).toBe(true);
    });
  });

  describe('completePartialJSON', () => {
    it('should close unclosed braces', () => {
      const partial = '{"key":"value"';
      const completed = decoder.completePartialJSON(partial);
      expect(completed).toBe('{"key":"value"}');
    });

    it('should close unclosed arrays', () => {
      const partial = '["item1","item2"';
      const completed = decoder.completePartialJSON(partial);
      expect(completed).toBe('["item1","item2"]');
    });

    it('should handle nested structures', () => {
      const partial = '{"outer":{"inner":"value"';
      const completed = decoder.completePartialJSON(partial);
      expect(completed).toBe('{"outer":{"inner":"value"}}');
    });
  });

  describe('simplifySchema', () => {
    it('should remove unnecessary fields', () => {
      const schema = {
        type: 'object',
        title: 'Test Schema',
        description: 'A test schema',
        examples: [{ test: true }],
        properties: {
          field: { type: 'string' }
        }
      };

      const simplified = decoder.simplifySchema(schema);
      expect(simplified).not.toHaveProperty('title');
      expect(simplified).not.toHaveProperty('examples');
    });
  });
});
```
