## Technical Notes

### MCP Server Registry

```typescript
// src/main/services/mcp/mcp-server-registry.ts
import { MCPClient } from './mcp-client';
import { EventEmitter } from 'events';

interface ServerConfig {
  id: string;
  name: string;
  command: string;
  args: string[];
  enabled: boolean;
  autoStart: boolean;
  timeout: number;
}

interface RegisteredServer {
  config: ServerConfig;
  client: MCPClient | null;
  status: ServerStatus;
  tools: MCPTool[];
  lastConnected: ISO8601String | null;
  reconnectAttempts: number;
}

type ServerStatus = 'disconnected' | 'connecting' | 'connected' | 'error';

export class MCPServerRegistry extends EventEmitter {
  private servers: Map<string, RegisteredServer> = new Map();
  private db: Database;

  constructor(db: Database) {
    super();
    this.db = db;
  }

  async initialize(): Promise<void> {
    // Load server configurations
    await this.loadConfigurations();

    // Start auto-start servers
    await this.startAutoStartServers();

    console.log(`MCP Server Registry initialized with ${this.servers.size} servers`);
  }

  async registerServer(config: ServerConfig): Promise<void> {
    // Validate config
    if (!config.id || !config.command) {
      throw new Error('Invalid server configuration');
    }

    // Check for duplicates
    if (this.servers.has(config.id)) {
      throw new Error(`Server ${config.id} already registered`);
    }

    // Create registered server
    const server: RegisteredServer = {
      config,
      client: null,
      status: 'disconnected',
      tools: [],
      lastConnected: null,
      reconnectAttempts: 0,
    };

    this.servers.set(config.id, server);

    // Persist configuration
    await this.saveConfiguration(config);

    // Auto-start if enabled
    if (config.enabled && config.autoStart) {
      await this.connectServer(config.id);
    }

    this.emit('server:registered', config.id);
  }

  async unregisterServer(serverId: string): Promise<void> {
    const server = this.servers.get(serverId);
    if (!server) {
      throw new Error(`Server ${serverId} not found`);
    }

    // Disconnect if connected
    if (server.client) {
      await server.client.disconnect();
    }

    // Remove from registry
    this.servers.delete(serverId);

    // Remove from database
    await this.db.run('DELETE FROM mcp_servers WHERE id = ?', [serverId]);

    this.emit('server:unregistered', serverId);
  }

  async connectServer(serverId: string): Promise<void> {
    const server = this.servers.get(serverId);
    if (!server) {
      throw new Error(`Server ${serverId} not found`);
    }

    if (server.status === 'connected' || server.status === 'connecting') {
      return;
    }

    try {
      server.status = 'connecting';
      this.emit('server:connecting', serverId);

      // Create client
      const client = new MCPClient({
        serverCommand: server.config.command,
        serverArgs: server.config.args,
        timeout: server.config.timeout,
        maxRetries: 3,
      });

      // Setup event listeners
      client.on('disconnected', () => {
        this.handleServerDisconnect(serverId);
      });

      // Connect
      await client.connect();

      // Update server
      server.client = client;
      server.status = 'connected';
      server.lastConnected = new Date().toISOString();
      server.reconnectAttempts = 0;

      // Load tools
      server.tools = await client.listTools();

      this.emit('server:connected', serverId, server.tools);
    } catch (error) {
      server.status = 'error';
      console.error(`Failed to connect to server ${serverId}:`, error);
      this.emit('server:error', serverId, error);

      // Schedule reconnect
      this.scheduleReconnect(serverId);
    }
  }

  async disconnectServer(serverId: string): Promise<void> {
    const server = this.servers.get(serverId);
    if (!server || !server.client) {
      return;
    }

    await server.client.disconnect();
    server.client = null;
    server.status = 'disconnected';

    this.emit('server:disconnected', serverId);
  }

  getServer(serverId: string): RegisteredServer | undefined {
    return this.servers.get(serverId);
  }

  getAllServers(): ServerInfo[] {
    return Array.from(this.servers.values()).map((s) => ({
      id: s.config.id,
      name: s.config.name,
      status: s.status,
      toolCount: s.tools.length,
      lastConnected: s.lastConnected,
      enabled: s.config.enabled,
    }));
  }

  getConnectedServers(): RegisteredServer[] {
    return Array.from(this.servers.values()).filter((s) => s.status === 'connected');
  }

  async getAllTools(): Promise<ToolInfo[]> {
    const tools: ToolInfo[] = [];

    for (const server of this.servers.values()) {
      if (server.status === 'connected' && server.tools.length > 0) {
        for (const tool of server.tools) {
          tools.push({
            serverId: server.config.id,
            serverName: server.config.name,
            name: tool.name,
            description: tool.description,
            inputSchema: tool.inputSchema,
          });
        }
      }
    }

    return tools;
  }

  async findTool(toolName: string): Promise<ToolLocation | null> {
    for (const server of this.servers.values()) {
      if (server.status === 'connected') {
        const tool = server.tools.find((t) => t.name === toolName);
        if (tool) {
          return {
            serverId: server.config.id,
            client: server.client!,
            tool,
          };
        }
      }
    }

    return null;
  }

  async callTool(serverId: string, toolName: string, args: object): Promise<any> {
    const server = this.servers.get(serverId);
    if (!server || !server.client) {
      throw new Error(`Server ${serverId} not connected`);
    }

    return server.client.callTool(toolName, args);
  }

  // Private methods

  private async loadConfigurations(): Promise<void> {
    // Load built-in configurations
    const builtinConfigs = this.getBuiltinConfigurations();
    for (const config of builtinConfigs) {
      await this.registerServer(config);
    }

    // Load custom configurations from database
    const rows = await this.db.all('SELECT * FROM mcp_servers');
    for (const row of rows) {
      const config: ServerConfig = {
        id: row.id,
        name: row.name,
        command: row.command,
        args: JSON.parse(row.args),
        enabled: row.enabled === 1,
        autoStart: row.auto_start === 1,
        timeout: row.timeout,
      };

      // Skip if already registered (builtin)
      if (!this.servers.has(config.id)) {
        await this.registerServer(config);
      }
    }
  }

  private getBuiltinConfigurations(): ServerConfig[] {
    return [
      {
        id: 'calendar',
        name: 'Calendar',
        command: 'npx',
        args: ['-y', '@modelcontextprotocol/server-calendar'],
        enabled: true,
        autoStart: true,
        timeout: 30000,
      },
      {
        id: 'filesystem',
        name: 'Filesystem',
        command: 'npx',
        args: ['-y', '@modelcontextprotocol/server-filesystem', process.env.HOME || '/'],
        enabled: true,
        autoStart: true,
        timeout: 30000,
      },
    ];
  }

  private async startAutoStartServers(): Promise<void> {
    const autoStartServers = Array.from(this.servers.values()).filter(
      (s) => s.config.enabled && s.config.autoStart
    );

    await Promise.all(autoStartServers.map((s) => this.connectServer(s.config.id)));
  }

  private async saveConfiguration(config: ServerConfig): Promise<void> {
    await this.db.run(
      `
      INSERT OR REPLACE INTO mcp_servers (
        id, name, command, args, enabled, auto_start, timeout
      ) VALUES (?, ?, ?, ?, ?, ?, ?)
    `,
      [
        config.id,
        config.name,
        config.command,
        JSON.stringify(config.args),
        config.enabled ? 1 : 0,
        config.autoStart ? 1 : 0,
        config.timeout,
      ]
    );
  }

  private handleServerDisconnect(serverId: string): void {
    const server = this.servers.get(serverId);
    if (!server) return;

    server.status = 'disconnected';
    server.client = null;

    this.emit('server:disconnected', serverId);

    // Attempt reconnect if auto-start enabled
    if (server.config.enabled && server.config.autoStart) {
      this.scheduleReconnect(serverId);
    }
  }

  private scheduleReconnect(serverId: string): void {
    const server = this.servers.get(serverId);
    if (!server) return;

    server.reconnectAttempts++;

    // Exponential backoff
    const delay = Math.min(1000 * Math.pow(2, server.reconnectAttempts), 60000);

    console.log(`Scheduling reconnect for ${serverId} in ${delay}ms`);

    setTimeout(() => {
      this.connectServer(serverId).catch(console.error);
    }, delay);
  }

  async shutdown(): Promise<void> {
    // Disconnect all servers
    await Promise.all(Array.from(this.servers.keys()).map((id) => this.disconnectServer(id)));
  }
}

// Types
export interface ServerInfo {
  id: string;
  name: string;
  status: ServerStatus;
  toolCount: number;
  lastConnected: ISO8601String | null;
  enabled: boolean;
}

export interface ToolInfo {
  serverId: string;
  serverName: string;
  name: string;
  description: string;
  inputSchema: object;
}

interface ToolLocation {
  serverId: string;
  client: MCPClient;
  tool: MCPTool;
}
```

## Test Cases

```typescript
describe('MCPServerRegistry', () => {
  let registry: MCPServerRegistry;
  let db: Database;

  beforeEach(async () => {
    db = await setupTestDatabase();
    registry = new MCPServerRegistry(db);
    await registry.initialize();
  });

  afterEach(async () => {
    await registry.shutdown();
  });

  it('should load builtin servers', () => {
    const servers = registry.getAllServers();
    expect(servers.length).toBeGreaterThan(0);
    expect(servers.some((s) => s.id === 'calendar')).toBe(true);
  });

  it('should register custom server', async () => {
    await registry.registerServer({
      id: 'custom',
      name: 'Custom Server',
      command: 'node',
      args: ['server.js'],
      enabled: true,
      autoStart: false,
      timeout: 30000,
    });

    const server = registry.getServer('custom');
    expect(server).toBeDefined();
  });

  it('should connect to server', async () => {
    await registry.connectServer('calendar');
    const server = registry.getServer('calendar');
    expect(server?.status).toBe('connected');
  });

  it('should aggregate tools from all servers', async () => {
    await registry.connectServer('calendar');
    const tools = await registry.getAllTools();
    expect(tools.length).toBeGreaterThan(0);
  });

  it('should find tool by name', async () => {
    await registry.connectServer('calendar');
    const location = await registry.findTool('get_events');
    expect(location).toBeDefined();
    expect(location?.tool.name).toBe('get_events');
  });
});
```
