## Technical Notes

```typescript
// tests/integration/mcp-calendar.test.ts
describe('MCP Calendar Integration', () => {
  let registry: MCPServerRegistry;

  beforeAll(async () => {
    registry = new MCPServerRegistry(db);
    await registry.initialize();
    await registry.connectServer('calendar');
  });

  it('should create calendar event', async () => {
    const result = await registry.callTool('calendar', 'create_event', {
      title: 'Test Meeting',
      start: '2026-01-29T10:00:00Z',
      end: '2026-01-29T11:00:00Z',
    });

    expect(result).toHaveProperty('id');
  });

  it('should handle errors gracefully', async () => {
    await expect(registry.callTool('calendar', 'invalid_tool', {})).rejects.toThrow();
  });
});
```
