## Technical Notes

### Audit Service Implementation

```typescript
// src/main/services/audit/audit.service.ts
import * as crypto from 'crypto';

interface AuditRecord {
  id: string;
  timestamp: ISO8601String;
  action_id: string;
  proposal_id: string;
  operation: OperationType;
  status: ActionOutcome;
  actor: ActorType;
  actor_id: string;
  input_hash: string;
  content_hash: string;
  data: object;
  rollback_of: string | null;
}

interface AuditFilter {
  startDate?: ISO8601String;
  endDate?: ISO8601String;
  operation?: OperationType;
  status?: ActionOutcome;
  actor?: ActorType;
  proposalId?: string;
  actionId?: string;
  limit?: number;
  offset?: number;
}

export class AuditService {
  private db: Database;

  constructor(db: Database) {
    this.db = db;
  }

  async initialize(): Promise<void> {
    // Ensure audit_log table exists
    await this.createTables();

    console.log('Audit service initialized');
  }

  async log(
    record: Omit<AuditRecord, 'id' | 'timestamp' | 'input_hash' | 'content_hash' | 'actor_id'>
  ): Promise<AuditRecord> {
    const id = crypto.randomUUID();
    const timestamp = new Date().toISOString();

    // Hash sensitive input data
    const inputHash = this.hashInput(record.data);

    // Compute content hash for integrity
    const contentHash = this.computeContentHash({
      id,
      timestamp,
      action_id: record.action_id,
      proposal_id: record.proposal_id,
      operation: record.operation,
      status: record.status,
    });

    const fullRecord: AuditRecord = {
      id,
      timestamp,
      ...record,
      actor_id: 'system', // TODO: Get from context
      input_hash: inputHash,
      content_hash: contentHash,
      rollback_of: record.rollback_of || null,
    };

    // Insert into append-only log
    await this.insertRecord(fullRecord);

    return fullRecord;
  }

  async query(filter: AuditFilter = {}): Promise<AuditRecord[]> {
    let sql = 'SELECT * FROM audit_log WHERE 1=1';
    const params: any[] = [];

    // Apply filters
    if (filter.startDate) {
      sql += ' AND timestamp >= ?';
      params.push(filter.startDate);
    }

    if (filter.endDate) {
      sql += ' AND timestamp <= ?';
      params.push(filter.endDate);
    }

    if (filter.operation) {
      sql += ' AND operation = ?';
      params.push(filter.operation);
    }

    if (filter.status) {
      sql += ' AND status = ?';
      params.push(filter.status);
    }

    if (filter.actor) {
      sql += ' AND actor = ?';
      params.push(filter.actor);
    }

    if (filter.proposalId) {
      sql += ' AND proposal_id = ?';
      params.push(filter.proposalId);
    }

    if (filter.actionId) {
      sql += ' AND action_id = ?';
      params.push(filter.actionId);
    }

    // Order by timestamp descending (most recent first)
    sql += ' ORDER BY timestamp DESC';

    // Apply pagination
    if (filter.limit) {
      sql += ' LIMIT ?';
      params.push(filter.limit);

      if (filter.offset) {
        sql += ' OFFSET ?';
        params.push(filter.offset);
      }
    }

    const rows = await this.db.all(sql, params);

    return rows.map((row) => ({
      id: row.id,
      timestamp: row.timestamp,
      action_id: row.action_id,
      proposal_id: row.proposal_id,
      operation: row.operation,
      status: row.status,
      actor: row.actor,
      actor_id: row.actor_id,
      input_hash: row.input_hash,
      content_hash: row.content_hash,
      data: JSON.parse(row.data),
      rollback_of: row.rollback_of,
    }));
  }

  async getById(id: string): Promise<AuditRecord | null> {
    const row = await this.db.get('SELECT * FROM audit_log WHERE id = ?', [id]);

    if (!row) return null;

    return {
      id: row.id,
      timestamp: row.timestamp,
      action_id: row.action_id,
      proposal_id: row.proposal_id,
      operation: row.operation,
      status: row.status,
      actor: row.actor,
      actor_id: row.actor_id,
      input_hash: row.input_hash,
      content_hash: row.content_hash,
      data: JSON.parse(row.data),
      rollback_of: row.rollback_of,
    };
  }

  async getActionHistory(actionId: string): Promise<AuditRecord[]> {
    return this.query({ actionId });
  }

  async getProposalHistory(proposalId: string): Promise<AuditRecord[]> {
    return this.query({ proposalId });
  }

  async getStatistics(filter: Partial<AuditFilter> = {}): Promise<AuditStatistics> {
    const records = await this.query(filter);

    const stats: AuditStatistics = {
      total: records.length,
      byStatus: {
        success: 0,
        failure: 0,
        pending: 0,
        rolled_back: 0,
      },
      byOperation: {},
      byActor: {},
    };

    for (const record of records) {
      // Count by status
      stats.byStatus[record.status]++;

      // Count by operation
      stats.byOperation[record.operation] = (stats.byOperation[record.operation] || 0) + 1;

      // Count by actor
      stats.byActor[record.actor] = (stats.byActor[record.actor] || 0) + 1;
    }

    return stats;
  }

  async verifyIntegrity(recordId: string): Promise<boolean> {
    const record = await this.getById(recordId);
    if (!record) return false;

    // Recompute content hash
    const expectedHash = this.computeContentHash({
      id: record.id,
      timestamp: record.timestamp,
      action_id: record.action_id,
      proposal_id: record.proposal_id,
      operation: record.operation,
      status: record.status,
    });

    return record.content_hash === expectedHash;
  }

  async applyRetentionPolicy(retentionDays: number): Promise<number> {
    const cutoffDate = new Date();
    cutoffDate.setDate(cutoffDate.getDate() - retentionDays);

    const result = await this.db.run('DELETE FROM audit_log WHERE timestamp < ?', [
      cutoffDate.toISOString(),
    ]);

    return result.changes || 0;
  }

  // Private methods

  private async createTables(): Promise<void> {
    await this.db.exec(`
      CREATE TABLE IF NOT EXISTS audit_log (
        id TEXT PRIMARY KEY,
        timestamp TEXT NOT NULL,
        action_id TEXT NOT NULL,
        proposal_id TEXT NOT NULL,
        operation TEXT NOT NULL,
        status TEXT NOT NULL,
        actor TEXT NOT NULL,
        actor_id TEXT NOT NULL,
        input_hash TEXT NOT NULL,
        content_hash TEXT NOT NULL,
        data TEXT NOT NULL,
        rollback_of TEXT
      );

      CREATE INDEX IF NOT EXISTS idx_audit_timestamp ON audit_log(timestamp);
      CREATE INDEX IF NOT EXISTS idx_audit_action_id ON audit_log(action_id);
      CREATE INDEX IF NOT EXISTS idx_audit_proposal_id ON audit_log(proposal_id);
      CREATE INDEX IF NOT EXISTS idx_audit_operation ON audit_log(operation);
      CREATE INDEX IF NOT EXISTS idx_audit_status ON audit_log(status);
    `);
  }

  private async insertRecord(record: AuditRecord): Promise<void> {
    await this.db.run(
      `
      INSERT INTO audit_log (
        id, timestamp, action_id, proposal_id, operation,
        status, actor, actor_id, input_hash, content_hash,
        data, rollback_of
      ) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)
    `,
      [
        record.id,
        record.timestamp,
        record.action_id,
        record.proposal_id,
        record.operation,
        record.status,
        record.actor,
        record.actor_id,
        record.input_hash,
        record.content_hash,
        JSON.stringify(record.data),
        record.rollback_of,
      ]
    );
  }

  private hashInput(data: object): string {
    // Hash sensitive input to protect PII while maintaining auditability
    const dataString = JSON.stringify(data);
    return crypto.createHash('sha256').update(dataString).digest('hex');
  }

  private computeContentHash(content: object): string {
    // Compute hash for integrity verification
    const canonical = JSON.stringify(content, Object.keys(content).sort());
    return crypto.createHash('sha256').update(canonical).digest('hex');
  }
}

// Types
interface AuditStatistics {
  total: number;
  byStatus: Record<ActionOutcome, number>;
  byOperation: Record<OperationType, number>;
  byActor: Record<ActorType, number>;
}

export { AuditRecord, AuditFilter };
```

### Usage Examples

```typescript
// Log action execution
await auditService.log({
  action_id: actionId,
  proposal_id: proposal.id,
  operation: proposal.operation,
  status: 'executing',
  actor: 'system',
  data: { plan },
});

// Log success
await auditService.log({
  action_id: actionId,
  proposal_id: proposal.id,
  operation: proposal.operation,
  status: 'success',
  actor: 'system',
  data: { outputs, duration },
});

// Query recent activity
const recentActivity = await auditService.query({
  startDate: new Date(Date.now() - 24 * 60 * 60 * 1000).toISOString(),
  limit: 50,
});

// Get action history
const actionHistory = await auditService.getActionHistory(actionId);

// Get statistics
const stats = await auditService.getStatistics({
  startDate: '2026-01-01T00:00:00Z',
});

// Verify integrity
const isValid = await auditService.verifyIntegrity(recordId);
```

## Test Cases

```typescript
describe('AuditService', () => {
  let auditService: AuditService;
  let db: Database;

  beforeEach(async () => {
    db = await setupTestDatabase();
    auditService = new AuditService(db);
    await auditService.initialize();
  });

  it('should log audit record', async () => {
    const record = await auditService.log({
      action_id: 'test-action',
      proposal_id: 'test-proposal',
      operation: 'create_ticket',
      status: 'success',
      actor: 'system',
      data: { test: true },
    });

    expect(record.id).toBeDefined();
    expect(record.timestamp).toBeDefined();
  });

  it('should query with filters', async () => {
    await auditService.log({
      action_id: 'action-1',
      proposal_id: 'proposal-1',
      operation: 'create_ticket',
      status: 'success',
      actor: 'system',
      data: {},
    });

    const results = await auditService.query({
      operation: 'create_ticket',
    });

    expect(results.length).toBeGreaterThan(0);
    expect(results[0].operation).toBe('create_ticket');
  });

  it('should get action history', async () => {
    const actionId = 'test-action';

    await auditService.log({
      action_id: actionId,
      proposal_id: 'proposal-1',
      operation: 'create_ticket',
      status: 'executing',
      actor: 'system',
      data: {},
    });

    await auditService.log({
      action_id: actionId,
      proposal_id: 'proposal-1',
      operation: 'create_ticket',
      status: 'success',
      actor: 'system',
      data: {},
    });

    const history = await auditService.getActionHistory(actionId);
    expect(history.length).toBe(2);
  });

  it('should verify integrity', async () => {
    const record = await auditService.log({
      action_id: 'test',
      proposal_id: 'test',
      operation: 'create_ticket',
      status: 'success',
      actor: 'system',
      data: {},
    });

    const isValid = await auditService.verifyIntegrity(record.id);
    expect(isValid).toBe(true);
  });

  it('should apply retention policy', async () => {
    // Insert old record
    await auditService.log({
      action_id: 'old',
      proposal_id: 'old',
      operation: 'create_ticket',
      status: 'success',
      actor: 'system',
      data: {},
    });

    const deleted = await auditService.applyRetentionPolicy(0);
    expect(deleted).toBeGreaterThan(0);
  });
});
```
