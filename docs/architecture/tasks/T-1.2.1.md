### Main Process Entry Point

```typescript
// src/main/index.ts
import { app, BrowserWindow } from 'electron';
import { createMainWindow } from './window';
import { registerGlobalShortcuts } from './shortcuts';
import { setupApplicationMenu } from './menu';
import { initializeAutoUpdater } from './updater';
import { createSystemTray } from './tray';

// Security: Disable insecure features
app.commandLine.appendSwitch('disable-features', 'OutOfBlinkCors');

// Single instance lock
const gotTheLock = app.requestSingleInstanceLock();
if (!gotTheLock) {
  app.quit();
} else {
  let mainWindow: BrowserWindow | null = null;

  app.on('second-instance', () => {
    if (mainWindow) {
      if (mainWindow.isMinimized()) mainWindow.restore();
      mainWindow.focus();
    }
  });

  app.on('ready', async () => {
    // Initialize main window
    mainWindow = createMainWindow();

    // Setup application components
    setupApplicationMenu(mainWindow);
    registerGlobalShortcuts(mainWindow);
    createSystemTray(mainWindow);

    // Initialize auto-updater
    if (process.env.NODE_ENV === 'production') {
      initializeAutoUpdater(mainWindow);
    }

    // Load renderer
    if (VITE_DEV_SERVER_URL) {
      mainWindow.loadURL(VITE_DEV_SERVER_URL);
    } else {
      mainWindow.loadFile(path.join(__dirname, '../renderer/index.html'));
    }
  });

  app.on('window-all-closed', () => {
    // On macOS, keep app running when windows closed
    if (process.platform !== 'darwin') {
      app.quit();
    }
  });

  app.on('activate', () => {
    // Re-create window on macOS when dock icon clicked
    if (BrowserWindow.getAllWindows().length === 0) {
      mainWindow = createMainWindow();
    }
  });

  app.on('before-quit', () => {
    // Cleanup
    if (mainWindow) {
      mainWindow.removeAllListeners('close');
    }
  });
}
```

### Window Configuration

```typescript
// src/main/window.ts
import { BrowserWindow, shell } from 'electron';

export function createMainWindow(): BrowserWindow {
  const window = new BrowserWindow({
    width: 1200,
    height: 800,
    minWidth: 800,
    minHeight: 600,
    title: 'Lumos',
    titleBarStyle: 'hiddenInset', // macOS native title bar
    trafficLightPosition: { x: 16, y: 16 },
    webPreferences: {
      // Security: Enable all protections
      contextIsolation: true,
      nodeIntegration: false,
      sandbox: true,
      webSecurity: true,
      allowRunningInsecureContent: false,

      // Preload script for IPC bridge
      preload: path.join(__dirname, '../preload/index.js'),

      // Performance
      backgroundThrottling: false
    },
    backgroundColor: '#1a1a1a',
    show: false // Show when ready to avoid flash
  });

  // Show window when ready
  window.once('ready-to-show', () => {
    window.show();
  });

  // Security: Prevent navigation to external URLs
  window.webContents.on('will-navigate', (event, url) => {
    const parsedUrl = new URL(url);
    if (parsedUrl.origin !== VITE_DEV_SERVER_URL) {
      event.preventDefault();
    }
  });

  // Security: Open external links in browser
  window.webContents.setWindowOpenHandler(({ url }) => {
    shell.openExternal(url);
    return { action: 'deny' };
  });

  // Handle window close
  window.on('close', (event) => {
    if (process.platform === 'darwin' && !app.isQuitting) {
      event.preventDefault();
      window.hide();
    }
  });

  return window;
}
```

### Global Shortcuts

```typescript
// src/main/shortcuts.ts
import { globalShortcut, BrowserWindow } from 'electron';

export function registerGlobalShortcuts(mainWindow: BrowserWindow): void {
  // Toggle main window
  globalShortcut.register('CommandOrControl+Shift+L', () => {
    if (mainWindow.isVisible()) {
      mainWindow.hide();
    } else {
      mainWindow.show();
      mainWindow.focus();
    }
  });

  // New chat
  globalShortcut.register('CommandOrControl+N', () => {
    if (mainWindow.isVisible()) {
      mainWindow.webContents.send('new-chat');
    }
  });
}

export function unregisterGlobalShortcuts(): void {
  globalShortcut.unregisterAll();
}
```

### System Tray

```typescript
// src/main/tray.ts
import { Tray, Menu, BrowserWindow, nativeImage } from 'electron';

let tray: Tray | null = null;

export function createSystemTray(mainWindow: BrowserWindow): Tray {
  // Create tray icon
  const icon = nativeImage.createFromPath(
    path.join(__dirname, '../../assets/tray-icon.png')
  );

  tray = new Tray(icon.resize({ width: 16, height: 16 }));
  tray.setToolTip('Lumos AI Assistant');

  // Create context menu
  const contextMenu = Menu.buildFromTemplate([
    {
      label: 'Show Lumos',
      click: () => {
        mainWindow.show();
        mainWindow.focus();
      }
    },
    { type: 'separator' },
    {
      label: 'Quit',
      click: () => {
        app.quit();
      }
    }
  ]);

  tray.setContextMenu(contextMenu);

  // Click to show window
  tray.on('click', () => {
    mainWindow.show();
    mainWindow.focus();
  });

  return tray;
}
```

## Security Best Practices

1. **Context Isolation**: Enabled to prevent renderer access to Node.js
2. **Sandbox**: Enabled for additional renderer protection
3. **No Node Integration**: Renderer cannot access Node.js APIs directly
4. **CSP Headers**: Content Security Policy enforced
5. **External Links**: Opened in system browser, not in app
6. **Navigation Prevention**: Block navigation to untrusted URLs
