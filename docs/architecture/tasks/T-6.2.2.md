## Technical Notes

```typescript
// src/renderer/views/PermissionsView.tsx
export default function PermissionsView() {
  const { invoke } = useIPC();
  const [trustLevels, setTrustLevels] = useState<Map<OperationType, TrustLevelRecord>>();
  const [attestationsDue, setAttestationsDue] = useState<AttestationItem[]>([]);

  useIPCEvent('trust:promoted', () => {
    loadTrustLevels();
  });

  const handlePromote = async (operation: OperationType) => {
    const result = await invoke('trust:promote', operation);
    if (result.allowed) {
      loadTrustLevels();
    }
  };

  const handleAttestation = async (operations: OperationType[]) => {
    await invoke('attestation:confirm', operations);
    loadTrustLevels();
  };

  return (
    <div className="p-6 space-y-4">
      <h1 className="text-2xl font-bold">Permissions</h1>

      {attestationsDue.length > 0 && (
        <AttestationAlert items={attestationsDue} onConfirm={handleAttestation} />
      )}

      <TrustLevelTable
        trustLevels={trustLevels}
        onPromote={handlePromote}
      />
    </div>
  );
}
```
