### Database Connection Manager

```typescript
import Database from 'better-sqlite3';
import type { Database as DatabaseType } from 'better-sqlite3';

const DB_PATH = process.env.LUMOS_DB_PATH || './data/lumos.db';

class DatabaseManager {
  private static instance: DatabaseType | null = null;

  static getConnection(): DatabaseType {
    if (!this.instance) {
      this.instance = new Database(DB_PATH, {
        verbose: process.env.NODE_ENV === 'development' ? console.log : undefined,
      });

      // Configure for optimal performance and safety
      this.instance.pragma('journal_mode = WAL');
      this.instance.pragma('synchronous = NORMAL');
      this.instance.pragma('foreign_keys = ON');
      this.instance.pragma('cache_size = -64000'); // 64MB cache
    }
    return this.instance;
  }

  static transaction<T>(fn: (db: DatabaseType) => T): T {
    const db = this.getConnection();
    return db.transaction(fn)(db);
  }

  static close(): void {
    if (this.instance) {
      this.instance.close();
      this.instance = null;
    }
  }

  static healthCheck(): boolean {
    try {
      const db = this.getConnection();
      db.prepare('SELECT 1').get();
      return true;
    } catch {
      return false;
    }
  }
}

export { DatabaseManager };
```

### Repository Pattern

```typescript
import type { Database, Statement } from 'better-sqlite3';
import type {
  Skill,
  SkillStatus,
  TrustLevel,
  SkillOrigin,
  DataDomain,
  ConflictResolution,
} from '@/types';

class SkillRepository {
  private db: Database;

  // Prepared statements (cached)
  private selectById: Statement;
  private selectByStatus: Statement;
  private insertStmt: Statement;
  private updateStatusStmt: Statement;

  constructor(db: Database) {
    this.db = db;
    this.selectById = db.prepare('SELECT * FROM skills WHERE id = ?');
    this.selectByStatus = db.prepare('SELECT * FROM skills WHERE status = ?');
    this.insertStmt = db.prepare(`
      INSERT INTO skills (
        id, name, description, version_hash, status, origin,
        enabled_operations, policy_constraints, required_trust_level,
        priority, data_domain, conflict_resolution, proposal_schema,
        rollback_specification, created_at, activated_at, activated_by
      ) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)
    `);
    this.updateStatusStmt = db.prepare('UPDATE skills SET status = ? WHERE id = ?');
  }

  findById(id: string): Skill | null {
    const row = this.selectById.get(id) as Record<string, unknown> | undefined;
    return row ? this.deserialize(row) : null;
  }

  findByStatus(status: SkillStatus): Skill[] {
    const rows = this.selectByStatus.all(status) as Record<string, unknown>[];
    return rows.map((row) => this.deserialize(row));
  }

  findActive(): Skill[] {
    return this.findByStatus('active');
  }

  create(skill: Skill): void {
    this.insertStmt.run(
      skill.id,
      skill.name,
      skill.description,
      skill.version_hash,
      skill.status,
      skill.origin,
      JSON.stringify(skill.enabled_operations),
      JSON.stringify(skill.policy_constraints),
      skill.required_trust_level,
      skill.priority,
      skill.data_domain,
      skill.conflict_resolution,
      JSON.stringify(skill.proposal_schema),
      JSON.stringify(skill.rollback_specification),
      skill.created_at,
      skill.activated_at,
      skill.activated_by
    );
  }

  updateStatus(id: string, status: SkillStatus): void {
    this.updateStatusStmt.run(status, id);
  }

  private deserialize(row: Record<string, unknown>): Skill {
    return {
      id: row.id as string,
      name: row.name as string,
      description: row.description as string,
      version_hash: row.version_hash as string,
      status: row.status as SkillStatus,
      origin: row.origin as SkillOrigin,
      enabled_operations: JSON.parse(row.enabled_operations as string),
      policy_constraints: JSON.parse(row.policy_constraints as string),
      required_trust_level: row.required_trust_level as TrustLevel,
      priority: row.priority as number,
      data_domain: row.data_domain as DataDomain | 'both',
      conflict_resolution: row.conflict_resolution as ConflictResolution,
      proposal_schema: JSON.parse(row.proposal_schema as string),
      rollback_specification: JSON.parse(row.rollback_specification as string),
      created_at: row.created_at as string,
      activated_at: row.activated_at as string | null,
      activated_by: row.activated_by as string | null,
    };
  }
}

export { SkillRepository };
```

### YAML Skill Loader

```typescript
import { readFile, readdir } from 'fs/promises';
import { join } from 'path';
import { parse as parseYaml } from 'yaml';
import Ajv from 'ajv';
import { createHash } from 'crypto';
import type { Skill } from '@/types';
import { SKILL_SCHEMA } from '@/types/schemas';

const ajv = new Ajv();
const validateSkill = ajv.compile(SKILL_SCHEMA);

class SkillFileLoader {
  async loadSkills(directory: string): Promise<Skill[]> {
    const files = await readdir(directory);
    const yamlFiles = files.filter((f) => f.endsWith('.yaml') || f.endsWith('.yml'));

    const skills: Skill[] = [];
    for (const file of yamlFiles) {
      const content = await readFile(join(directory, file), 'utf-8');
      const parsed = parseYaml(content);

      // Validate against schema
      const valid = validateSkill(parsed);
      if (!valid) {
        throw new SkillValidationError(file, validateSkill.errors ?? []);
      }

      // Calculate version hash
      const versionHash = this.calculateHash(parsed);

      skills.push({
        ...parsed,
        version_hash: versionHash,
      } as Skill);
    }

    return skills;
  }

  private calculateHash(skill: Record<string, unknown>): string {
    const canonical = JSON.stringify(skill, Object.keys(skill).sort());
    return createHash('sha256').update(canonical).digest('hex');
  }
}

export { SkillFileLoader };
```

### Error Handling

```typescript
class StorageError extends Error {
  constructor(
    message: string,
    public cause?: Error
  ) {
    super(message);
    this.name = 'StorageError';
  }
}

class DatabaseConnectionError extends StorageError {
  constructor(message: string, cause?: Error) {
    super(message, cause);
    this.name = 'DatabaseConnectionError';
  }
}

class QueryError extends StorageError {
  constructor(
    message: string,
    public query?: string,
    cause?: Error
  ) {
    super(message, cause);
    this.name = 'QueryError';
  }
}

class ValidationError extends StorageError {
  constructor(
    message: string,
    public errors?: unknown[]
  ) {
    super(message);
    this.name = 'ValidationError';
  }
}

class SkillValidationError extends ValidationError {
  constructor(
    public filename: string,
    errors: unknown[]
  ) {
    super(`Skill validation failed for ${filename}`, errors);
    this.name = 'SkillValidationError';
  }
}

export { StorageError, DatabaseConnectionError, QueryError, ValidationError, SkillValidationError };
```

```typescript
      repo.create(skill);

      repo.updateStatus('update-test', 'active');
```

### Transaction Tests

```typescript
// tests/storage/database.transaction.test.ts
import { describe, it, expect, beforeEach, afterEach } from 'vitest';
import Database from 'better-sqlite3';
import { DatabaseManager } from '@/main/storage/database';
import { SkillRepository } from '@/main/storage/repositories/skill.repository';
import { createTestSkill, setupTestDatabase, cleanupTestDatabase } from '../helpers';

describe('DatabaseManager Transactions', () => {
  let db: Database.Database;

  beforeEach(() => {
    db = setupTestDatabase();
  });

  afterEach(() => {
    cleanupTestDatabase(db);
  });

  it('should commit transaction on success', () => {
    const repo = new SkillRepository(db);

    db.transaction(() => {
      repo.create(createTestSkill({ id: 'tx-1' }));
      repo.create(createTestSkill({ id: 'tx-2' }));
    })();

    expect(repo.findById('tx-1')).not.toBeNull();
    expect(repo.findById('tx-2')).not.toBeNull();
  });

  it('should rollback transaction on error', () => {
    const repo = new SkillRepository(db);

    expect(() => {
      db.transaction(() => {
        repo.create(createTestSkill({ id: 'rollback-1' }));
        throw new Error('Simulated error');
      })();
    }).toThrow('Simulated error');

    // Should be rolled back
    expect(repo.findById('rollback-1')).toBeNull();
  });

  it('should support nested transactions (savepoints)', () => {
    const repo = new SkillRepository(db);

    db.transaction(() => {
      repo.create(createTestSkill({ id: 'outer-1' }));

      try {
        db.transaction(() => {
          repo.create(createTestSkill({ id: 'inner-1' }));
          throw new Error('Inner error');
        })();
      } catch {
        // Inner transaction rolled back
      }

      repo.create(createTestSkill({ id: 'outer-2' }));
    })();

    expect(repo.findById('outer-1')).not.toBeNull();
    expect(repo.findById('outer-2')).not.toBeNull();
    expect(repo.findById('inner-1')).toBeNull();
  });
});
```
