## Technical Notes

### Trust Management Service

```typescript
// src/main/services/governance/trust-management.service.ts
import { TrustLevelStateMachine, TrustState } from './trust-level-state-machine';
import { EventEmitter } from 'events';

export class TrustManagementService extends EventEmitter {
  private stateMachine: TrustLevelStateMachine;
  private trustStates: Map<OperationType, TrustState> = new Map();
  private db: Database; // Storage layer

  constructor(db: Database) {
    super();
    this.db = db;
    this.stateMachine = new TrustLevelStateMachine();
  }

  async initialize(): Promise<void> {
    // Load trust states from database
    await this.loadStates();

    // Setup periodic attestation checks
    this.startAttestationMonitoring();

    console.log(`Initialized trust management for ${this.trustStates.size} operations`);
  }

  /**
   * Get trust level for an operation
   */
  getTrustLevel(operation: OperationType): TrustLevel {
    const state = this.getOrCreateState(operation);
    return state.level;
  }

  /**
   * Get all trust levels
   */
  getAllTrustLevels(): Map<OperationType, TrustLevelRecord> {
    const records = new Map<OperationType, TrustLevelRecord>();

    for (const [operation, state] of this.trustStates) {
      records.set(operation, this.stateToRecord(state));
    }

    return records;
  }

  /**
   * Request trust level promotion
   */
  async promoteTrust(operation: OperationType): Promise<PromotionResult> {
    const state = this.getOrCreateState(operation);

    // Check if promotion is allowed
    if (!this.stateMachine.canPromote(state)) {
      const progress = this.stateMachine.getPromotionProgress(state);
      return {
        allowed: false,
        newLevel: null,
        reason: progress.reason
      };
    }

    // Perform promotion
    const result = this.stateMachine.promote(state, 'User requested promotion');

    if (result.success) {
      // Persist state
      await this.persistState(operation, state);

      // Emit event
      this.emit('trust:promoted', {
        operation,
        oldLevel: result.oldLevel,
        newLevel: result.newLevel
      });

      return {
        allowed: true,
        newLevel: result.newLevel!,
        reason: 'Promotion successful'
      };
    }

    return {
      allowed: false,
      newLevel: null,
      reason: result.error || 'Promotion failed'
    };
  }

  /**
   * Demote trust level
   */
  async demoteTrust(operation: OperationType, targetLevel: TrustLevel): Promise<void> {
    const state = this.getOrCreateState(operation);

    const result = this.stateMachine.demote(
      state,
      targetLevel,
      'Manual demotion by user'
    );

    if (result.success) {
      // Persist state
      await this.persistState(operation, state);

      // Emit event
      this.emit('trust:demoted', {
        operation,
        oldLevel: result.oldLevel,
        newLevel: result.newLevel
      });
    }
  }

  /**
   * Record successful operation
   */
  async recordSuccess(operation: OperationType): Promise<void> {
    const state = this.getOrCreateState(operation);

    this.stateMachine.recordSuccess(state);

    // Check if now eligible for promotion
    if (this.stateMachine.canPromote(state)) {
      this.emit('trust:promotion-eligible', { operation });
    }

    // Persist state
    await this.persistState(operation, state);
  }

  /**
   * Record rollback (automatic demotion)
   */
  async recordRollback(operation: OperationType): Promise<void> {
    const state = this.getOrCreateState(operation);

    const result = this.stateMachine.demoteOnRollback(state);

    if (result.success) {
      // Persist state
      await this.persistState(operation, state);

      // Emit event
      this.emit('trust:auto-demoted', {
        operation,
        oldLevel: result.oldLevel,
        newLevel: result.newLevel,
        reason: 'Rollback occurred'
      });
    }
  }

  /**
   * Check which operations need attestation
   */
  checkAttestationDue(): AttestationCheck[] {
    const checks: AttestationCheck[] = [];

    for (const [operation, state] of this.trustStates) {
      if (this.stateMachine.isAttestationDue(state)) {
        checks.push({
          operation,
          dueDate: state.attestationDue!,
          daysPastDue: this.calculateDaysPastDue(state.attestationDue!)
        });
      }
    }

    return checks;
  }

  /**
   * Confirm attestation for operations
   */
  async confirmAttestation(operations: OperationType[]): Promise<void> {
    for (const operation of operations) {
      const state = this.trustStates.get(operation);
      if (!state) continue;

      if (state.level === 'DELEGATED') {
        this.stateMachine.confirmAttestation(state);
        await this.persistState(operation, state);

        this.emit('trust:attestation-confirmed', { operation });
      }
    }
  }

  /**
   * Get promotion progress for an operation
   */
  getPromotionProgress(operation: OperationType): PromotionProgress {
    const state = this.getOrCreateState(operation);
    return this.stateMachine.getPromotionProgress(state);
  }

  /**
   * Get trust statistics
   */
  getStatistics(): TrustStatistics {
    const stats: TrustStatistics = {
      totalOperations: this.trustStates.size,
      byLevel: {
        OBSERVE: 0,
        SUPERVISED: 0,
        DELEGATED: 0
      },
      attestationDue: 0,
      totalSuccesses: 0,
      totalFailures: 0
    };

    for (const state of this.trustStates.values()) {
      stats.byLevel[state.level]++;
      stats.totalSuccesses += state.successes;
      stats.totalFailures += state.failures;

      if (this.stateMachine.isAttestationDue(state)) {
        stats.attestationDue++;
      }
    }

    return stats;
  }

  // Private methods

  private getOrCreateState(operation: OperationType): TrustState {
    if (!this.trustStates.has(operation)) {
      const newState: TrustState = {
        operation,
        level: 'OBSERVE',
        successes: 0,
        failures: 0,
        lastAttestation: null,
        attestationDue: null,
        lastUpdated: new Date().toISOString(),
        transitionHistory: []
      };

      this.trustStates.set(operation, newState);

      // Persist new state
      this.persistState(operation, newState).catch(console.error);
    }

    return this.trustStates.get(operation)!;
  }

  private async loadStates(): Promise<void> {
    const rows = await this.db.all(`
      SELECT * FROM trust_levels
    `);

    for (const row of rows) {
      const state: TrustState = {
        operation: row.operation,
        level: row.level,
        successes: row.successes,
        failures: row.failures,
        lastAttestation: row.last_attestation,
        attestationDue: row.attestation_due,
        lastUpdated: row.last_updated,
        transitionHistory: JSON.parse(row.transition_history || '[]')
      };

      this.trustStates.set(row.operation, state);
    }
  }

  private async persistState(operation: OperationType, state: TrustState): Promise<void> {
    await this.db.run(`
      INSERT OR REPLACE INTO trust_levels (
        operation,
        level,
        successes,
        failures,
        last_attestation,
        attestation_due,
        last_updated,
        transition_history
      ) VALUES (?, ?, ?, ?, ?, ?, ?, ?)
    `, [
      operation,
      state.level,
      state.successes,
      state.failures,
      state.lastAttestation,
      state.attestationDue,
      state.lastUpdated,
      JSON.stringify(state.transitionHistory)
    ]);
  }

  private stateToRecord(state: TrustState): TrustLevelRecord {
    return {
      operation: state.operation,
      level: state.level,
      successes: state.successes,
      failures: state.failures,
      last_attestation: state.lastAttestation,
      attestation_due: state.attestationDue,
      last_updated: state.lastUpdated
    };
  }

  private calculateDaysPastDue(dueDate: ISO8601String): number {
    const now = new Date();
    const due = new Date(dueDate);
    const diffMs = now.getTime() - due.getTime();
    return Math.floor(diffMs / (1000 * 60 * 60 * 24));
  }

  private startAttestationMonitoring(): void {
    // Check every 24 hours
    const INTERVAL = 24 * 60 * 60 * 1000;

    setInterval(() => {
      const due = this.checkAttestationDue();

      if (due.length > 0) {
        this.emit('trust:attestation-reminder', { due });
      }
    }, INTERVAL);

    // Immediate check on startup
    setTimeout(() => {
      const due = this.checkAttestationDue();
      if (due.length > 0) {
        this.emit('trust:attestation-reminder', { due });
      }
    }, 1000);
  }

  async shutdown(): Promise<void> {
    // Persist all states before shutdown
    for (const [operation, state] of this.trustStates) {
      await this.persistState(operation, state);
    }
  }
}

// Types
export interface TrustLevelRecord {
  operation: OperationType;
  level: TrustLevel;
  successes: number;
  failures: number;
  last_attestation: ISO8601String | null;
  attestation_due: ISO8601String | null;
  last_updated: ISO8601String;
}

export interface PromotionResult {
  allowed: boolean;
  newLevel: TrustLevel | null;
  reason: string;
}

export interface AttestationCheck {
  operation: OperationType;
  dueDate: ISO8601String;
  daysPastDue: number;
}

export interface TrustStatistics {
  totalOperations: number;
  byLevel: Record<TrustLevel, number>;
  attestationDue: number;
  totalSuccesses: number;
  totalFailures: number;
}
```

### Database Schema

```sql
-- Trust levels table
CREATE TABLE IF NOT EXISTS trust_levels (
  operation TEXT PRIMARY KEY,
  level TEXT NOT NULL,
  successes INTEGER DEFAULT 0,
  failures INTEGER DEFAULT 0,
  last_attestation TEXT,
  attestation_due TEXT,
  last_updated TEXT NOT NULL,
  transition_history TEXT
);

CREATE INDEX idx_trust_levels_level ON trust_levels(level);
CREATE INDEX idx_trust_levels_attestation_due ON trust_levels(attestation_due);
```

```typescript
describe('TrustManagementService', () => {
  let service: TrustManagementService;
  let db: Database;

  beforeEach(async () => {
    db = await setupTestDatabase();
    service = new TrustManagementService(db);
    await service.initialize();
  });

  it('should start with OBSERVE level', () => {
    const level = service.getTrustLevel('create_ticket');
    expect(level).toBe('OBSERVE');
  });

  it('should allow immediate promotion to SUPERVISED', async () => {
    const result = await service.promoteTrust('create_ticket');
    expect(result.allowed).toBe(true);
    expect(result.newLevel).toBe('SUPERVISED');
  });

  it('should record successes', async () => {
    await service.recordSuccess('create_ticket');
    const progress = service.getPromotionProgress('create_ticket');
    expect(progress.currentSuccesses).toBe(1);
  });

  it('should demote on rollback', async () => {
    await service.promoteTrust('create_ticket');
    await service.promoteTrust('create_ticket');

    expect(service.getTrustLevel('create_ticket')).toBe('DELEGATED');

    await service.recordRollback('create_ticket');
    expect(service.getTrustLevel('create_ticket')).toBe('SUPERVISED');
  });

  it('should detect attestation due', async () => {
    // Setup DELEGATED with past due attestation

    await service.promoteTrust('create_ticket');
```
