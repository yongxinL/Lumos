## Technical Notes

```typescript
// src/renderer/views/ChatView.tsx
import { useState, useEffect, useRef } from 'react';
import { useIPC, useIPCEvent } from '../hooks/useIPC';
import { MessageList } from '../components/chat/MessageList';
import { ChatInput } from '../components/chat/ChatInput';
import { ProposalCard } from '../components/chat/ProposalCard';

export default function ChatView() {
  const { invoke } = useIPC();
  const [messages, setMessages] = useState<Message[]>([]);
  const [isRecording, setIsRecording] = useState(false);
  const [transcriptionPreview, setTranscriptionPreview] = useState('');
  const messagesEndRef = useRef<HTMLDivElement>(null);

  // Handle new proposals
  useIPCEvent('proposal:new', (proposal: ActionProposal) => {
    setMessages(prev => [
      ...prev,
      {
        id: crypto.randomUUID(),
        type: 'proposal',
        proposal,
        timestamp: new Date().toISOString()
      }
    ]);
  });

  // Handle execution updates
  useIPCEvent('action:executing', (data) => {
    // Update message with execution status
  });

  // Handle transcription updates
  useIPCEvent('transcription:update', (partial: string) => {
    setTranscriptionPreview(partial);
  });

  const handleSubmit = async (text: string) => {
    // Add user message
    setMessages(prev => [
      ...prev,
      {
        id: crypto.randomUUID(),
        type: 'user',
        content: text,
        timestamp: new Date().toISOString()
      }
    ]);

    // Submit to backend
    await invoke('input:submit-text', text);
  };

  const handleStartRecording = async () => {
    setIsRecording(true);
    await invoke('input:start-recording');
  };

  const handleStopRecording = async () => {
    setIsRecording(false);
    const preview = await invoke('input:stop-recording');
    // Show transcription for confirmation
  };

  useEffect(() => {
    messagesEndRef.current?.scrollIntoView({ behavior: 'smooth' });
  }, [messages]);

  return (
    <div className="flex flex-col h-full">
      <div className="flex-1 overflow-auto p-4">
        <MessageList messages={messages} />
        <div ref={messagesEndRef} />
      </div>

      {transcriptionPreview && (
        <div className="p-4 bg-accent border-t">
          <p className="text-sm text-muted-foreground">
            Transcribing: {transcriptionPreview}
          </p>
        </div>
      )}

      <ChatInput
        onSubmit={handleSubmit}
        onStartRecording={handleStartRecording}
        onStopRecording={handleStopRecording}
        isRecording={isRecording}
      />
    </div>
  );
}
```
