## Technical Notes

### Ollama Client Implementation

```typescript
// src/main/services/ai/ollama-client.ts
interface OllamaConfig {
  baseUrl: string;
  timeout: number;
  maxRetries: number;
}

interface GenerateRequest {
  model: string;
  prompt: string;
  format?: object; // JSON Schema for constrained decoding
  stream?: boolean;
  options?: {
    temperature?: number;
    top_p?: number;
    top_k?: number;
  };
}

interface GenerateResponse {
  model: string;
  created_at: string;
  response: string;
  done: boolean;
}

export class OllamaClient {
  private config: OllamaConfig;

  constructor(config?: Partial<OllamaConfig>) {
    this.config = {
      baseUrl: 'http://localhost:11434',
      timeout: 120000, // 2 minutes
      maxRetries: 3,
      ...config,
    };
  }

  async generate(request: GenerateRequest): Promise<string> {
    const response = await this.post('/api/generate', {
      ...request,
      stream: false,
    });

    if (!response.ok) {
      throw new OllamaError(`Generate failed: ${response.statusText}`);
    }

    const data: GenerateResponse = await response.json();
    return data.response;
  }

  async generateStream(
    request: GenerateRequest,
    onChunk: (chunk: string) => void
  ): Promise<string> {
    const response = await this.post('/api/generate', {
      ...request,
      stream: true,
    });

    if (!response.ok) {
      throw new OllamaError(`Generate stream failed: ${response.statusText}`);
    }

    let fullResponse = '';
    const reader = response.body?.getReader();
    const decoder = new TextDecoder();

    if (!reader) {
      throw new OllamaError('No response body');
    }

    while (true) {
      const { done, value } = await reader.read();
      if (done) break;

      const text = decoder.decode(value);
      const lines = text.split('\n').filter((l) => l.trim());

      for (const line of lines) {
        try {
          const data: GenerateResponse = JSON.parse(line);
          if (data.response) {
            fullResponse += data.response;
            onChunk(data.response);
          }
        } catch {
          // Skip malformed JSON
        }
      }
    }

    return fullResponse;
  }

  async listModels(): Promise<ModelInfo[]> {
    const response = await this.get('/api/tags');

    if (!response.ok) {
      throw new OllamaError(`List models failed: ${response.statusText}`);
    }

    const data = await response.json();
    return data.models || [];
  }

  async pullModel(model: string, onProgress?: (progress: number) => void): Promise<void> {
    const response = await this.post('/api/pull', {
      name: model,
      stream: true,
    });

    if (!response.ok) {
      throw new OllamaError(`Pull model failed: ${response.statusText}`);
    }

    const reader = response.body?.getReader();
    const decoder = new TextDecoder();

    if (!reader) return;

    while (true) {
      const { done, value } = await reader.read();
      if (done) break;

      const text = decoder.decode(value);
      const lines = text.split('\n').filter((l) => l.trim());

      for (const line of lines) {
        try {
          const data = JSON.parse(line);
          if (data.total && data.completed && onProgress) {
            const progress = (data.completed / data.total) * 100;
            onProgress(progress);
          }
        } catch {
          // Skip malformed JSON
        }
      }
    }
  }

  async checkHealth(): Promise<boolean> {
    try {
      const response = await fetch(this.config.baseUrl, {
        method: 'GET',
        signal: AbortSignal.timeout(5000),
      });
      return response.ok;
    } catch {
      return false;
    }
  }

  async getModelInfo(model: string): Promise<ModelInfo | null> {
    try {
      const response = await this.post('/api/show', { name: model });
      if (!response.ok) return null;
      return await response.json();
    } catch {
      return null;
    }
  }

  private async get(path: string): Promise<Response> {
    return this.request('GET', path);
  }

  private async post(path: string, body: any): Promise<Response> {
    return this.request('POST', path, body);
  }

  private async request(method: string, path: string, body?: any, attempt = 1): Promise<Response> {
    try {
      const response = await fetch(`${this.config.baseUrl}${path}`, {
        method,
        headers: {
          'Content-Type': 'application/json',
        },
        body: body ? JSON.stringify(body) : undefined,
        signal: AbortSignal.timeout(this.config.timeout),
      });

      return response;
    } catch (error) {
      if (attempt < this.config.maxRetries) {
        await this.delay(1000 * attempt);
        return this.request(method, path, body, attempt + 1);
      }

      if (error instanceof Error && error.name === 'AbortError') {
        throw new OllamaTimeoutError('Request timeout');
      }

      throw new OllamaConnectionError('Failed to connect to Ollama');
    }
  }

  private delay(ms: number): Promise<void> {
    return new Promise((resolve) => setTimeout(resolve, ms));
  }
}

// Types
export interface ModelInfo {
  name: string;
  modified_at: string;
  size: number;
  digest: string;
  details?: {
    format: string;
    family: string;
    parameter_size: string;
    quantization_level: string;
  };
}

// Errors
export class OllamaError extends Error {
  constructor(message: string) {
    super(message);
    this.name = 'OllamaError';
  }
}

export class OllamaTimeoutError extends OllamaError {
  constructor(message: string) {
    super(message);
    this.name = 'OllamaTimeoutError';
  }
}

export class OllamaConnectionError extends OllamaError {
  constructor(message: string) {
    super(message);
    this.name = 'OllamaConnectionError';
  }
}
```

## Test Cases

```typescript
describe('OllamaClient', () => {
  let client: OllamaClient;

  beforeEach(() => {
    client = new OllamaClient();
  });

  it('should check Ollama health', async () => {
    const healthy = await client.checkHealth();
    expect(typeof healthy).toBe('boolean');
  });

  it('should list available models', async () => {
    const models = await client.listModels();
    expect(Array.isArray(models)).toBe(true);
  });

  it('should generate text response', async () => {
    const response = await client.generate({
      model: 'qwen3-vl-4b',
      prompt: 'Say hello',
    });
    expect(typeof response).toBe('string');
  });

  it('should handle JSON mode', async () => {
    const response = await client.generate({
      model: 'qwen3-vl-4b',
      prompt: 'Generate JSON with field "name"',
      format: { type: 'object', properties: { name: { type: 'string' } } },
    });
    const parsed = JSON.parse(response);
    expect(parsed).toHaveProperty('name');
  });

  it('should retry on connection failure', async () => {
    // Mock network failure
  });

  it('should timeout long requests', async () => {
    const shortTimeoutClient = new OllamaClient({ timeout: 100 });
    await expect(shortTimeoutClient.generate({ model: 'test', prompt: 'test' })).rejects.toThrow(
      'timeout'
    );
  });
});
```
