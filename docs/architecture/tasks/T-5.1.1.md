### MCP Client Implementation (Using Official SDK)

```typescript
// src/main/services/mcp/client.ts
import { Client } from '@modelcontextprotocol/sdk/client/index.js';
import { StdioClientTransport } from '@modelcontextprotocol/sdk/client/stdio.js';
import type { Tool, Resource, CallToolResult } from '@modelcontextprotocol/sdk/types.js';

interface MCPClientConfig {
  serverCommand: string;
  serverArgs: string[];
  name: string;
  timeout?: number;
}

interface MCPServerInfo {
  name: string;
  version: string;
  protocolVersion: string;
  capabilities: {
    tools?: boolean;
    resources?: boolean;
    prompts?: boolean;
  };
}

export class LumosMCPClient {
  private client: Client;
  private transport: StdioClientTransport | null = null;
  private config: MCPClientConfig;
  private serverInfo: MCPServerInfo | null = null;
  private _connected = false;

  constructor(config: MCPClientConfig) {
    this.config = config;
    this.client = new Client({
      name: 'lumos',
      version: '1.0.0'
    });
  }

  async connect(): Promise<void> {
    // Create stdio transport
    this.transport = new StdioClientTransport({
      command: this.config.serverCommand,
      args: this.config.serverArgs
    });

    // Connect to server
    await this.client.connect(this.transport);

    // Get server info (from client's connection result)
    this.serverInfo = {
      name: this.config.name,
      version: '1.0.0',
      protocolVersion: '1.0',
      capabilities: {
        tools: true,
        resources: true,
        prompts: true
      }
    };

    this._connected = true;
  }

  async disconnect(): Promise<void> {
    if (this.client) {
      await this.client.close();
    }
    this._connected = false;
    this.serverInfo = null;
  }

  async listTools(): Promise<Tool[]> {
    this.ensureConnected();

    const toolsList = await this.client.listTools();
    return toolsList.tools;
  }

  async callTool(name: string, args: Record<string, unknown>): Promise<CallToolResult> {
    this.ensureConnected();

    const result = await this.client.callTool({
      name,
      arguments: args
    });

    return result;
  }

  async listResources(): Promise<Resource[]> {
    this.ensureConnected();

    const resourcesList = await this.client.listResources();
    return resourcesList.resources;
  }

  async readResource(uri: string): Promise<{ contents: unknown[] }> {
    this.ensureConnected();

    const result = await this.client.readResource({ uri });
    return result;
  }

  isConnected(): boolean {
    return this._connected;
  }

  getServerInfo(): MCPServerInfo | null {
    return this.serverInfo;
  }

  private ensureConnected(): void {
    if (!this._connected) {
      throw new MCPConnectionError('Client not connected');
    }
  }
}

// Error types
export class MCPError extends Error {
  constructor(message: string) {
    super(message);
    this.name = 'MCPError';
  }
}

export class MCPConnectionError extends MCPError {
  constructor(message: string) {
    super(message);
    this.name = 'MCPConnectionError';
  }
}

export class MCPToolError extends MCPError {
  constructor(message: string, public toolName: string) {
    super(message);
    this.name = 'MCPToolError';
  }
}
```

### Connection Manager

```typescript
// src/main/services/mcp/manager.ts
import { LumosMCPClient, MCPConnectionError } from './client';

interface MCPServerConfig {
  id: string;
  name: string;
  command: string;
  args: string[];
  autoConnect?: boolean;
}

export class MCPConnectionManager {
  private clients: Map<string, LumosMCPClient> = new Map();
  private configs: Map<string, MCPServerConfig> = new Map();

  registerServer(config: MCPServerConfig): void {
    this.configs.set(config.id, config);

    if (config.autoConnect) {
      this.connect(config.id).catch(err => {
        console.error(`Failed to auto-connect to ${config.name}:`, err);
      });
    }
  }

  async connect(serverId: string): Promise<LumosMCPClient> {
    const config = this.configs.get(serverId);
    if (!config) {
      throw new MCPConnectionError(`Server not registered: ${serverId}`);
    }

    // Check if already connected
    const existing = this.clients.get(serverId);
    if (existing?.isConnected()) {
      return existing;
    }

    // Create and connect new client
    const client = new LumosMCPClient({
      serverCommand: config.command,
      serverArgs: config.args,
      name: config.name
    });

    await client.connect();
    this.clients.set(serverId, client);

    return client;
  }

  async disconnect(serverId: string): Promise<void> {
    const client = this.clients.get(serverId);
    if (client) {
      await client.disconnect();
      this.clients.delete(serverId);
    }
  }

  async disconnectAll(): Promise<void> {
    const disconnectPromises = Array.from(this.clients.values())
      .map(client => client.disconnect());
    await Promise.allSettled(disconnectPromises);
    this.clients.clear();
  }

  getClient(serverId: string): LumosMCPClient | undefined {
    return this.clients.get(serverId);
  }

  getConnectedServers(): string[] {
    return Array.from(this.clients.entries())
      .filter(([_, client]) => client.isConnected())
      .map(([id]) => id);
  }
}
```

### Usage Example

```typescript
// Create connection manager
const mcpManager = new MCPConnectionManager();

// Register MCP servers
mcpManager.registerServer({
  id: 'calendar',
  name: 'Calendar MCP Server',
  command: 'npx',
  args: ['-y', '@anthropic/mcp-server-calendar'],
  autoConnect: true
});

mcpManager.registerServer({
  id: 'filesystem',
  name: 'Filesystem MCP Server',
  command: 'npx',
  args: ['-y', '@anthropic/mcp-server-filesystem', '--allowed-directories', '/home/user'],
  autoConnect: false
});

// Connect and use
const calendarClient = await mcpManager.connect('calendar');

// List available tools
const tools = await calendarClient.listTools();
console.log('Available tools:', tools.map(t => t.name));

// Call a tool
const result = await calendarClient.callTool('get_events', {
  start: '2026-01-28T00:00:00Z',
  end: '2026-01-29T00:00:00Z'
});

console.log('Events:', result.content);

// Disconnect when done
await mcpManager.disconnectAll();
```
