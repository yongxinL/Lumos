### Preload Script

```typescript
// src/preload/index.ts
import { contextBridge, ipcRenderer, IpcRendererEvent } from 'electron';
import type { IPCChannel, IPCEventType } from '../types/ipc';

// Type-safe IPC API
const api = {
  // Request-response pattern
  invoke: async <T = unknown>(channel: IPCChannel, payload: unknown): Promise<T> => {
    const response = await ipcRenderer.invoke(channel, payload);
    if (!response.success) {
      throw new Error(response.error?.message || 'IPC request failed');
    }
    return response.data;
  },

  // Event subscription
  on: (event: IPCEventType, callback: (data: unknown) => void): (() => void) => {
    const wrappedCallback = (_: IpcRendererEvent, data: unknown) => {
      callback(data);
    };
    ipcRenderer.on(event, wrappedCallback);

    // Return unsubscribe function
    return () => {
      ipcRenderer.removeListener(event, wrappedCallback);
    };
  },

  // One-time event listener
  once: (event: IPCEventType, callback: (data: unknown) => void): void => {
    ipcRenderer.once(event, (_: IpcRendererEvent, data: unknown) => {
      callback(data);
    });
  },
};

// Expose to renderer
contextBridge.exposeInMainWorld('api', api);

// Type declaration for window.api
declare global {
  interface Window {
    api: typeof api;
  }
}
```

### Main Process Handlers

```typescript
// src/main/ipc/handlers.ts
import { ipcMain, IpcMainInvokeEvent } from 'electron';
import type { IPCChannel, IPCResponse } from '../../types/ipc';

// Handler wrapper with error handling and validation
export function registerHandler<TPayload = unknown, TResult = unknown>(
  channel: IPCChannel,
  handler: (payload: TPayload) => Promise<TResult>
): void {
  ipcMain.handle(channel, async (event: IpcMainInvokeEvent, payload: TPayload) => {
    const response: IPCResponse<TResult> = {
      id: crypto.randomUUID(),
      success: false,
      timestamp: new Date().toISOString(),
    };

    try {
      // Execute handler
      const data = await handler(payload);
      response.success = true;
      response.data = data;
    } catch (error) {
      response.error = {
        code: error instanceof Error ? error.name : 'UNKNOWN_ERROR',
        message: error instanceof Error ? error.message : 'Unknown error',
      };
    }

    return response;
  });
}

// Example: Input handlers
import { InputHandlerService } from '../services/input-handler';

export function registerInputHandlers(service: InputHandlerService): void {
  registerHandler('input:submit-text', async (text: string) => {
    return service.submitText(text);
  });

  registerHandler('input:start-recording', async () => {
    return service.startRecording();
  });

  registerHandler('input:stop-recording', async () => {
    return service.stopRecording();
  });

  registerHandler('input:confirm-transcription', async (preview: TranscriptionPreview) => {
    return service.confirmTranscription(preview);
  });
}
```

### Event Emission

```typescript
// src/main/ipc/events.ts
import { BrowserWindow } from 'electron';
import type { IPCEvent, IPCEventType } from '../../types/ipc';

export class IPCEventEmitter {
  constructor(private window: BrowserWindow) {}

  emit<T>(type: IPCEventType, payload: T): void {
    const event: IPCEvent<T> = {
      type,
      payload,
      timestamp: new Date().toISOString(),
    };

    this.window.webContents.send(type, event);
  }
}

// Usage
const emitter = new IPCEventEmitter(mainWindow);

// Emit transcription updates
emitter.emit('transcription:update', {
  partial: 'Hello world...',
});

// Emit proposal events
emitter.emit('proposal:new', {
  proposal: actionProposal,
});
```

### React IPC Hook

```typescript
// src/renderer/hooks/useIPC.ts
import { useEffect, useCallback } from 'react';
import type { IPCChannel, IPCEventType } from '../../types/ipc';

export function useIPC() {
  const invoke = useCallback(
    async <T = unknown>(channel: IPCChannel, payload?: unknown): Promise<T> => {
      return window.api.invoke<T>(channel, payload);
    },
    []
  );

  return { invoke };
}

export function useIPCEvent<T = unknown>(
  event: IPCEventType,
  callback: (data: T) => void
): void {
  useEffect(() => {
    const unsubscribe = window.api.on(event, (data) => {
      callback(data as T);
    });

    return unsubscribe;
  }, [event, callback]);
}

// Usage in component
function ChatView() {
  const { invoke } = useIPC();

  // Handle proposal events
  useIPCEvent<ActionProposal>('proposal:new', (proposal) => {
    console.log('New proposal:', proposal);
  });

  // Submit text input
  const handleSubmit = async (text: string) => {
    const result = await invoke('input:submit-text', text);
    console.log('Input processed:', result);
  };

  return <div>...</div>;
}
```

### Message Validation

```typescript
// src/main/ipc/validation.ts
import Ajv from 'ajv';

const ajv = new Ajv();

// Define schemas for each channel
const schemas: Record<IPCChannel, object> = {
  'input:submit-text': {
    type: 'string',
    minLength: 1,
    maxLength: 10000,
  },
  'skills:activate': {
    type: 'object',
    required: ['skillId'],
    properties: {
      skillId: { type: 'string', format: 'uuid' },
    },
  },
  // ... more schemas
};

export function validatePayload(channel: IPCChannel, payload: unknown): boolean {
  const schema = schemas[channel];
  if (!schema) return true; // No schema = no validation

  const validate = ajv.compile(schema);
  return validate(payload);
}
```

## Security Considerations

1. **Context Isolation**: Preload script isolated from renderer
2. **No Direct Node Access**: Renderer cannot access Node.js APIs
3. **Validated Messages**: All IPC payloads validated before processing
4. **Error Boundaries**: IPC errors don't crash application
5. **Rate Limiting**: Prevent IPC flooding attacks (future enhancement)
