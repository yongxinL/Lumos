## Technical Notes

```yaml
# resources/skills/calendar-operations.yaml
id: "skill-calendar-001"
name: "Calendar Operations"
description: "Create, update, and delete calendar events"
status: active
origin: human
enabled_operations:
  - create_calendar_event
  - update_calendar_event
  - delete_calendar_event
policy_constraints:
  - type: precondition
    rule: "proposal.confidence >= 0.7"
    error_message: "Confidence too low for calendar operations"
  - type: precondition
    rule: "proposal.data_classification !== 'restricted'"
    error_message: "Cannot operate on restricted data"
required_trust_level: SUPERVISED
priority: 80
data_domain: both
conflict_resolution: policy_escalation
proposal_schema:
  type: object
  required:
    - intent
    - operation
  properties:
    intent:
      type: string
    operation:
      enum:
        - create_calendar_event
        - update_calendar_event
        - delete_calendar_event
rollback_specification:
  type: FULL
  method: "Reverse calendar operation via MCP"
created_at: "2026-01-28T00:00:00Z"

# resources/skills/filesystem-operations.yaml
id: "skill-filesystem-001"
name: "Filesystem Operations"
description: "Read, write, and manage files"
status: active
origin: human
enabled_operations:
  - read_file
  - write_file
  - delete_file
  - create_note
  - update_note
  - delete_note
policy_constraints:
  - type: precondition
    rule: "proposal.confidence >= 0.8"
    error_message: "High confidence required for file operations"
  - type: invariant
    rule: "!proposal.operation.includes('delete') || proposal.requires_confirmation"
    error_message: "Delete operations must require confirmation"
required_trust_level: SUPERVISED
priority: 75
data_domain: personal
conflict_resolution: user_choice
proposal_schema:
  type: object
  required:
    - intent
    - operation
rollback_specification:
  type: PARTIAL
  method: "Restore file from backup if available"
created_at: "2026-01-28T00:00:00Z"
```

```typescript
// Package skills during build
// electron-builder.config.js
module.exports = {
  extraResources: [
    {
      from: 'resources/skills',
      to: 'skills'
    }
  ]
};
```
