### Audio Handler

```swift
// swift/Sources/LumosHelper/AudioHandler.swift
import Foundation
import FluidAudio
import AVFoundation

class AudioHandler: FluidAudioDelegate {
    private var engine: FluidAudioEngine?
    private let messageHandler: MessageHandler
    private var isRecording = false
    private var recordingStartTime: Date?

    init(messageHandler: MessageHandler) {
        self.messageHandler = messageHandler
    }

    func startRecording() throws {
        // Check microphone permission
        guard checkMicrophonePermission() else {
            throw AudioError.permissionDenied
        }

        // Initialize engine
        engine = try FluidAudioEngine(
            sampleRate: 16000,
            channels: 1,
            delegate: self
        )

        // Start recording
        try engine?.startRecording()
        isRecording = true
        recordingStartTime = Date()

        // Notify success
        messageHandler.sendEvent("recording:started", params: [:])
    }

    func stopRecording() throws -> TranscriptionPreview {
        guard isRecording else {
            throw AudioError.notRecording
        }

        // Stop recording
        let finalTranscript = try engine?.stopRecording()
        isRecording = false

        let duration = Date().timeIntervalSince(recordingStartTime ?? Date())

        // Create preview
        let preview = TranscriptionPreview(
            text: finalTranscript?.text ?? "",
            confidence: finalTranscript?.confidence ?? 0.0,
            durationMs: Int(duration * 1000),
            language: finalTranscript?.language ?? "en"
        )

        // Notify completion
        messageHandler.sendEvent("recording:stopped", params: [
            "preview": preview.toDictionary()
        ])

        return preview
    }

    func cancelRecording() {
        guard isRecording else { return }

        engine?.cancelRecording()
        isRecording = false

        messageHandler.sendEvent("recording:cancelled", params: [:])
    }

    // MARK: - FluidAudioDelegate

    func fluidAudio(_ engine: FluidAudioEngine, didReceivePartialTranscript transcript: String) {
        // Stream partial transcription
        messageHandler.sendEvent("transcription:update", params: [
            "partial": transcript
        ])
    }

    func fluidAudio(_ engine: FluidAudioEngine, didEncounterError error: Error) {
        messageHandler.sendEvent("recording:error", params: [
            "error": error.localizedDescription
        ])
    }

    // MARK: - Permissions

    private func checkMicrophonePermission() -> Bool {
        let status = AVCaptureDevice.authorizationStatus(for: .audio)

        switch status {
        case .authorized:
            return true
        case .notDetermined:
            // Request permission synchronously
            var granted = false
            let semaphore = DispatchSemaphore(value: 0)

            AVCaptureDevice.requestAccess(for: .audio) { result in
                granted = result
                semaphore.signal()
            }

            semaphore.wait()
            return granted
        case .denied, .restricted:
            return false
        @unknown default:
            return false
        }
    }
}

// MARK: - Models

struct TranscriptionPreview: Codable {
    let text: String
    let confidence: Double
    let durationMs: Int
    let language: String

    func toDictionary() -> [String: Any] {
        return [
            "text": text,
            "confidence": confidence,
            "duration_ms": durationMs,
            "language": language
        ]
    }
}

enum AudioError: Error {
    case permissionDenied
    case notRecording
    case engineInitFailed
}
```

### Update Message Handler

```swift
// swift/Sources/LumosHelper/MessageHandler.swift
class MessageHandler {
    private let audioHandler: AudioHandler

    init() {
        self.audioHandler = AudioHandler(messageHandler: self)
    }

    private func handleRequest(_ message: IPCMessage) {
        guard let method = message.method else {
            sendError(id: message.id, code: "INVALID_REQUEST", message: "Missing method")
            return
        }

        do {
            switch method {
            case "ping":
                sendResponse(id: message.id, result: ["status": "ok"])

            case "health":
                sendResponse(id: message.id, result: [
                    "status": "healthy",
                    "uptime": ProcessInfo.processInfo.systemUptime
                ])

            case "audio:start-recording":
                try audioHandler.startRecording()
                sendResponse(id: message.id, result: ["status": "started"])

            case "audio:stop-recording":
                let preview = try audioHandler.stopRecording()
                sendResponse(id: message.id, result: preview.toDictionary())

            case "audio:cancel-recording":
                audioHandler.cancelRecording()
                sendResponse(id: message.id, result: ["status": "cancelled"])

            default:
                sendError(
                    id: message.id,
                    code: "METHOD_NOT_FOUND",
                    message: "Unknown method: \(method)"
                )
            }
        } catch let error as AudioError {
            sendError(
                id: message.id,
                code: "AUDIO_ERROR",
                message: error.localizedDescription
            )
        } catch {
            sendError(
                id: message.id,
                code: "INTERNAL_ERROR",
                message: error.localizedDescription
            )
        }
    }
}
```

### TypeScript Bridge Updates

```typescript
// src/main/bridges/swift-bridge.ts
export class SwiftBridge extends EventEmitter {
  async startRecording(): Promise<void> {
    await this.sendRequest('audio:start-recording');
  }

  async stopRecording(): Promise<TranscriptionPreview> {
    const result = await this.sendRequest('audio:stop-recording');
    return result as TranscriptionPreview;
  }

  cancelRecording(): void {
    this.sendRequest('audio:cancel-recording').catch(console.error);
  }

  // Subscribe to streaming updates
  onTranscriptionUpdate(callback: (partial: string) => void): void {
    this.on('transcription:update', (params) => {
      callback(params.partial);
    });
  }
}
```
