## Technical Notes

### Constraint Evaluator Implementation

```typescript
// src/main/services/governance/constraint-evaluator.ts
import * as vm from 'vm';
import * as crypto from 'crypto';

interface EvaluationContext {
  proposal: ActionProposal;
  trustLevel: TrustLevel;
  user: {
    id: string;
    permissions: string[];
  };
  system: {
    timestamp: string;
    environment: 'development' | 'production';
  };
}

interface EvaluationResult {
  passed: boolean;
  constraint: PolicyConstraint;
  executionTime: number;
  error?: string;
}

export class ConstraintEvaluator {
  private cache: Map<string, boolean> = new Map();
  private maxCacheSize = 1000;
  private timeout = 5000; // 5 seconds max per constraint

  evaluateConstraint(constraint: PolicyConstraint, context: EvaluationContext): EvaluationResult {
    const startTime = performance.now();

    try {
      // Check cache
      const cacheKey = this.getCacheKey(constraint, context);
      if (this.cache.has(cacheKey)) {
        const cached = this.cache.get(cacheKey)!;
        return {
          passed: cached,
          constraint,
          executionTime: performance.now() - startTime,
        };
      }

      // Evaluate constraint
      const passed = this.evaluateExpression(constraint.rule, context);

      // Cache result
      this.cacheResult(cacheKey, passed);

      const executionTime = performance.now() - startTime;

      // Performance monitoring
      if (executionTime > 10) {
        console.warn(`Slow constraint evaluation (${executionTime}ms): ${constraint.rule}`);
      }

      return {
        passed,
        constraint,
        executionTime,
      };
    } catch (error) {
      const executionTime = performance.now() - startTime;
      return {
        passed: false,
        constraint,
        executionTime,
        error: error instanceof Error ? error.message : 'Unknown error',
      };
    }
  }

  evaluateConstraints(
    constraints: PolicyConstraint[],
    context: EvaluationContext
  ): EvaluationResult[] {
    return constraints.map((c) => this.evaluateConstraint(c, context));
  }

  evaluateByType(
    constraints: PolicyConstraint[],
    type: 'precondition' | 'invariant' | 'postcondition',
    context: EvaluationContext
  ): EvaluationResult[] {
    const filtered = constraints.filter((c) => c.type === type);
    return this.evaluateConstraints(filtered, context);
  }

  clearCache(): void {
    this.cache.clear();
  }

  private evaluateExpression(expression: string, context: EvaluationContext): boolean {
    // Create sandboxed context
    const sandbox = this.createSandbox(context);

    // Compile and run in VM
    const script = new vm.Script(`(${expression})`, {
      timeout: this.timeout,
      displayErrors: true,
    });

    const vmContext = vm.createContext(sandbox);
    const result = script.runInContext(vmContext, {
      timeout: this.timeout,
    });

    // Ensure boolean result
    return Boolean(result);
  }

  private createSandbox(context: EvaluationContext): any {
    // Whitelist of safe functions
    const safeFunctions = {
      // String operations
      includes: (str: string, search: string) => str.includes(search),
      startsWith: (str: string, search: string) => str.startsWith(search),
      endsWith: (str: string, search: string) => str.endsWith(search),
      toLowerCase: (str: string) => str.toLowerCase(),
      toUpperCase: (str: string) => str.toUpperCase(),

      // Array operations
      contains: (arr: any[], item: any) => arr.includes(item),
      isEmpty: (arr: any[]) => arr.length === 0,
      length: (arr: any[]) => arr.length,

      // Number operations
      abs: Math.abs,
      min: Math.min,
      max: Math.max,

      // Date operations (limited)
      now: () => new Date().toISOString(),

      // Comparison helpers
      equals: (a: any, b: any) => a === b,
      notEquals: (a: any, b: any) => a !== b,
      greaterThan: (a: number, b: number) => a > b,
      lessThan: (a: number, b: number) => a < b,
    };

    // Create sandbox with limited scope
    return {
      // Context data (read-only)
      proposal: Object.freeze(JSON.parse(JSON.stringify(context.proposal))),
      trustLevel: context.trustLevel,
      user: Object.freeze(JSON.parse(JSON.stringify(context.user))),
      system: Object.freeze(JSON.parse(JSON.stringify(context.system))),

      // Safe functions
      ...safeFunctions,

      // Blocked globals
      require: undefined,
      module: undefined,
      exports: undefined,
      process: undefined,
      global: undefined,
      __dirname: undefined,
      __filename: undefined,
      setTimeout: undefined,
      setInterval: undefined,
      setImmediate: undefined,
      clearTimeout: undefined,
      clearInterval: undefined,
      clearImmediate: undefined,
    };
  }

  private getCacheKey(constraint: PolicyConstraint, context: EvaluationContext): string {
    const data = {
      rule: constraint.rule,
      proposalId: context.proposal.id,
      operation: context.proposal.operation,
      trustLevel: context.trustLevel,
    };

    return crypto.createHash('md5').update(JSON.stringify(data)).digest('hex');
  }

  private cacheResult(key: string, result: boolean): void {
    // LRU cache eviction
    if (this.cache.size >= this.maxCacheSize) {
      const firstKey = this.cache.keys().next().value;
      this.cache.delete(firstKey);
    }

    this.cache.set(key, result);
  }

  // Validate constraint syntax (called when loading skills)
  validateConstraintSyntax(constraint: PolicyConstraint): SyntaxValidationResult {
    try {
      // Try to compile the expression
      new vm.Script(`(${constraint.rule})`);

      return {
        valid: true,
      };
    } catch (error) {
      return {
        valid: false,
        error: error instanceof Error ? error.message : 'Syntax error',
      };
    }
  }
}

interface SyntaxValidationResult {
  valid: boolean;
  error?: string;
}

// Example constraint expressions
const EXAMPLE_CONSTRAINTS = {
  // Preconditions
  confidenceTooLow: {
    type: 'precondition' as const,
    rule: 'proposal.confidence >= 0.7',
    error_message: 'Confidence must be at least 0.7',
  },

  restrictedDataDomain: {
    type: 'precondition' as const,
    rule: 'proposal.data_classification !== "restricted"',
    error_message: 'Cannot operate on restricted data',
  },

  requiresHighTrust: {
    type: 'precondition' as const,
    rule: 'trustLevel === "DELEGATED"',
    error_message: 'Operation requires delegated trust level',
  },

  // Invariants
  targetMustBeServiceNow: {
    type: 'invariant' as const,
    rule: 'proposal.target_entity === null || proposal.target_entity.source === "servicenow"',
    error_message: 'Target must be ServiceNow or null',
  },

  operationMustMatch: {
    type: 'invariant' as const,
    rule: 'proposal.operation === "create_ticket"',
    error_message: 'Operation must be create_ticket',
  },

  // Postconditions
  mustHaveRollbackPlan: {
    type: 'postcondition' as const,
    rule: 'proposal.rollback_plan !== null',
    error_message: 'Rollback plan is required',
  },
};
```

### Usage Example

```typescript
// Example: Evaluate constraints for a proposal
const evaluator = new ConstraintEvaluator();

const context: EvaluationContext = {
  proposal: actionProposal,
  trustLevel: 'SUPERVISED',
  user: {
    id: 'user-123',
    permissions: ['create_ticket', 'update_ticket'],
  },
  system: {
    timestamp: new Date().toISOString(),
    environment: 'production',
  },
};

// Evaluate all constraints
const results = evaluator.evaluateConstraints(skill.policy_constraints, context);

// Check if all passed
const allPassed = results.every((r) => r.passed);

// Get failures
const failures = results.filter((r) => !r.passed);
for (const failure of failures) {
  console.error(`Constraint failed: ${failure.constraint.error_message}`);
  if (failure.error) {
    console.error(`Error: ${failure.error}`);
  }
}

// Evaluate by type
const preconditions = evaluator.evaluateByType(skill.policy_constraints, 'precondition', context);
```

## Test Cases

```typescript
describe('ConstraintEvaluator', () => {
  let evaluator: ConstraintEvaluator;
  let context: EvaluationContext;

  beforeEach(() => {
    evaluator = new ConstraintEvaluator();
    context = {
      proposal: {
        id: 'test',
        operation: 'create_ticket',
        confidence: 0.9,
        data_classification: 'internal',
        // ... other fields
      } as ActionProposal,
      trustLevel: 'SUPERVISED',
      user: { id: 'test-user', permissions: [] },
      system: { timestamp: new Date().toISOString(), environment: 'development' },
    };
  });

  it('should evaluate simple constraint', () => {
    const constraint: PolicyConstraint = {
      type: 'precondition',
      rule: 'proposal.confidence >= 0.7',
      error_message: 'Confidence too low',
    };

    const result = evaluator.evaluateConstraint(constraint, context);
    expect(result.passed).toBe(true);
  });

  it('should fail on false constraint', () => {
    const constraint: PolicyConstraint = {
      type: 'precondition',
      rule: 'proposal.confidence >= 0.95',
      error_message: 'Confidence too low',
    };

    const result = evaluator.evaluateConstraint(constraint, context);
    expect(result.passed).toBe(false);
  });

  it('should cache results', () => {
    const constraint: PolicyConstraint = {
      type: 'precondition',
      rule: 'proposal.confidence >= 0.7',
      error_message: 'Test',
    };

    const result1 = evaluator.evaluateConstraint(constraint, context);
    const result2 = evaluator.evaluateConstraint(constraint, context);

    expect(result2.executionTime).toBeLessThan(result1.executionTime);
  });

  it('should timeout on infinite loop', () => {
    const constraint: PolicyConstraint = {
      type: 'precondition',
      rule: 'while(true) {}',
      error_message: 'Test',
    };

    const result = evaluator.evaluateConstraint(constraint, context);
    expect(result.passed).toBe(false);
    expect(result.error).toBeDefined();
  });

  it('should sandbox dangerous operations', () => {
    const constraint: PolicyConstraint = {
      type: 'precondition',
      rule: 'require("fs").readFileSync("/etc/passwd")',
      error_message: 'Test',
    };

    const result = evaluator.evaluateConstraint(constraint, context);
    expect(result.passed).toBe(false);
    expect(result.error).toBeDefined();
  });

  it('should validate syntax', () => {
    const valid = evaluator.validateConstraintSyntax({
      type: 'precondition',
      rule: 'proposal.confidence >= 0.7',
      error_message: 'Test',
    });
    expect(valid.valid).toBe(true);

    const invalid = evaluator.validateConstraintSyntax({
      type: 'precondition',
      rule: 'this is not valid syntax',
      error_message: 'Test',
    });
    expect(invalid.valid).toBe(false);
  });
});
```

## Security Considerations

1. **VM Sandbox**: All constraints run in isolated VM context
2. **No Node.js Access**: `require`, `process`, `fs` blocked
3. **Timeout Protection**: 5-second timeout prevents infinite loops
4. **Read-Only Context**: Context objects frozen, cannot be modified
5. **Whitelist Functions**: Only explicitly allowed functions available
6. **No Side Effects**: Constraints cannot modify system state
