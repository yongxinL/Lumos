### Execution Layer Implementation

```typescript
// src/main/services/execution/execution-layer.ts
import { MCPServerRegistry } from '../mcp/mcp-server-registry';
import { AuditService } from '../audit/audit.service';

interface ExecutionContext {
  proposal: ActionProposal;
  plan: ActionPlan;
  userId: string;
}

interface ExecutionResult {
  success: boolean;
  actionId: string;
  outputs: any[];
  duration: number;
  stateSnapshot: StateSnapshot | null;
  error?: string;
}

interface StateSnapshot {
  operation: OperationType;
  timestamp: ISO8601String;
  preState: any;
  captureMethod: string;
}

export class ExecutionLayer {
  private mcpRegistry: MCPServerRegistry;
  private auditService: AuditService;
  private activeExecutions: Map<string, ExecutionContext> = new Map();

  constructor(mcpRegistry: MCPServerRegistry, auditService: AuditService) {
    this.mcpRegistry = mcpRegistry;
    this.auditService = auditService;
  }

  async execute(context: ExecutionContext): Promise<ExecutionResult> {
    const startTime = Date.now();
    const actionId = crypto.randomUUID();

    // Track active execution
    this.activeExecutions.set(actionId, context);

    try {
      // Validate tools available
      await this.validateToolsAvailable(context);

      // Capture pre-execution state
      const stateSnapshot = await this.captureState(context);

      // Log execution start
      await this.auditService.log({
        action_id: actionId,
        proposal_id: context.proposal.id,
        operation: context.proposal.operation,
        status: 'executing',
        actor: 'system',
        data: { plan: context.plan }
      });

      // Execute action steps
      const outputs = await this.executeSteps(context, actionId);

      // Validate results
      await this.validateResults(context, outputs);

      const duration = Date.now() - startTime;

      // Log success
      await this.auditService.log({
        action_id: actionId,
        proposal_id: context.proposal.id,
        operation: context.proposal.operation,
        status: 'success',
        actor: 'system',
        data: { outputs, duration }
      });

      return {
        success: true,
        actionId,
        outputs,
        duration,
        stateSnapshot
      };

    } catch (error) {
      const duration = Date.now() - startTime;

      // Log failure
      await this.auditService.log({
        action_id: actionId,
        proposal_id: context.proposal.id,
        operation: context.proposal.operation,
        status: 'failure',
        actor: 'system',
        data: {
          error: error instanceof Error ? error.message : 'Unknown error',
          duration
        }
      });

      // Attempt automatic rollback
      try {
        await this.rollback(actionId);
      } catch (rollbackError) {
        console.error('Rollback failed:', rollbackError);
      }

      return {
        success: false,
        actionId,
        outputs: [],
        duration,
        stateSnapshot: null,
        error: error instanceof Error ? error.message : 'Execution failed'
      };

    } finally {
      this.activeExecutions.delete(actionId);
    }
  }

  async rollback(actionId: string): Promise<RollbackResult> {
    // Get execution context
    const context = this.activeExecutions.get(actionId);
    if (!context) {
      throw new Error(`No active execution found for ${actionId}`);
    }

    // Rollback logic handled by RollbackManager (T-5.2.1)
    throw new Error('Rollback not yet implemented');
  }

  getRegisteredTools(): ToolInfo[] {
    return this.mcpRegistry.getAllTools();
  }

  isExecuting(actionId: string): boolean {
    return this.activeExecutions.has(actionId);
  }

  getActiveExecutions(): string[] {
    return Array.from(this.activeExecutions.keys());
  }

  // Private methods

  private async validateToolsAvailable(context: ExecutionContext): Promise<void> {
    // Map operation to required tool
    const toolName = this.operationToTool(context.proposal.operation);

    // Check if tool exists
    const location = await this.mcpRegistry.findTool(toolName);
    if (!location) {
      throw new ExecutionError(
        `Required tool ${toolName} not available`,
        'TOOL_NOT_FOUND'
      );
    }

    // Validate tool capabilities match requirements
    // (Future enhancement)
  }

  private async captureState(context: ExecutionContext): Promise<StateSnapshot | null> {
    // Only capture state for reversible operations
    if (context.proposal.reversibility === 'IRREVERSIBLE') {
      return null;
    }

    const toolName = this.operationToTool(context.proposal.operation);

    try {
      // Capture current state before execution
      let preState: any = null;

      if (context.proposal.operation.includes('calendar')) {
        // Capture calendar state
        preState = await this.captureCalendarState(context);
      } else if (context.proposal.operation.includes('file')) {
        // Capture file state
        preState = await this.captureFileState(context);
      }

      return {
        operation: context.proposal.operation,
        timestamp: new Date().toISOString(),
        preState,
        captureMethod: toolName
      };

    } catch (error) {
      console.warn('Failed to capture state:', error);
      return null;
    }
  }

  private async executeSteps(
    context: ExecutionContext,
    actionId: string
  ): Promise<any[]> {
    const outputs: any[] = [];

    for (const step of context.plan.steps) {
      try {
        // Map operation to tool and arguments
        const { toolName, args } = this.planStepToTool(
          step,
          context.proposal
        );

        // Find tool
        const location = await this.mcpRegistry.findTool(toolName);
        if (!location) {
          throw new ExecutionError(
            `Tool ${toolName} not found`,
            'TOOL_NOT_FOUND'
          );
        }

        // Execute tool
        const result = await this.mcpRegistry.callTool(
          location.serverId,
          toolName,
          args
        );

        outputs.push({
          step: step.order,
          tool: toolName,
          result
        });

      } catch (error) {
        throw new ExecutionError(
          `Step ${step.order} failed: ${error instanceof Error ? error.message : 'Unknown'}`,
          'STEP_FAILED'
        );
      }
    }

    return outputs;
  }

  private async validateResults(
    context: ExecutionContext,
    outputs: any[]
  ): Promise<void> {
    // Validate all expected outputs present
    if (outputs.length !== context.plan.steps.length) {
      throw new ExecutionError(
        'Not all steps completed',
        'INCOMPLETE_EXECUTION'
      );
    }

    // Validate against expected outcomes
    for (const expectedOutcome of context.plan.expectedOutcomes) {

      // Real implementation would be more sophisticated
    }
  }

  private operationToTool(operation: OperationType): string {
    // Map Lumos operations to MCP tool names
    const mapping: Record<string, string> = {
      'create_calendar_event': 'create_event',
      'update_calendar_event': 'update_event',
      'delete_calendar_event': 'delete_event',
      'read_file': 'read_file',
      'write_file': 'write_file',
      'create_note': 'write_file',
      'update_note': 'write_file',
      'delete_note': 'delete_file'
    };

    return mapping[operation] || operation;
  }

  private planStepToTool(
    step: ActionStep,
    proposal: ActionProposal
  ): { toolName: string; args: object } {
    const toolName = this.operationToTool(proposal.operation);

    // Extract arguments from step and proposal
    const args: any = {};

    if (proposal.operation === 'create_calendar_event') {
      args.title = step.description;
      args.start = new Date().toISOString();
      args.end = new Date(Date.now() + 3600000).toISOString();
    }

    return { toolName, args };
  }

  private async captureCalendarState(context: ExecutionContext): Promise<any> {
    // Capture current calendar events that might be affected
    if (context.proposal.target_entity?.id) {
      const location = await this.mcpRegistry.findTool('get_event');
      if (location) {
        return await this.mcpRegistry.callTool(
          location.serverId,
          'get_event',
          { id: context.proposal.target_entity.id }
        );
      }
    }
    return null;
  }

  private async captureFileState(context: ExecutionContext): Promise<any> {
    // Capture current file content
    if (context.proposal.target_entity?.id) {
      const location = await this.mcpRegistry.findTool('read_file');
      if (location) {
        return await this.mcpRegistry.callTool(
          location.serverId,
          'read_file',
          { path: context.proposal.target_entity.id }
        );
      }
    }
    return null;
  }
}

// Errors
export class ExecutionError extends Error {
  constructor(message: string, public code: string) {
    super(message);
    this.name = 'ExecutionError';
  }
}

interface RollbackResult {
  success: boolean;
  error?: string;
}
```
