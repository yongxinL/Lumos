### AI Provider Interface

```typescript
// src/main/services/ai/providers/ai-provider.interface.ts

export interface AIProviderConfig {
  apiKey?: string;
  baseUrl?: string;
  model: string;
  temperature?: number;
  maxTokens?: number;
  timeout?: number;
}

export interface AIProviderCapabilities {
  streaming: boolean;
  functionCalling: boolean;
  vision: boolean;
  jsonMode: boolean;
  maxContextLength: number;
}

export interface GenerateOptions {
  temperature?: number;
  maxTokens?: number;
  topP?: number;
  stopSequences?: string[];
  systemPrompt?: string;
}

export interface GenerateRequest {
  messages: Message[];
  options?: GenerateOptions;
}

export interface Message {
  role: 'system' | 'user' | 'assistant';
  content: string;
}

export interface GenerateResponse {
  content: string;
  finishReason: 'stop' | 'length' | 'error';
  usage: TokenUsage;
  model: string;
}

export interface TokenUsage {
  promptTokens: number;
  completionTokens: number;
  totalTokens: number;
  estimatedCost?: number;
}

export interface StreamChunk {
  delta: string;
  isComplete: boolean;
}

export abstract class AIProvider {
  protected config: AIProviderConfig;

  constructor(config: AIProviderConfig) {
    this.config = config;
  }

  // Core methods
  abstract generate(request: GenerateRequest): Promise<GenerateResponse>;

  abstract generateStream(
    request: GenerateRequest,
    onChunk: (chunk: StreamChunk) => void
  ): Promise<GenerateResponse>;

  abstract getCapabilities(): AIProviderCapabilities;

  abstract checkHealth(): Promise<boolean>;

  abstract listModels(): Promise<ModelInfo[]>;

  // Helper methods
  getName(): string {
    return this.constructor.name;
  }

  getModel(): string {
    return this.config.model;
  }

  getType(): 'local' | 'cloud' {
    return 'cloud'; // Override in subclasses
  }

  // Validation
  protected validateConfig(): void {
    if (!this.config.model) {
      throw new Error('Model is required');
    }
  }

  protected validateRequest(request: GenerateRequest): void {
    if (!request.messages || request.messages.length === 0) {
      throw new Error('At least one message is required');
    }
  }
}

export interface ModelInfo {
  id: string;
  name: string;
  contextLength: number;
  costPer1kTokens?: {
    prompt: number;
    completion: number;
  };
}
```

### Provider Registry

```typescript
// src/main/services/ai/providers/provider-registry.ts

export class AIProviderRegistry {
  private providers: Map<string, AIProvider> = new Map();
  private defaultProvider?: string;

  register(name: string, provider: AIProvider): void {
    this.providers.set(name, provider);

    if (this.providers.size === 1) {
      this.defaultProvider = name;
    }
  }

  unregister(name: string): void {
    this.providers.delete(name);

    if (this.defaultProvider === name) {
      this.defaultProvider = this.providers.keys().next().value;
    }
  }

  get(name: string): AIProvider | undefined {
    return this.providers.get(name);
  }

  getDefault(): AIProvider | undefined {
    if (this.defaultProvider) {
      return this.providers.get(this.defaultProvider);
    }
    return undefined;
  }

  setDefault(name: string): void {
    if (!this.providers.has(name)) {
      throw new Error(`Provider ${name} not registered`);
    }
    this.defaultProvider = name;
  }

  list(): string[] {
    return Array.from(this.providers.keys());
  }

  async listAvailable(): Promise<ProviderStatus[]> {
    const statuses: ProviderStatus[] = [];

    for (const [name, provider] of this.providers) {
      const healthy = await provider.checkHealth();
      const capabilities = provider.getCapabilities();

      statuses.push({
        name,
        model: provider.getModel(),
        type: provider.getType(),
        healthy,
        capabilities
      });
    }

    return statuses;
  }

  async getHealthyProvider(preferLocal = false): Promise<AIProvider | null> {
    // Try default first
    if (this.defaultProvider) {
      const provider = this.providers.get(this.defaultProvider);
      if (provider && await provider.checkHealth()) {
        return provider;
      }
    }

    // Find any healthy provider
    for (const provider of this.providers.values()) {
      if (preferLocal && provider.getType() !== 'local') {
        continue;
      }

      if (await provider.checkHealth()) {
        return provider;
      }
    }

    return null;
  }
}

export interface ProviderStatus {
  name: string;
  model: string;
  type: 'local' | 'cloud';
  healthy: boolean;
  capabilities: AIProviderCapabilities;
}
```

### Base Provider Implementation

```typescript
// src/main/services/ai/providers/base-provider.ts

export abstract class BaseAIProvider extends AIProvider {
  protected async generateWithRetry(
    fn: () => Promise<GenerateResponse>,
    maxRetries = 3
  ): Promise<GenerateResponse> {
    let lastError: Error | undefined;

    for (let attempt = 1; attempt <= maxRetries; attempt++) {
      try {
        return await fn();
      } catch (error) {
        lastError = error as Error;
        console.error(`Provider attempt ${attempt} failed:`, error);

        if (attempt < maxRetries) {
          await this.delay(1000 * attempt);
        }
      }
    }

    throw new AIProviderError(
      `Failed after ${maxRetries} attempts: ${lastError?.message}`
    );
  }

  protected calculateCost(usage: TokenUsage, costPer1k: { prompt: number; completion: number }): number {
    const promptCost = (usage.promptTokens / 1000) * costPer1k.prompt;
    const completionCost = (usage.completionTokens / 1000) * costPer1k.completion;
    return promptCost + completionCost;
  }

  protected async delay(ms: number): Promise<void> {
    return new Promise(resolve => setTimeout(resolve, ms));
  }

  protected mapError(error: any): AIProviderError {
    if (error instanceof AIProviderError) {
      return error;
    }

    // Map common HTTP errors
    if (error.status === 401 || error.status === 403) {
      return new AIProviderAuthError('Authentication failed');
    }

    if (error.status === 429) {
      return new AIProviderRateLimitError('Rate limit exceeded');
    }

    if (error.status === 500 || error.status === 503) {
      return new AIProviderUnavailableError('Provider unavailable');
    }

    return new AIProviderError(error.message || 'Unknown error');
  }
}

// Error classes
export class AIProviderError extends Error {
  constructor(message: string, public readonly code?: string) {
    super(message);
    this.name = 'AIProviderError';
  }
}

export class AIProviderAuthError extends AIProviderError {
  constructor(message: string) {
    super(message, 'AUTH_ERROR');
    this.name = 'AIProviderAuthError';
  }
}

export class AIProviderRateLimitError extends AIProviderError {
  constructor(message: string) {
    super(message, 'RATE_LIMIT');
    this.name = 'AIProviderRateLimitError';
  }
}

export class AIProviderUnavailableError extends AIProviderError {
  constructor(message: string) {
    super(message, 'UNAVAILABLE');
    this.name = 'AIProviderUnavailableError';
  }
}
```

### Ollama Provider Implementation

```typescript
// src/main/services/ai/providers/ollama-provider.ts

export class OllamaProvider extends BaseAIProvider {
  private client: OllamaClient;

  constructor(config: AIProviderConfig) {
    super(config);
    this.client = new OllamaClient({
      baseUrl: config.baseUrl || 'http://localhost:11434'
    });
  }

  async generate(request: GenerateRequest): Promise<GenerateResponse> {
    this.validateRequest(request);

    const prompt = this.formatMessages(request.messages);
    const response = await this.client.generate({
      model: this.config.model,
      prompt,
      options: {
        temperature: request.options?.temperature ?? this.config.temperature,
        top_p: request.options?.topP
      }
    });

    return {
      content: response,
      finishReason: 'stop',
      usage: {
        promptTokens: 0, // Ollama doesn't provide token counts
        completionTokens: 0,
        totalTokens: 0
      },
      model: this.config.model
    };
  }

  async generateStream(
    request: GenerateRequest,
    onChunk: (chunk: StreamChunk) => void
  ): Promise<GenerateResponse> {
    this.validateRequest(request);

    const prompt = this.formatMessages(request.messages);
    let fullContent = '';

    const response = await this.client.generateStream(
      {
        model: this.config.model,
        prompt,
        options: {
          temperature: request.options?.temperature ?? this.config.temperature
        }
      },
      (delta) => {
        fullContent += delta;
        onChunk({ delta, isComplete: false });
      }
    );

    onChunk({ delta: '', isComplete: true });

    return {
      content: response,
      finishReason: 'stop',
      usage: {
        promptTokens: 0,
        completionTokens: 0,
        totalTokens: 0
      },
      model: this.config.model
    };
  }

  getCapabilities(): AIProviderCapabilities {
    return {
      streaming: true,
      functionCalling: false,
      vision: false,
      jsonMode: true,
      maxContextLength: 8192
    };
  }

  async checkHealth(): Promise<boolean> {
    return this.client.checkHealth();
  }

  async listModels(): Promise<ModelInfo[]> {
    const models = await this.client.listModels();
    return models.map(m => ({
      id: m.name,
      name: m.name,
      contextLength: 8192
    }));
  }

  getType(): 'local' | 'cloud' {
    return 'local';
  }

  private formatMessages(messages: Message[]): string {
    return messages
      .map(m => {
        if (m.role === 'system') {
          return `System: ${m.content}`;
        }
        return m.content;
      })
      .join('\n\n');
  }
}
```
