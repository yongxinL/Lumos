## Technical Notes

### Service Implementation

```typescript
// src/main/services/input-handler.service.ts
import { EventEmitter } from 'events';
import { FastPathService } from './fast-path.service';
import { SwiftBridge } from '../bridges/swift-bridge';

export class InputHandlerService extends EventEmitter {
  private fastPath: FastPathService;
  private swiftBridge: SwiftBridge;
  private inputHistory: ProcessedInput[] = [];
  private rateLimiter: Map<string, number[]> = new Map();

  constructor(fastPath: FastPathService, swiftBridge: SwiftBridge) {
    super();
    this.fastPath = fastPath;
    this.swiftBridge = swiftBridge;
  }

  async submitText(text: string): Promise<ProcessedInput> {
    // Validate and sanitize
    const sanitized = this.sanitizeText(text);
    if (!this.validateInput(sanitized)) {
      throw new InputValidationError('Invalid input');
    }

    // Rate limiting
    if (!this.checkRateLimit('text')) {
      throw new RateLimitError('Too many requests');
    }

    // Create processed input
    const processed: ProcessedInput = {
      id: crypto.randomUUID(),
      text: sanitized,
      source: 'text',
      timestamp: new Date().toISOString(),
      metadata: {
        length: sanitized.length,
        word_count: sanitized.split(/\s+/).length,
      },
    };

    // Classify and route
    const classification = this.fastPath.classify(processed);
    processed.classification = classification;

    // Track history
    this.inputHistory.push(processed);
    this.trimHistory();

    // Emit event
    this.emit('input:processed', processed);

    return processed;
  }

  async startRecording(): Promise<void> {
    // Delegate to Swift bridge
    await this.swiftBridge.startRecording();
    this.emit('recording:started');
  }

  async stopRecording(): Promise<TranscriptionPreview> {
    // Delegate to Swift bridge
    const preview = await this.swiftBridge.stopRecording();
    this.emit('recording:stopped', preview);
    return preview;
  }

  async confirmTranscription(preview: TranscriptionPreview): Promise<ProcessedInput> {
    // Process as text input
    return this.submitText(preview.text);
  }

  cancelTranscription(): void {
    this.swiftBridge.cancelRecording();
    this.emit('recording:cancelled');
  }

  private sanitizeText(text: string): string {
    // Remove excessive whitespace
    let sanitized = text.trim().replace(/\s+/g, ' ');

    // Remove control characters
    sanitized = sanitized.replace(/[\x00-\x1F\x7F]/g, '');

    // Limit length
    if (sanitized.length > 10000) {
      sanitized = sanitized.substring(0, 10000);
    }

    return sanitized;
  }

  private validateInput(text: string): boolean {
    // Minimum length
    if (text.length < 1) return false;

    // Maximum length
    if (text.length > 10000) return false;

    // Must contain at least one alphanumeric character
    if (!/[a-zA-Z0-9]/.test(text)) return false;

    return true;
  }

  private checkRateLimit(source: string): boolean {
    const now = Date.now();
    const window = 60000; // 1 minute
    const limit = 10; // 10 requests per minute

    // Get recent timestamps
    const timestamps = this.rateLimiter.get(source) || [];
    const recent = timestamps.filter((t) => now - t < window);

    // Check limit
    if (recent.length >= limit) {
      return false;
    }

    // Add current timestamp
    recent.push(now);
    this.rateLimiter.set(source, recent);

    return true;
  }

  private trimHistory(): void {
    // Keep last 100 inputs
    if (this.inputHistory.length > 100) {
      this.inputHistory = this.inputHistory.slice(-100);
    }
  }

  getRecentInputs(count: number = 10): ProcessedInput[] {
    return this.inputHistory.slice(-count);
  }

  clearHistory(): void {
    this.inputHistory = [];
  }
}

// Custom errors
export class InputValidationError extends Error {
  constructor(message: string) {
    super(message);
    this.name = 'InputValidationError';
  }
}

export class RateLimitError extends Error {
  constructor(message: string) {
    super(message);
    this.name = 'RateLimitError';
  }
}
```

### Types

```typescript
export interface ProcessedInput {
  id: string;
  text: string;
  source: 'text' | 'audio';
  timestamp: string;
  classification?: FastPathResult;
  metadata: {
    length: number;
    word_count: number;
    audio_duration_ms?: number;
    confidence?: number;
  };
}

export interface TranscriptionPreview {
  text: string;
  confidence: number;
  duration_ms: number;
  language: string;
}

export interface RecordingState {
  isRecording: boolean;
  duration_ms: number;
}
```

### IPC Handler Registration

```typescript
// src/main/ipc/input.handlers.ts
import { InputHandlerService } from '../services/input-handler.service';

export function registerInputHandlers(service: InputHandlerService): void {
  registerHandler('input:submit-text', async (text: string) => {
    return service.submitText(text);
  });

  registerHandler('input:start-recording', async () => {
    return service.startRecording();
  });

  registerHandler('input:stop-recording', async () => {
    return service.stopRecording();
  });

  registerHandler('input:confirm-transcription', async (preview: TranscriptionPreview) => {
    return service.confirmTranscription(preview);
  });

  registerHandler('input:cancel-transcription', async () => {
    service.cancelTranscription();
  });

  registerHandler('input:get-recent', async (count: number) => {
    return service.getRecentInputs(count);
  });

  // Forward events to renderer
  service.on('transcription:update', (partial: string) => {
    mainWindow.webContents.send('transcription:update', partial);
  });

  service.on('recording:started', () => {
    mainWindow.webContents.send('recording:started');
  });

  service.on('recording:stopped', (preview: TranscriptionPreview) => {
    mainWindow.webContents.send('recording:stopped', preview);
  });
}
```
