### Enhanced Swift Bridge

```typescript
// src/main/bridges/swift-bridge.ts
import { spawn, ChildProcess } from 'child_process';
import { EventEmitter } from 'events';

interface BridgeConfig {
  maxRestarts: number;
  restartDelay: number;
  healthCheckInterval: number;
  requestTimeout: number;
}

interface BridgeMetrics {
  totalRequests: number;
  successfulRequests: number;
  failedRequests: number;
  avgLatencyMs: number;
  uptime: number;
  restartCount: number;
}

export class SwiftBridge extends EventEmitter {
  private process: ChildProcess | null = null;
  private requestCallbacks: Map<string, RequestCallback> = new Map();
  private requestQueue: QueuedRequest[] = [];
  private isReady = false;
  private restartCount = 0;
  private metrics: BridgeMetrics;
  private config: BridgeConfig;
  private healthCheckTimer?: NodeJS.Timeout;
  private startTime: number;

  constructor(config?: Partial<BridgeConfig>) {
    super();

    this.config = {
      maxRestarts: 5,
      restartDelay: 1000,
      healthCheckInterval: 30000,
      requestTimeout: 30000,
      ...config
    };

    this.metrics = {
      totalRequests: 0,
      successfulRequests: 0,
      failedRequests: 0,
      avgLatencyMs: 0,
      uptime: 0,
      restartCount: 0
    };

    this.startTime = Date.now();
  }

  async start(): Promise<void> {
    try {
      await this.spawnProcess();
      await this.waitForReady();
      this.startHealthCheck();
      this.processQueue();
    } catch (error) {
      console.error('Failed to start Swift bridge:', error);
      throw error;
    }
  }

  async stop(): Promise<void> {
    this.stopHealthCheck();

    if (this.process) {
      // Send graceful shutdown
      try {
        await this.sendRequest('shutdown', {}, 5000);
      } catch {
        // Force kill if graceful shutdown fails
        this.process.kill('SIGKILL');
      }

      this.process = null;
      this.isReady = false;
    }
  }

  async sendRequest(method: string, params: any = {}, timeout?: number): Promise<any> {
    // Queue if not ready
    if (!this.isReady) {
      return new Promise((resolve, reject) => {
        this.requestQueue.push({ method, params, resolve, reject });
      });
    }

    return new Promise((resolve, reject) => {
      const id = crypto.randomUUID();
      const startTime = Date.now();

      const message = {
        id,
        type: 'request',
        method,
        params
      };

      // Store callback with timeout
      const timeoutId = setTimeout(() => {
        if (this.requestCallbacks.has(id)) {
          this.requestCallbacks.delete(id);
          this.recordFailure(Date.now() - startTime);
          reject(new Error(`Request timeout: ${method}`));
        }
      }, timeout || this.config.requestTimeout);

      this.requestCallbacks.set(id, {
        resolve: (result) => {
          clearTimeout(timeoutId);
          this.recordSuccess(Date.now() - startTime);
          resolve(result);
        },
        reject: (error) => {
          clearTimeout(timeoutId);
          this.recordFailure(Date.now() - startTime);
          reject(error);
        }
      });

      // Send to Swift
      try {
        this.process?.stdin?.write(JSON.stringify(message) + '\n');
        this.metrics.totalRequests++;
      } catch (error) {
        this.requestCallbacks.delete(id);
        clearTimeout(timeoutId);
        reject(error);
      }
    });
  }

  // Audio methods
  async startRecording(): Promise<void> {
    await this.sendRequest('audio:start-recording');
  }

  async stopRecording(): Promise<TranscriptionPreview> {
    const result = await this.sendRequest('audio:stop-recording');
    return result as TranscriptionPreview;
  }

  cancelRecording(): void {
    this.sendRequest('audio:cancel-recording').catch(console.error);
  }

  onTranscriptionUpdate(callback: (partial: string) => void): void {
    this.on('transcription:update', (params) => {
      callback(params.partial);
    });
  }

  async healthCheck(): Promise<boolean> {
    try {
      const result = await this.sendRequest('health', {}, 5000);
      return result?.status === 'healthy';
    } catch {
      return false;
    }
  }

  getMetrics(): BridgeMetrics {
    return {
      ...this.metrics,
      uptime: Date.now() - this.startTime
    };
  }

  // Private methods

  private async spawnProcess(): Promise<void> {
    const executablePath = this.getExecutablePath();

    this.process = spawn(executablePath, [], {
      stdio: ['pipe', 'pipe', 'pipe']
    });

    // Handle stdout (messages from Swift)
    this.process.stdout?.on('data', (data) => {
      const lines = data.toString().split('\n').filter((l: string) => l.trim());
      for (const line of lines) {
        this.handleMessage(line);
      }
    });

    // Handle stderr (errors from Swift)
    this.process.stderr?.on('data', (data) => {
      console.error('[Swift Helper Error]', data.toString());
    });

    // Handle process exit
    this.process.on('exit', (code, signal) => {
      console.log(`Swift helper exited: code=${code}, signal=${signal}`);
      this.process = null;
      this.isReady = false;

      // Attempt restart
      if (this.restartCount < this.config.maxRestarts) {
        this.restartCount++;
        this.metrics.restartCount++;

        setTimeout(() => {
          console.log(`Restarting Swift helper (attempt ${this.restartCount})`);
          this.start().catch(console.error);
        }, this.config.restartDelay);
      } else {
        console.error('Swift helper exceeded max restart attempts');
        this.emit('fatal-error', new Error('Swift bridge failed'));
      }
    });
  }

  private getExecutablePath(): string {
    if (process.env.NODE_ENV === 'development') {
      return path.join(__dirname, '../../swift/.build/debug/lumos-helper');
    } else {
      return path.join(process.resourcesPath, 'bin/lumos-helper');
    }
  }

  private handleMessage(line: string): void {
    try {
      const message = JSON.parse(line);

      if (message.type === 'response') {
        const callback = this.requestCallbacks.get(message.id);
        if (callback) {
          callback.resolve(message.result);
          this.requestCallbacks.delete(message.id);
        }
      } else if (message.type === 'event') {
        if (message.method === 'ready') {
          this.isReady = true;
          this.emit('ready');
        } else {
          this.emit(message.method, message.params);
        }
      } else if (message.type === 'error') {
        const callback = this.requestCallbacks.get(message.id);
        if (callback) {
          callback.reject(new Error(message.error?.message || 'Unknown error'));
          this.requestCallbacks.delete(message.id);
        }
      }
    } catch (error) {
      console.error('Failed to parse message from Swift:', error);
    }
  }

  private async waitForReady(timeout: number = 10000): Promise<void> {
    return new Promise((resolve, reject) => {
      if (this.isReady) {
        resolve();
        return;
      }

      const timer = setTimeout(() => {
        reject(new Error('Swift bridge ready timeout'));
      }, timeout);

      this.once('ready', () => {
        clearTimeout(timer);
        resolve();
      });
    });
  }

  private processQueue(): void {
    while (this.requestQueue.length > 0 && this.isReady) {
      const request = this.requestQueue.shift();
      if (request) {
        this.sendRequest(request.method, request.params)
          .then(request.resolve)
          .catch(request.reject);
      }
    }
  }

  private startHealthCheck(): void {
    this.healthCheckTimer = setInterval(async () => {
      const healthy = await this.healthCheck();
      if (!healthy) {
        console.error('Swift bridge health check failed');
        this.emit('health-check-failed');

        // Force restart
        if (this.process) {
          this.process.kill('SIGTERM');
        }
      }
    }, this.config.healthCheckInterval);
  }

  private stopHealthCheck(): void {
    if (this.healthCheckTimer) {
      clearInterval(this.healthCheckTimer);
      this.healthCheckTimer = undefined;
    }
  }

  private recordSuccess(latencyMs: number): void {
    this.metrics.successfulRequests++;
    this.updateAvgLatency(latencyMs);
  }

  private recordFailure(latencyMs: number): void {
    this.metrics.failedRequests++;
    this.updateAvgLatency(latencyMs);
  }

  private updateAvgLatency(latencyMs: number): void {
    const total = this.metrics.successfulRequests + this.metrics.failedRequests;
    this.metrics.avgLatencyMs =
      (this.metrics.avgLatencyMs * (total - 1) + latencyMs) / total;
  }
}

// Types
interface RequestCallback {
  resolve: (result: any) => void;
  reject: (error: Error) => void;
}

interface QueuedRequest {
  method: string;
  params: any;
  resolve: (result: any) => void;
  reject: (error: Error) => void;
}
```

## Integration Tests

```typescript
// src/main/bridges/__tests__/swift-bridge.test.ts
describe('SwiftBridge', () => {
  let bridge: SwiftBridge;

  beforeEach(async () => {
    bridge = new SwiftBridge();
    await bridge.start();
  });

  afterEach(async () => {
    await bridge.stop();
  });

  it('should start and become ready', async () => {
    expect(bridge.getMetrics().uptime).toBeGreaterThan(0);
  });

  it('should handle ping request', async () => {
    const result = await bridge.sendRequest('ping');
    expect(result.status).toBe('ok');
  });

  it('should handle health check', async () => {
    const healthy = await bridge.healthCheck();
    expect(healthy).toBe(true);
  });

  it('should queue requests before ready', async () => {
    const newBridge = new SwiftBridge();
    const promise = newBridge.sendRequest('ping');
    await newBridge.start();
    const result = await promise;
    expect(result.status).toBe('ok');
    await newBridge.stop();
  });

  it('should timeout long requests', async () => {
    await expect(
      bridge.sendRequest('slow-operation', {}, 100)
    ).rejects.toThrow('timeout');
  });

  it('should restart on crash', async () => {
    // Simulate crash
    bridge['process']?.kill('SIGKILL');

    // Wait for restart
    await new Promise(resolve => setTimeout(resolve, 2000));

    // Should be healthy again
    const healthy = await bridge.healthCheck();
    expect(healthy).toBe(true);
  });
});
```
