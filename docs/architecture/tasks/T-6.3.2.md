## Technical Notes

```typescript
// src/renderer/hooks/useKeyboardShortcuts.ts
export function useKeyboardShortcuts() {
  useEffect(() => {
    const handleKeyDown = (e: KeyboardEvent) => {
      // Command palette
      if ((e.metaKey || e.ctrlKey) && e.key === 'k') {
        e.preventDefault();
        // Open command palette
      }

      // Navigation
      if ((e.metaKey || e.ctrlKey) && e.key >= '1' && e.key <= '5') {
        e.preventDefault();
        const routes = ['/chat', '/activity', '/permissions', '/skills', '/settings'];
        navigate(routes[parseInt(e.key) - 1]);
      }

      // Help
      if ((e.metaKey || e.ctrlKey) && e.key === '/') {
        e.preventDefault();
        // Show shortcuts modal
      }
    };

    window.addEventListener('keydown', handleKeyDown);
    return () => window.removeEventListener('keydown', handleKeyDown);
  }, []);
}

// Shortcuts configuration
export const SHORTCUTS = {
  commandPalette: { key: 'k', modifiers: ['meta'] },
  navigation: {
    chat: { key: '1', modifiers: ['meta'] },
    activity: { key: '2', modifiers: ['meta'] },
    permissions: { key: '3', modifiers: ['meta'] },
    skills: { key: '4', modifiers: ['meta'] },
    settings: { key: '5', modifiers: ['meta'] },
  },
  actions: {
    submit: { key: 'Enter' },
    cancel: { key: 'Escape' },
  },
};
```
