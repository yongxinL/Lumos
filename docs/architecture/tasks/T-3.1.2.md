## Objective

Implement the Evaluation LLM Service that uses Ollama with constrained JSON decoding to generate structured ActionProposal objects from user input, ensuring fast, deterministic, and local-only operation.

- [ ] **AC-3.1.2.4**: 2-3 second generation time target
- [ ] **AC-3.1.2.5**: Fallback retry logic for malformed JSON
- [ ] **AC-3.1.2.6**: Confidence scoring based on model output
- [ ] **AC-3.1.2.7**: Context window management for long inputs

### Service Implementation

```typescript
// src/main/services/ai/evaluation-llm.service.ts
import { OllamaClient } from './ollama-client';
import Ajv from 'ajv';
import { ACTION_PROPOSAL_SCHEMA } from '../../types/governance/proposal';

interface EvaluationLLMConfig {
  model: string;
  temperature: number;
  maxRetries: number;
  timeoutMs: number;
}

export class EvaluationLLMService {
  private ollama: OllamaClient;
  private config: EvaluationLLMConfig;
  private validator: Ajv;
  private validateProposal: any;

  constructor(ollama: OllamaClient, config?: Partial<EvaluationLLMConfig>) {
    this.ollama = ollama;
    this.config = {
      model: 'qwen3-vl-4b',
      temperature: 0.1,
      maxRetries: 3,
      timeoutMs: 30000,
      ...config,
    };

    this.validator = new Ajv({ allErrors: true });
    this.validateProposal = this.validator.compile(ACTION_PROPOSAL_SCHEMA);
  }

  async initialize(): Promise<void> {
    // Check if model is available
    const models = await this.ollama.listModels();
    const hasModel = models.some((m) => m.name.includes(this.config.model));

    if (!hasModel) {
      throw new Error(`Model ${this.config.model} not found. Please pull it first.`);
    }

    // Verify model works
    const healthy = await this.ollama.checkHealth();
    if (!healthy) {
      throw new Error('Ollama service not available');
    }
  }

  async generateProposal(input: ProcessedInput): Promise<ActionProposal> {
    const startTime = Date.now();

    // Build prompt
    const prompt = this.buildPrompt(input);

    // Generate with retries
    for (let attempt = 1; attempt <= this.config.maxRetries; attempt++) {
      try {
        const response = await this.ollama.generate({
          model: this.config.model,
          prompt,
          format: ACTION_PROPOSAL_SCHEMA,
          options: {
            temperature: this.config.temperature,
            top_p: 0.9,
          },
        });

        // Parse and validate
        const proposal = this.parseAndValidate(response, input);
        const duration = Date.now() - startTime;

        console.log(`Proposal generated in ${duration}ms (attempt ${attempt})`);
        return proposal;
      } catch (error) {
        console.error(`Proposal generation attempt ${attempt} failed:`, error);

        if (attempt === this.config.maxRetries) {
          throw new EvaluationError('Failed to generate valid proposal after retries', error);
        }

        // Wait before retry
        await this.delay(1000 * attempt);
      }
    }

    throw new EvaluationError('Max retries exceeded');
  }

  async isAvailable(): Promise<boolean> {
    return this.ollama.checkHealth();
  }

  getModelInfo(): ModelInfo {
    return {
      name: this.config.model,
      type: 'local',
      provider: 'ollama',
      capabilities: ['json_mode', 'low_latency'],
    };
  }

  private buildPrompt(input: ProcessedInput): string {
    return `You are an AI assistant that analyzes user requests and generates structured action proposals.

User Input: "${input.text}"

Classification: ${input.classification?.classification || 'action'}

Your task:
1. Analyze the user's intent
2. Determine the operation type (e.g., create_ticket, schedule_meeting, etc.)
3. Identify the target entity if applicable
4. Assess data domain (enterprise/personal) and classification
5. Determine reversibility type
6. Assign confidence score (0.0-1.0)
7. Determine if confirmation is required
8. Assess risk level (low/medium/high)

Available operation types:
- create_ticket, update_ticket_status, add_ticket_comment
- create_calendar_event, update_calendar_event, delete_calendar_event
- create_note, update_note, delete_note
- read_file, write_file

Guidelines:
- Data domain: "enterprise" for work-related, "personal" for individual tasks
- Data classification: "public" (no sensitive data), "internal" (company info), "confidential" (restricted access), "restricted" (highly sensitive)
- Reversibility: "FULL" (easily undone), "PARTIAL" (some parts undoable), "COMPENSATABLE" (can be compensated), "IRREVERSIBLE" (cannot undo)
- Confidence: Based on clarity of intent (0.9+ for clear, 0.5-0.8 for ambiguous, <0.5 for unclear)
- Requires confirmation: true for write/delete operations, false for reads
- Risk level: "low" (reads, views), "medium" (updates, creates), "high" (deletes, bulk operations)

Generate a valid ActionProposal JSON object.`;
  }

  private parseAndValidate(response: string, input: ProcessedInput): ActionProposal {
    // Parse JSON
    let parsed: any;
    try {
      parsed = JSON.parse(response);
    } catch (error) {
      throw new ValidationError('Invalid JSON response', error);
    }

    // Add required fields not in LLM output
    parsed.id = crypto.randomUUID();
    parsed.timestamp = new Date().toISOString();
    parsed.evaluation_model = this.config.model;
    parsed.raw_user_input = input.text;
    parsed.related_skills = []; // Populated by policy engine
    parsed.rollback_plan = null; // Generated by expert AI if needed

    // Validate against schema
    const valid = this.validateProposal(parsed);
    if (!valid) {
      const errors = this.validator.errorsText(this.validateProposal.errors);
      throw new ValidationError(`Schema validation failed: ${errors}`);
    }

    // Additional business logic validation
    this.validateBusinessRules(parsed);

    return parsed as ActionProposal;
  }

  private validateBusinessRules(proposal: any): void {
    // Confidence must be reasonable
    if (proposal.confidence < 0 || proposal.confidence > 1) {
      throw new ValidationError('Confidence out of range');
    }

    // High-risk operations must require confirmation
    if (proposal.risk_level === 'high' && !proposal.requires_confirmation) {
      proposal.requires_confirmation = true;
    }

    // Delete operations should be FULL or IRREVERSIBLE
    if (proposal.operation.includes('delete')) {
      if (!['FULL', 'IRREVERSIBLE'].includes(proposal.reversibility)) {
        proposal.reversibility = 'IRREVERSIBLE';
      }
    }

    // Read operations should be low risk
    if (proposal.operation.includes('read') || proposal.operation.includes('view')) {
      proposal.risk_level = 'low';
      proposal.requires_confirmation = false;
    }
  }

  private delay(ms: number): Promise<void> {
    return new Promise((resolve) => setTimeout(resolve, ms));
  }
}

// Errors
export class EvaluationError extends Error {
  constructor(
    message: string,
    public cause?: any
  ) {
    super(message);
    this.name = 'EvaluationError';
  }
}

export class ValidationError extends Error {
  constructor(
    message: string,
    public cause?: any
  ) {
    super(message);
    this.name = 'ValidationError';
  }
}

// Types
export interface ModelInfo {
  name: string;
  type: 'local' | 'cloud';
  provider: string;
  capabilities: string[];
}
```
